vibefpmgui.test_monitor

Live table of the tests fpm test runs, with an estimated progress bar.

The GUI first runs fpm test --list to learn the test names, then runs the tests with --runner pointing at vibefpmgui.test_runner, which prints a marker before and after each test (fpm runs them one at a time)::

@@vibefpm:start:check_fast@@
@@vibefpm:end:check_fast:0:0.012@@

How long each test took is remembered per project, so later runs can weight the progress bar by expected duration and estimate the time left.

 57MARKER_RE = re.compile(r"@@vibefpm:(start|end):([^:@]+)(?::(-?\d+):([\d.]+))?@@")
 58# A test's command line, from "test_runner.py --collect" (for a parallel run).
 59EXE_RE = re.compile(r"@@vibefpm:exe:([^:@]+)@@(.*)$")
 60# "fpm test --list" prints " Matched names:" followed by the names.
 61MATCHED_RE = re.compile(r"^\s*Matched names:\s*(.*)$")
 62NO_TESTS_RE = re.compile(r"No tests to run")

 64QUEUED, RUNNING, PASSED, FAILED_T, NOT_RUN = "Queued", "Running", "Passed", "Failed", "Not run"
 65STATUSES = (RUNNING, PASSED, FAILED_T, NOT_RUN, QUEUED)
 66ICONS = {QUEUED: "·", RUNNING: "⟳", PASSED: "✓", FAILED_T: "✕", NOT_RUN: "–"}
 67# Test statuses borrow the build palette's colors.
 68COLOR_KEY = {RUNNING: COMPILING, PASSED: DONE, FAILED_T: FAILED, NOT_RUN: ABORTED, QUEUED: ABORTED}

 72def parse_test_list(text: str) -> list[str] | None:
 73    """Test names from ``fpm test --list`` output ([] if there are none,
 74    None if the output wasn't understood)."""
 75    if NO_TESTS_RE.search(text):
 76        return []
 77    lines = text.splitlines()
 78    for i, line in enumerate(lines):
 79        match = MATCHED_RE.match(line)
 80        if match:
 81            names = match.group(1).split()
 82            for rest in lines[i + 1:]:
 83                if not rest.strip() or rest.lstrip().startswith(("<", "STOP")):
 84                    break
 85                names += rest.split()
 86            return names
 87    return None

 90@dataclass
 91class TestCase:
 92    name: str
 93    row: int
 94    expected: float | None = None  # duration last time, if known
 95    seq: int = 0  # original (listed) order, for "unsorted"
 96    status: str = QUEUED
 97    started: float | None = None
 98    finished: float | None = None
 99    seconds: float | None = None  # as measured by the runner
100    exit_code: int | None = None
101    stopped: bool = False  # was running when Stop was pressed
102    output: list[str] = field(default_factory=list)
103    partial: str = ""  # (a parallel run: the start of a line still being written)
104
105    def elapsed(self) -> float:
106        if self.seconds is not None:
107            return self.seconds
108        if self.started is None:
109            return 0.0
110        return (self.finished or time.monotonic()) - self.started

121    def paint(self, p: QPainter, option, index):
122        mon = self.monitor
123        if index.row() >= len(mon.tests):
124            return

133        elif index.row() == mon.hover_row:
134            p.fillRect(r, QColor(pal["hover"]))
135        if test.finished is not None and time.monotonic() - test.finished < FLASH_SECONDS:
136            age = time.monotonic() - test.finished
137            p.fillRect(r, _tint(color, int(35 * (1 - age / FLASH_SECONDS))))
138        if test.name == mon.selected and index.column() == 0:
139            p.fillRect(QRectF(r.left(), r.top() + 6, 3, r.height() - 12), QColor(pal["accent1"]))
140        p.setPen(QPen(QColor(pal["line"]), 1))
141        p.drawLine(r.bottomLeft(), r.bottomRight())
142
143        small = QFont(option.font)
144        small.setPointSizeF(max(option.font.pointSizeF() - 2, 7))
145        if index.column() == 0:
146            self._paint_name(p, r, test, option.font, small, pal)
147        elif index.column() == 1:
148            self._paint_status(p, r, test, small, color)
149        else:

153    def _paint_name(self, p, r, test, font, small, pal):
154        bold = QFont(font)
155        bold.setWeight(QFont.DemiBold)
156        detail = {RUNNING: "running…", NOT_RUN: "not run"}.get(test.status, "")
157        if test.stopped:
158            detail = f"stopped after {test.elapsed():.1f} s"
159        if test.status == FAILED_T and test.exit_code is not None:
160            detail = f"exit code {test.exit_code}"
161        lines = len(test.output)
162        if lines and test.status != RUNNING:
163            count = f"{lines} line{'s' * (lines != 1)} of output"
164            detail = f"{detail} · {count}" if detail else count
165        paint_name_and_detail(p, r.adjusted(14, 0, -6, 0), test.name, detail, bold, small,
166                              QColor(pal["text"] if test.status != QUEUED else pal["muted"]), QColor(pal["muted"]))

168    def _paint_status(self, p, r, test, small, color):
169        font = QFont(small)
170        font.setWeight(QFont.DemiBold)
171        text = "Stopped" if test.stopped else self.monitor.word(test.status)
172        pill = QRectF(r.left() + 8, r.center().y() - 10,
173                      QFontMetricsF(font).horizontalAdvance(text) + 36, 20)
174        p.setPen(Qt.NoPen)
175        p.setBrush(_tint(color, 40 if test.status != QUEUED else 22))
176        p.drawRoundedRect(pill, 10, 10)
177        icon = QRectF(pill.left() + 8, pill.center().y() - 6, 12, 12)
178        if test.status == RUNNING:
179            p.setPen(QPen(_tint(color, 70), 2))
180            p.drawEllipse(icon)
181            p.setPen(QPen(color, 2, Qt.SolidLine, Qt.RoundCap))
182            p.drawArc(icon, int(-self.monitor.spinner_angle * 16), -110 * 16)
183        else:
184            p.setFont(font)
185            p.setPen(color)
186            p.drawText(icon.adjusted(-3, -3, 3, 3), Qt.AlignCenter, ICONS[test.status])
187        p.setFont(font)
188        p.setPen(color.darker(115) if self.monitor.is_light else color)
189        p.drawText(pill.adjusted(27, 0, -6, 0), Qt.AlignVCenter | Qt.AlignLeft, text)

191    def _paint_time(self, p, r, test, small, pal, color):
192        scale = self.monitor.time_scale()
193        track = QRectF(r.left() + 10, r.center().y() - 3, max(r.width() - 74, 10), 6)
194        p.setPen(Qt.NoPen)
195        p.setBrush(QColor(pal["track"]))
196        p.drawRoundedRect(track, 3, 3)
197        mono = QFontDatabase.systemFont(QFontDatabase.FixedFont)
198        mono.setPointSizeF(small.pointSizeF())
199        p.setFont(mono)
200        if test.status == QUEUED:
201            if test.expected is not None and scale > 0:  # ghost bar: last run's time
202                p.setBrush(_tint(QColor(pal["muted"]), 70))
203                p.drawRoundedRect(QRectF(track.left(), track.top(),
204                                         max(track.width() * test.expected / scale, 3), 6), 3, 3)
205                p.setPen(QColor(pal["muted"]))
206                p.drawText(r.adjusted(0, 0, -12, 0), Qt.AlignVCenter | Qt.AlignRight,
207                           f"~{test.expected:.1f} s")
208            return
209        elapsed = test.elapsed()
210        if test.status == RUNNING and test.expected:  # where it finished last time
211            x = track.left() + track.width() * min(test.expected / scale, 1)
212            p.setPen(QPen(QColor(pal["muted"]), 1, Qt.DotLine))
213            p.drawLine(x, track.top() - 3, x, track.bottom() + 3)
214            p.setPen(Qt.NoPen)
215        p.setBrush(color if test.status == RUNNING else QColor(pal["muted"]))
216        fraction = elapsed / scale if scale > 0 else 0
217        p.drawRoundedRect(QRectF(track.left(), track.top(), max(track.width() * min(fraction, 1), 3), 6), 3, 3)
218        p.setPen(QColor(pal["text"] if test.status == RUNNING else pal["muted"]))
219        p.drawText(r.adjusted(0, 0, -12, 0), Qt.AlignVCenter | Qt.AlignRight, f"{elapsed:.1f} s")

427    def finish(self, exit_code: int, stopped: bool = False, no_tests: bool = False):
428        if not self._active:
429            return
430        build_failed = self._result == "building" and exit_code != 0 and not stopped
431        self._active = False
432        self._finished_at = time.monotonic()
433        now = self._finished_at
434        for test in self.tests:
435            if test.status == RUNNING:
436                test.status = FAILED_T if not stopped else NOT_RUN
437                test.stopped = stopped
438                test.finished = now
439            elif test.status == QUEUED:
440                test.status = NOT_RUN
441        if no_tests:
442            self._result = "none"
443            self._show_empty(self.word("none"))
444        elif stopped:
445            self._result = "stopped"
446        elif exit_code != 0 or any(t.status == FAILED_T for t in self.tests):
447            self._result = "failed"
448        else:
449            self._result = "ok"
450        if build_failed:
451            self._result = "buildfailed"
452        if self._result == "failed" and not self.tests:
453            self._show_empty(self.word("unlisted"))
454        state = {"ok": "ok", "failed": "failed", "stopped": "stopped", "none": "ok",
455                 "buildfailed": "failed"}[self._result]
456        self.progress.set(1.0 if self._result in ("ok", "none") else self.fraction(), state)
457        self.update_header()
458        self.table.viewport().update()
459        self.schedule_sort()

476    # ------------------------------------------------------------- parsing
477    def feed_line(self, line: str) -> str | None:
478        """Process one line of ``fpm test`` output. Returns the text to show
479        in the console (markers removed), or None to drop the line."""
480        match = MARKER_RE.search(line)
481        if not match:
482            if self.current is not None:
483                self.current.output.append(line)
484            return line
485        before = line[:match.start()]
486        if before.strip() and self.current is not None:
487            self.current.output.append(before)
488        kind, name = match.group(1), match.group(2)
489        if kind == "start":
490            self.current = self.test_started(name)
491        else:
492            self.current = None
493            self.test_finished(name, int(match.group(3)), float(match.group(4)))
494        return before if before.strip() else None

571    def _expected(self) -> dict[str, float]:
572        """Expected duration for every test, or {} when nothing is known.
573
574        Tests without a recorded time get the median of the known ones, and a
575        running test that has already taken longer than expected is assumed
576        to need a bit longer still, so the bar never sits at ~100%."""
577        known = [t.expected for t in self.tests if t.expected is not None]
578        if not known:
579            return {}
580        guess = statistics.median(known)
581        expected = {}
582        for t in self.tests:
583            e = t.expected if t.expected is not None else guess
584            if t.status == RUNNING:
585                e = max(e, t.elapsed() * 1.25)
586            expected[t.name] = e
587        return expected

589    def fraction(self) -> float:
590        if not self.tests:
591            return 0.0
592        expected = self._expected()
593        if not expected:  # nothing known yet: count tests
594            done = sum(t.status in (PASSED, FAILED_T, NOT_RUN) for t in self.tests)
595            return done / len(self.tests)
596        total = sum(max(v, 1e-3) for v in expected.values())
597        progress = 0.0
598        for t in self.tests:
599            e = max(expected[t.name], 1e-3)
600            if t.status in (PASSED, FAILED_T, NOT_RUN):
601                progress += e
602            elif t.status == RUNNING:
603                progress += min(t.elapsed(), 0.95 * e)
604        return progress / total

606    def remaining(self) -> float | None:
607        """Estimated time left, only when every test has a recorded time."""
608        if any(t.expected is None for t in self.tests):
609            return None
610        expected = self._expected()
611        if not expected:
612            return None
613        left = 0.0
614        for t in self.tests:

616                left += expected[t.name]
617            elif t.status == RUNNING:

572    def _expected(self) -> dict[str, float]:
573        """Expected duration for every test, or {} when nothing is known.
574
575        Tests without a recorded time get the median of the known ones, and a
576        running test that has already taken longer than expected is assumed
577        to need a bit longer still, so the bar never sits at ~100%."""
578        known = [t.expected for t in self.tests if t.expected is not None]
579        if not known:
580            return {}
581        guess = statistics.median(known)
582        expected = {}
583        for t in self.tests:
584            e = t.expected if t.expected is not None else guess
585            if t.status == RUNNING:
586                e = max(e, t.elapsed() * 1.25)
587            expected[t.name] = e
588        return expected

590    def fraction(self) -> float:
591        if not self.tests:
592            return 0.0
593        expected = self._expected()
594        if not expected:  # nothing known yet: count tests
595            done = sum(t.status in (PASSED, FAILED_T, NOT_RUN) for t in self.tests)
596            return done / len(self.tests)
597        total = sum(max(v, 1e-3) for v in expected.values())
598        progress = 0.0
599        for t in self.tests:
600            e = max(expected[t.name], 1e-3)
601            if t.status in (PASSED, FAILED_T, NOT_RUN):
602                progress += e
603            elif t.status == RUNNING:
604                progress += min(t.elapsed(), 0.95 * e)
605        return progress / total

607    def remaining(self) -> float | None:
608        """Estimated time left, only when every test has a recorded time."""
609        if any(t.expected is None for t in self.tests):
610            return None
611        expected = self._expected()
612        if not expected:
613            return None
614        left = 0.0
615        for t in self.tests:
616            if t.status == QUEUED:
617                left += expected[t.name]
618            elif t.status == RUNNING:
619                left += max(expected[t.name] - t.elapsed(), 0.0)
620        return left / min(self.jobs, max(1, sum(t.status in (QUEUED, RUNNING) for t in self.tests)))
