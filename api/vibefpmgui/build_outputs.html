vibefpmgui.build_outputs

What fpm has left in a project's build/ folder, and a tab to show it.

fpm splits one build over several folders named <compiler>_<hash>, where the hash covers the settings that affect that step: one folder for compiled files and modules, one for the library archive, one for linked programs. A build with other settings (profile, flags, features) adds another set and old sets are never removed.

The hashes can't be turned back into settings, so the GUI records, for every build it runs, which folders that build created or changed together with the settings used (ConfigRecords). Folders it has no record for (built before this feature, or by fpm on the command line) are grouped by when they were last written.

build/compile_commands.json (the exact compiler command for each source file of the most recent build) is read by read_compile_commands.

113def build_folders(project_dir: Path) -> list[Path]:
114    build = project_dir / "build"

120def _is_program(path: Path) -> bool:
121    if os.name == "nt":
122        return path.suffix.lower() == ".exe"
123    return os.access(path, os.X_OK) and not path.suffix

126def scan_folder(path: Path) -> BuildFolder:
127    match = BUILD_DIR_RE.match(path.name)
128    folder = BuildFolder(path, match.group("compiler") if match else "?")
129    for dirpath, _dirs, files in os.walk(path):
130        rel_top = Path(dirpath).relative_to(path).parts[:1]
131        for name in files:
132            file = Path(dirpath) / name
133            try:
134                stat = file.stat()
135            except OSError:
136                continue
137            folder.size += stat.st_size
138            folder.newest = max(folder.newest, stat.st_mtime)
139            suffix = file.suffix.lower()
140            if suffix in (".mod", ".smod"):
141                folder.modules += 1
142            elif suffix in (".o", ".obj"):
143                folder.objects += 1
144            elif suffix in (".a", ".lib") or (suffix in (".so", ".dylib", ".dll") and name.startswith("lib")):
145                folder.libraries.append(file)
146            elif rel_top and rel_top[0] in PROGRAM_KINDS and _is_program(file):
147                folder.programs.append((PROGRAM_KINDS[rel_top[0]], file))
148    return folder

161def read_compile_commands(project_dir: Path) -> tuple[dict[str, list[str]], str | None]:
162    """{resolved source path: argument list} from build/compile_commands.json,
163    plus the name of the build folder the commands compile into."""
164    path = project_dir / "build" / "compile_commands.json"
165    try:
166        entries = json.loads(path.read_text())
167    except (OSError, ValueError):
168        return {}, None
169    commands: dict[str, list[str]] = {}
170    folder = None
171    for entry in entries if isinstance(entries, list) else []:
172        if not isinstance(entry, dict) or "file" not in entry:
173            continue
174        args = entry.get("arguments") or shlex.split(entry.get("command", ""))
175        base = Path(entry.get("directory") or project_dir)
176        source = (base / entry["file"]).resolve()
177        commands[str(source)] = list(args)
178        if folder is None and "-J" in args:
179            j = args.index("-J")
180            if j + 1 < len(args):
181                folder = Path(args[j + 1]).name
182    return commands, folder

185def format_command(args: list[str]) -> str:
186    """A shell-quoted command, one option per line for readability."""
187    if not args:
188        return ""
189    lines = [quote_arg(args[0])]
190    i = 1
191    while i < len(args):
192        part = quote_arg(args[i])
193        # Keep an option and its value together ("-o file", "-J dir").
194        if args[i] in ("-o", "-J", "-I", "-c") and i + 1 < len(args):
195            part += " " + quote_arg(args[i + 1])
196            i += 1
197        lines.append(part)
198        i += 1
199    # Line continuation: "\" in POSIX shells, "^" in the Windows shell.
200    return f" {'^' if WINDOWS else chr(92)}\n    ".join(lines)

204class ConfigRecords:
205    """Per-project record of which folders each GUI build wrote, stored in
206    the app's settings as JSON: [{label, flags, folders, built}, ...]."""
207
208    def __init__(self, settings, key: str):
209        self.settings, self.key = settings, key
210
211    def load(self) -> list[dict]:
212        try:
213            data = json.loads(self.settings.value(self.key, "[]"))
214            return [r for r in data if isinstance(r, dict) and r.get("folders")]
215        except (TypeError, ValueError):
216            return []
217
218    def save(self, records: list[dict]):
219        self.settings.setValue(self.key, json.dumps(records))
220
221    def last_used(self) -> tuple[str, str] | None:
222        """(label, flags) of the last build the GUI ran, if any."""
223        try:
224            data = json.loads(self.settings.value(self.key + "-last", "null"))
225            return (data["label"], data["flags"]) if data else None
226        except (TypeError, ValueError, KeyError):
227            return None
228
229    def _set_last(self, label: str, flags: str):
230        self.settings.setValue(self.key + "-last", json.dumps({"label": label, "flags": flags}))
231
232    def record_build(self, label: str, flags: str, folders: list[str]):
233        """``folders`` were just written by a build with these settings."""
234        self._set_last(label, flags)
235        if not folders:
236            return
237        records = self.load()
238        for r in records:  # a folder belongs to one configuration only
239            r["folders"] = [f for f in r["folders"] if f not in folders]
240        existing = next((r for r in records if r.get("label") == label and r.get("flags") == flags), None)
241        if existing is None:
242            existing = {"label": label, "flags": flags, "folders": []}
243            records.append(existing)
244        existing["folders"] = sorted(set(existing["folders"]) | set(folders))
245        existing["built"] = existing["used"] = time.time()
246        self.save([r for r in records if r["folders"]])
247
248    def mark_used(self, label: str, flags: str):
249        """A build with these settings found nothing to do: it's still current."""
250        self._set_last(label, flags)
251        records = self.load()
252        for r in records:
253            if r.get("label") == label and r.get("flags") == flags:
254                r["used"] = time.time()
255                self.save(records)
256                return
257
258    def forget(self, folders: list[str]):
259        records = self.load()
260        for r in records:
261            r["folders"] = [f for f in r["folders"] if f not in folders]
262        self.save([r for r in records if r["folders"]])

265def configurations(project_dir: Path, records: list[dict],
266                   last_used: tuple[str, str] | None = None) -> list[Configuration]:
267    """Group the build folders into configurations, newest first.
268    ``last_used`` is the (label, flags) of the GUI's last build."""
269    folders = {p.name: scan_folder(p) for p in build_folders(project_dir)}
270    _, current_compile = read_compile_commands(project_dir)
271    configs: list[Configuration] = []
272    claimed: set[str] = set()
273    for r in records:
274        present = [folders[n] for n in r["folders"] if n in folders and n not in claimed]
275        if not present:
276            continue
277        claimed.update(f.name for f in present)
278        config = Configuration(r.get("label", "?"), present, True, r.get("flags", ""),
279                               r.get("built") or max(f.newest for f in present))
280        config.used = r.get("used") or config.built
281        configs.append(config)
282    # Folders without a record: group those written at about the same time.
283    loose = sorted((f for n, f in folders.items() if n not in claimed), key=lambda f: f.newest)
284    group: list[BuildFolder] = []
285    for f in loose + [None]:
286        if f is not None and (not group or f.newest - group[-1].newest <= GROUP_GAP_SECONDS):
287            group.append(f)
288            continue
289        if group:
290            compilers = sorted({g.compiler for g in group})
291            configs.append(Configuration(f"{', '.join(compilers)} · built outside the app",
292                                         group, False, "", max(g.newest for g in group)))
293        group = [f] if f is not None else []
294    configs.sort(key=lambda c: c.built, reverse=True)
295    # "Current": the configuration of the GUI's last build. If that build's
296    # settings have no recorded folders (they were built outside the app),
297    # nothing is marked. Without any GUI build, use the folder the latest
298    # compile commands point at, else the newest.
299    if last_used is not None:
300        for c in configs:
301            if c.recorded and (c.label, c.flags) == tuple(last_used):
302                c.current = True
303                break
304    else:
305        for c in configs:
306            if current_compile and any(f.name == current_compile for f in c.folders):
307                c.current = True
308                break
309        else:
310            if configs:
311                configs[0].current = True
312    return configs

315def human_size(n: int) -> str:
316    for unit in ("B", "KB", "MB", "GB"):
317        if n < 1024 or unit == "GB":
318            return f"{n:.0f} {unit}" if unit == "B" else f"{n:.1f} {unit}"
319        n /= 1024
320    return f"{n} B"

323def human_age(t: float) -> str:
324    seconds = max(0, time.time() - t)
325    if seconds < 60:
326        return "just now"
327    for unit, size in (("day", 86400), ("hour", 3600), ("minute", 60)):
328        if seconds >= size:
329            n = int(seconds // size)
330            return f"{n} {unit}{'s' * (n != 1)} ago"
331    return ""

334def delete_folders(paths: list[Path]) -> list[str]:
335    """Remove build folders; returns error messages (empty if all went)."""
