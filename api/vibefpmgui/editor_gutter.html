 47    def event(self, event):
 48        if event.type() == event.Type.ToolTip:
 49            line = self.editor.line_at_y(event.pos().y())
 50            if event.pos().x() < self.editor._breakpoint_column():
 51                tip = self.editor.breakpoint_tip(line)
 52                if tip:
 53                    QToolTip.showText(event.globalPos(), tip, self)
 54                else:
 55                    QToolTip.hideText()
 56                return True
 57            origin = self.editor._breakpoint_column()
 58            if origin <= event.pos().x() < origin + self.editor._coverage_column() and self.editor.coverage_note(line):
 59                QToolTip.showText(event.globalPos(), self.editor.coverage_note(line), self)
 60                return True
 61            change_tip = self._change_tip(event.pos(), line)
 62            if change_tip:  # over a change bar or a deletion marker
 63                QToolTip.showText(event.globalPos(), change_tip, self)
 64                return True
 65            notes = self.editor.markers.get(line)
 66            # (Spelling has no gutter marks: its notes are on the words only.)
 67            issues = [i for i in self.editor.lint_issues if i.line == line and i.severity != "spelling"]
 68            popup = self.editor.hover_popup
 69            if issues:  # the same card as hovering the text (with its fix links)
 70                if not (popup.isVisible() and popup.anchor == ("gutter", line)):
 71                    # (fortitude's and fortls's notes are the card's issues already)
 72                    others = [m for severity, m in notes or [] if severity not in hover_card.LINT_COLORS]
 73                    popup.show_at(event.globalPos(), self.editor.lint_card_html(issues, others), ("gutter", line))
 74                QToolTip.hideText()
 75            elif notes or line in self.editor.gremlins:
 76                lines = [m for _, m in notes or []] + self.editor.gremlin_notes(line)
 77                QToolTip.showText(event.globalPos(), "\n".join(lines), self)
 78            else:
 79                QToolTip.hideText()
 80            return True
 81        return super().event(event)

 83    def _change_tip(self, pos, line: int) -> str:
 84        """Tooltip for the change bars at the gutter's right edge."""
 85        if pos.x() < self.width() - 9:
 86            return ""
 87        editor = self.editor
 88        block = editor.document().findBlockByNumber(line - 1)
 89        if block.isValid():
 90            top = editor.blockBoundingGeometry(block).translated(editor.contentOffset()).top()
 91            bottom = top + editor.blockBoundingRect(block).height()
 92            for deleted_at, y in ((line, top), (line + 1, bottom)):
 93                if deleted_at in editor.deletions and abs(pos.y() - y) <= 5:
 94                    kind, count = editor.deletions[deleted_at]
 95                    what = f"{count} line{'s' * (count != 1)} deleted here"
 96                    return what + (" — not saved yet" if kind == "unsaved" else " (saved)")
 97        change = editor.change_marks.get(line)
 98        if change == "unsaved":
 99            return "Changed — not saved yet"
100        if change == "saved":
101            return "Changed since the file was opened (saved)"
102        return ""

141    def breakpoint_lines(self) -> list[int]:
142        """The lines (1-based) with a breakpoint, enabled or not, as they are
143        now (they move with edits)."""
144        return sorted({cursor.blockNumber() + 1 for cursor, _on in self._breakpoints})

146    def disabled_breakpoints(self) -> list[int]:
147        """The lines whose breakpoint is disabled (kept, drawn hollow; the debugger doesn't stop there)."""
148        enabled = {cursor.blockNumber() + 1 for cursor, on in self._breakpoints if on}
149        return sorted({cursor.blockNumber() + 1 for cursor, on in self._breakpoints if not on} - enabled)

151    def enabled_breakpoints(self) -> list[int]:
152        disabled = set(self.disabled_breakpoints())
153        return [n for n in self.breakpoint_lines() if n not in disabled]

155    def set_breakpoint_lines(self, lines: list[int], disabled: list[int] | None = None):
156        """Breakpoints on these lines (those past the end: dropped); ``disabled``: these of them, disabled."""
157        doc = self.document()
158        off = set(disabled or ())
159        self._breakpoints = []
160        for line in sorted(set(lines)):
161            block = doc.findBlockByNumber(line - 1)
162            if block.isValid():
163                self._breakpoints.append((QTextCursor(block), line not in off))
164        self.breakpoint_problems = {}
165        self.gutter.update()

167    def toggle_breakpoint(self, line: int):
168        """A breakpoint on ``line`` set, or removed (gutter click, F9)."""
169        if not self.debuggable:
170            return
171        lines, off = self.breakpoint_lines(), self.disabled_breakpoints()
172        self.set_breakpoint_lines([n for n in lines if n != line] if line in lines else [*lines, line], off)
173        self.breakpoints_changed.emit()

175    def toggle_breakpoint_enabled(self, line: int):
176        """The breakpoint on ``line`` disabled (kept, hollow), or enabled again
177        (⌘/Ctrl-click, Ctrl+F9); none there: one set."""
178        if not self.debuggable:
179            return
180        if line not in self.breakpoint_lines():
181            self.toggle_breakpoint(line)
182            return
183        self.set_breakpoints_enabled([line], line in self.disabled_breakpoints())

185    def set_breakpoints_enabled(self, lines: list[int], enabled: bool):
186        """These lines' breakpoints enabled, or disabled (Debug → Disable All Breakpoints)."""
187        if not self.debuggable:
188            return
189        wanted = set(lines)
190        changed = [(cursor, enabled if cursor.blockNumber() + 1 in wanted else on) for cursor, on in self._breakpoints]
191        if changed != self._breakpoints:

228    def set_coverage(self, lines: dict[int, int] | None, branches: dict[int, tuple[int, int]] | None = None):
229        """The last test run's coverage of this file: line -> times run (lines
230        with code only), shown as green and red bars in the gutter (amber: it
231        ran, but not all of its ``branches``: line -> (taken, total))."""
232        lines, branches = dict(lines or {}), dict(branches or {}) if lines else {}
233        if lines == self.coverage and branches == self.coverage_branches:
234            return
235        self.coverage, self.coverage_branches = lines, branches
236        self.refresh_gutter()

238    def coverage_state(self, line: int) -> str | None:
239        """ "ran", "partial" (ran; not every branch taken), "missed", or None
240        (no code there, or no coverage)."""
241        if line not in self.coverage:
242            return None
243        if not self.coverage[line]:
244            return "missed"
245        taken, total = self.coverage_branches.get(line, (0, 0))
246        return "partial" if taken < total else "ran"

248    def missed_starts(self) -> list[int]:
249        """The first line of each stretch of lines the tests didn't run
250        (lines without code in between don't end a stretch)."""
251        starts, previous = [], None
252        for line in sorted(self.coverage):
253            missed = self.coverage[line] == 0
254            if missed and previous is not True:
255                starts.append(line)
256            previous = missed
257        return starts

259    def go_to_missed(self, forward: bool = True) -> bool:
260        """Edit → Go To → Next / Previous Missed Line: to the next stretch of lines
261        the tests didn't run (around to the start at the end); False if
262        there's none."""
263        starts = self.missed_starts()
264        if not starts:
265            return False
266        here = self.textCursor().blockNumber() + 1
267        if forward:
268            target = next((n for n in starts if n > here), starts[0])
269        else:
270            # (From inside a stretch, the one before it: not its own start.)
271            current = max((n for n in starts if n <= here), default=None)
272            target = next((n for n in reversed(starts) if n < (current if current is not None else here)),
273                          starts[-1])
274        self.go_to_line(target)
275        return True

287    def coverage_note(self, line: int) -> str:
288        """The gutter's tooltip for a coverage bar."""
289        if line not in self.coverage:
290            return ""
291        count = self.coverage[line]
292        if not count:
293            return "Not run by the tests"

329        if width != self.viewportMargins().left():
330            self.setViewportMargins(width, 0, 0, 0)

340    def paint_gutter(self, event):
341        pal = self.palette()
342        painter = QPainter(self.gutter)
343        base = pal.color(QPalette.Base)
344        gutter_bg = base.darker(104) if base.lightness() >= 128 else base.lighter(118)
345        painter.fillRect(event.rect(), gutter_bg)
346        text = pal.color(QPalette.Text)
347        muted = QColor(text)
348        muted.setAlpha(110)
349        current = self.textCursor().blockNumber()
350        block = self.firstVisibleBlock()
351        number = block.blockNumber()
352        top = round(self.blockBoundingGeometry(block).translated(self.contentOffset()).top())
353        bottom = top + round(self.blockBoundingRect(block).height())
354        height = self.fontMetrics().height()
355        width = self.gutter.width()
356        painter.setRenderHint(QPainter.Antialiasing)
357        origin = self._breakpoint_column()  # (the breakpoints, at the left edge: the rest after them)
358        breakpoints = set(self.breakpoint_lines()) if origin else set()
359        disabled = set(self.disabled_breakpoints()) if origin else set()
360        while block.isValid() and top <= event.rect().bottom():
361            if block.isVisible() and bottom >= event.rect().top():
362                line = number + 1
363                if origin:
364                    self._paint_breakpoint(painter, line, line in breakpoints, line in disabled, top, height)
365                left = self._coverage_column()
366                state = self.coverage_state(line) if left else None
367                if state:  # (ran: green; ran, not every branch: amber; didn't: red)
368                    painter.setPen(Qt.NoPen)
369                    painter.setBrush(QColor(COVERAGE_COLORS[state]))
370                    painter.drawRect(QRectF(origin, top, 3, bottom - top))
371                left += origin
372                if line in self.gremlins:
373                    size = min(GREMLIN_WIDTH - 2, height)
374                    gremlins_mod.paint_gremlin(painter, QRectF(1 + left, top + (height - size) / 2, size, size))
375                font = QFont(self.font())
376                font.setBold(number == current)
377                painter.setFont(font)
378                painter.setPen(text if number == current else muted)
379                painter.drawText(0, top, width - 8, height, Qt.AlignRight | Qt.AlignVCenter, str(line))
380                notes = self.markers.get(line)
381                if notes:
382                    worst = worst_severity(notes)
383                    painter.setPen(Qt.NoPen)
384                    painter.setBrush(QColor(MARKER_COLORS.get(worst, "#868e96")))
385                    d = max(6, height // 2)
386                    painter.drawEllipse(4 + left + self._gremlin_column(), top + (height - d) // 2, d, d)
387                # Change bar (full line height) at the edge next to the text.
388                change = self.change_marks.get(line)
389                if change:
390                    painter.setPen(Qt.NoPen)
391                    painter.setBrush(QColor(CHANGE_COLORS[change]))
392                    painter.drawRect(QRectF(width - 4, top, 3, bottom - top))
393                # Deleted lines: a small triangle on the boundary above this
394                # line (and below the last one, for a deletion at the end).
395                for deleted_at, y in ((line, top), (line + 1, bottom)):
396                    if deleted_at == line + 1 and block.next().isValid():
397                        continue
398                    if deleted_at in self.deletions:
399                        kind, _count = self.deletions[deleted_at]
400                        painter.setPen(Qt.NoPen)
401                        painter.setBrush(QColor(CHANGE_COLORS[kind]))
402                        painter.drawPolygon(QPolygonF([QPointF(width - 7, y - 4.5), QPointF(width - 1, y),
403                                                       QPointF(width - 7, y + 4.5)]))
404            block = block.next()
405            top = bottom
406            bottom = top + round(self.blockBoundingRect(block).height())
407            number += 1
408        painter.end()

410    def _paint_breakpoint(self, painter: QPainter, line: int, on: bool, disabled: bool, top: int, height: int):
411        """A line's breakpoint (a red dot; hollow: disabled; grey: the debugger
412        couldn't place it) and, where the program is stopped, an arrow over it."""
413        d = min(BREAKPOINT_WIDTH - 5, height - 4)
414        x, y = (BREAKPOINT_WIDTH - d) / 2, top + (height - d) / 2
415        if on:
416            color = QColor(BREAKPOINT_COLOR)
417            if disabled:
418                painter.setPen(QPen(color, 1.5))
419                painter.setBrush(Qt.NoBrush)
420            else:
421                painter.setPen(Qt.NoPen)
422                painter.setBrush(QColor(BREAKPOINT_UNPLACED) if line in self.breakpoint_problems else color)
423            painter.drawEllipse(QRectF(x, y, d, d))
424        if line == self.execution_line:
425            painter.setPen(QPen(QColor(EXECUTION_COLOR).darker(140), 1))
426            painter.setBrush(QColor(EXECUTION_COLOR))
427            mid = top + height / 2
428            painter.drawPolygon(QPolygonF([QPointF(1.5, mid - d / 2), QPointF(1.5 + d * 0.55, mid - d / 2),
429                                           QPointF(BREAKPOINT_WIDTH - 1.5, mid),
430                                           QPointF(1.5 + d * 0.55, mid + d / 2), QPointF(1.5, mid + d / 2)]))
