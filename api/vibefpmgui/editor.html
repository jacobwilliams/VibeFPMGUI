vibefpmgui.editor

A small built-in code editor: line numbers, syntax highlighting (Pygments, chosen by file extension), find/replace, go to line, and compiler warning / error markers in the gutter, fortitude lint findings (see vibefpmgui.lint), and for Fortran files the language server's hover, go to definition and outline (see vibefpmgui.lsp).

EditorTab wraps a CodeEditor (vibefpmgui.code_editor) with a find bar (vibefpmgui.find_bar), the outline and a status line, and takes care of loading and saving the file (keeping its encoding and line endings) and of noticing when it changes on disk. Its language-server features are in vibefpmgui.editor_language, its checks (fortitude, spelling, gremlins, the compiler's and the AI's markers) in vibefpmgui.editor_checks.

 83    def paint(self, painter, option, index):
 84        opt = QStyleOptionViewItem(option)
 85        self.initStyleOption(opt, index)
 86        style = opt.widget.style() if opt.widget else QApplication.style()
 87        selected = bool(opt.state & QStyle.State_Selected) or bool(opt.state & QStyle.State_MouseOver)
 88        painter.save()
 89        if selected:
 90            painter.fillRect(opt.rect, opt.palette.color(QPalette.Active, QPalette.Highlight))
 91        else:
 92            style.drawPrimitive(QStyle.PE_PanelItemViewItem, opt, painter, opt.widget)
 93        x = opt.rect.left() + 8 + SYMBOL_INDENT * int(index.data(DEPTH_ROLE) or 0)
 94        size = opt.decorationSize
 95        top = opt.rect.top() + (opt.rect.height() - size.height()) // 2
 96        opt.icon.paint(painter, QRect(x, top, size.width(), size.height()))
 97        painter.setPen(opt.palette.color(QPalette.Active, QPalette.HighlightedText) if selected
 98                       else opt.palette.color(QPalette.Text))
 99        painter.setFont(opt.font)
100        text_rect = QRect(x + size.width() + 6, opt.rect.top(), opt.rect.right() - x - size.width() - 6,
101                          opt.rect.height())
102        painter.drawText(text_rect, Qt.AlignVCenter | Qt.AlignLeft, opt.text)
103        painter.restore()

331    def load(self):

358    def convert_line_endings(self, newline: str) -> bool:
359        """Edit → Convert Line Endings: the file is saved with ``newline``
360        ("\\n" Unix, "\\r\\n" Windows) at every line end (a file with mixed
361        ones, or old Mac ones, which Qt reads as line breaks, gets the same
362        everywhere). False if it already has them."""
363        if newline == self.newline and not self.mixed_newlines:
364            return False
365        self.newline = newline
366        self.mixed_newlines = False  # (all the same from the next save)
367        self.dirty_changed.emit(self.is_dirty())
368        self.update_status()
369        return True

371    def trim_trailing_whitespace(self) -> int:
372        """Edit → Trim Trailing Whitespace: the spaces and tabs at the ends of
373        the lines removed (one step to undo); how many lines had them. The
374        lines are found in the text at once (going through a big file's
375        every line took a tenth of a second at every save)."""
376        text = self.editor.toPlainText()
377        lines, line, at = [], 0, 0
378        for match in TRAILING_SPACE_RE.finditer(text):
379            line += text.count("\n", at, match.start())
380            at = match.start()
381            lines.append(line)
382        if not lines:
383            return 0
384        document = self.editor.document()
385        cursor = QTextCursor(document)
386        cursor.beginEditBlock()
387        for number in lines:
388            block = document.findBlockByNumber(number)
389            text = block.text()
390            # (Qt's positions count a character beyond U+FFFF as two.)
391            cursor.setPosition(block.position() + _qt_length(text.rstrip()))
392            cursor.setPosition(block.position() + _qt_length(text), QTextCursor.KeepAnchor)
393            cursor.removeSelectedText()
394        cursor.endEditBlock()
395        return len(lines)

397    def save(self) -> bool:
398        if self.untitled:
399            return self.save_as()  # (a name first)
400        if self.trim_on_save and self.path.suffix.lower() not in TRIM_KEEPS:
401            self.trim_trailing_whitespace()
402        text = self.editor.toPlainText()
403        try:
404            data = text.replace("\n", self.newline).encode(self.encoding)
405        except UnicodeEncodeError:
406            data = text.replace("\n", self.newline).encode("utf-8")
407            self.encoding = "utf-8"
408        try:
409            self.path.write_bytes(data)
410        except OSError as exc:
411            QMessageBox.warning(self, "Save failed", f"Could not save {self.path}:\n{exc}")
412            return False
413        self._disk_text = text
414        self._disk_newline = self.newline
415        self.mixed_newlines = self._disk_mixed = False  # (saved: every line ends the same way now)
416        self._disk_change = None  # (the save replaced any other version on disk)
417        self.disk_bar.hide()
418        self.editor.document().setModified(False)
419        self.dirty_changed.emit(False)  # (also when only the line endings had changed)
420        # Line numbers in the compiler/review notes may no longer be right.
421        self._compiler_markers, self._review_markers = {}, {}
422        self.editor.set_markers({})
423        self._watch()
424        if self.lsp is not None:
425            self._flush_lsp()
426            self.lsp.save_document(self.path)
427        self.update_change_marks()  # unsaved bars become saved ones
428        self.refresh_blame()
429        self.saved.emit(str(self.path))
430        return True

432    def save_as(self) -> bool:
433        filename, _ = QFileDialog.getSaveFileName(self, "Save As", str(self.path))
434        if not filename:
435            return False
436        client = self.lsp
437        self.detach_lsp()  # (under the old name)
438        self.untitled = False
439        new = not Path(filename).exists()
440        self._set_path(filename)  # (its type known now: colors, comments…)
441        saved = self.save()
442        if saved and new:
443            self.created.emit(str(self.path))
444        self.lint()  # it may (no longer) be a Fortran file
445        self.attach_lsp(client)

607    def _issue_counts(self) -> list[str]:
608        """The status line's counts ("12 fortitude issues"…). Counted again
609        only when what's counted changed (each is replaced, not changed in
610        place, when it does): the status line follows the cursor, and a big
611        file's thousands of issues were counted at every step."""
612        counted = (self._compiler_markers, self._review_markers, self._lint_markers, self._lsp_markers,
613                   self._spelling_issues, self.editor.gremlins)
614        cached = getattr(self, "_counts", None)
615        if cached is not None and all(a is b for a, b in zip(cached[0], counted, strict=True)):
616            return cached[1]
617        parts = []
618        if self._compiler_markers:
619            notes = [s for ns in self._compiler_markers.values() for s, _ in ns]
620            errors = sum(1 for s in notes if s in ("error", "fatal error"))
621            warnings = sum(1 for s in notes if s == "warning")
622            parts.append(f"{errors} errors, {warnings} warnings from the last build")
623        for markers, what in ((self._review_markers, "AI review note"), (self._lint_markers, "fortitude issue"),
624                              (self._lsp_markers, "fortls issue")):
625            if markers:
626                count = sum(len(ns) for ns in markers.values())
627                parts.append(f"{count} {what}{'s' * (count != 1)}")
628        if self._spelling_issues:
629            count = len(self._spelling_issues)
630            parts.append(f"{count} misspelled word{'s' * (count != 1)}")
631        if self.editor.gremlins:
632            count = sum(len(g) for g in self.editor.gremlins.values())
633            parts.append(f"{count} gremlin{'s' * (count != 1)}")
634        self._counts = (counted, parts)
635        return parts
