vibefpmgui.command_palette

View → Command Palette… (⌘⇧P / Ctrl+Shift+P): every command in the menus, found by typing part of its name, as in VS Code.

The commands aren't listed here: they're the menu bar's actions, read when the palette opens (collect), so a command added to a menu is in the palette too. A menu that fills itself when it opens (Syntax Colors, Window Style, Recent Projects…) is asked to, as it is when it's opened. Running a command triggers its action, as choosing it in its menu does.

 50def clean(text: str) -> str:
 51    """A menu text as shown: no "&" for its mnemonic ("&&" is an "&"), no
 52    tab and what's after it (a shortcut some menus put in their text)."""
 53    text = text.split("\t", 1)[0]
 54    return re.sub(r"&(.)", r"\1", text).strip()

 57def collect(menu_bar: QMenuBar) -> list[Command]:
 58    """Every command in the menu bar's menus, in menu order: not separators,
 59    submenus themselves or hidden actions.
 60
 61    A submenu is found from its menu (``QMenu.menuAction()``), never with
 62    ``QAction.menu()``: in PySide that leaves a menu the menu bar made (and
 63    its actions) deleted as far as Python knows, which broke the app's own
 64    menus."""
 65    commands: list[Command] = []
 66    seen: set[int] = set()
 67    submenus = {m.menuAction(): m for m in menu_bar.window().findChildren(QMenu)}
 68
 69    def walk(menu: QMenu, path: tuple[str, ...]):
 70        menu.aboutToShow.emit()  # (a menu filled when it opens: filled now, as then)
 71        submenus.update({m.menuAction(): m for m in menu.findChildren(QMenu)})  # (made by filling it)
 72        for action in menu.actions():
 73            if action.isSeparator() or not action.isVisible() or not clean(action.text()):
 74                continue
 75            here = (*path, clean(action.text()))
 76            submenu = submenus.get(action)
 77            if submenu is not None:
 78                walk(submenu, here)
 79            elif id(action) not in seen:
 80                seen.add(id(action))
 81                commands.append(Command(here, action))
 82
 83    for action in menu_bar.actions():
 84        submenu = submenus.get(action)
 85        if submenu is not None and action.isVisible():
 86            walk(submenu, (clean(action.text()),))
 87    return commands

 90def _word_starts(text: str) -> set[int]:
 91    return {0} | {i for i in range(1, len(text)) if not text[i - 1].isalnum() and text[i].isalnum()}

 94def _subsequence(word: str, text: str, starts: set[int], prefer: bool = True) -> int | None:
 95    """How well ``word``'s letters appear, in order, in ``text`` (None: they
 96    don't): each letter counts, more at a word's start or right after the
 97    one before. Each letter goes at the next word's start that has it, if
 98    any (``prefer``); if that leaves a later letter unmatched, at the next
 99    place it is."""
100    score, at, last, first = 0, 0, -2, None
101    for ch in word:
102        found = text.find(ch, at)
103        if found < 0:
104            return None if prefer is False else _subsequence(word, text, starts, prefer=False)
105        if prefer and found != last + 1 and found not in starts:
106            later = found
107            while later >= 0 and later not in starts:
108                later = text.find(ch, later + 1)
109            if later >= 0:
110                found = later
111        gap = found - last - 1 if last >= 0 else 0
112        score += 1 + (3 if found in starts else 0) + (2 if found == last + 1 else 0) - min(gap, 4)
113        first = found if first is None else first
114        last, at = found, found + 1
115    if first is not None and last - first + 1 > max(3 * len(word), len(word) + 8):
116        return None  # (its letters scattered far apart: not what was meant)
117    return score

120def score(query: str, command: Command) -> int | None:
121    """How well a command matches what's typed (None: it doesn't). Each word
122    typed must match, letters in order (so "mvup" finds Move Line Up); in
123    the command's own name counts more than in its menus' names."""
124    words = query.lower().split()
125    if not words:
126        return 0
127    name, title = command.name.lower(), command.title.lower()
128    name_starts, title_starts = _word_starts(name), _word_starts(title)
129    total = 0
130    for word in words:
131        in_name = _subsequence(word, name, name_starts)
132        in_title = _subsequence(word, title, title_starts)
133        if in_name is None and in_title is None:
134            return None
135        total += max(in_name * 2 if in_name is not None else 0, in_title or 0)
136        if name.startswith(word):
137            total += 5
138    if name.rstrip("…") == " ".join(words):
139        total += 10  # (just its name: it before others that start the same)
140    return total

143def ranked(query: str, commands: list[Command], recent: list[str]) -> list[Command]:
144    """The commands matching ``query``, best first: the best match, then the
145    most recently used, then those available now (the others are listed,
146    greyed), then the fewest menus deep, then the shortest name, then menu
147    order. Nothing typed: the
148    recently used first, then the menus' order."""
149    order = {title: n for n, title in enumerate(recent)}
150    matched = []
151    for index, command in enumerate(commands):
152        points = score(query, command)
153        if points is not None:
154            matched.append((-points, order.get(command.title, len(recent)), not command.action.isEnabled(),
155                            len(command.path), len(command.name), index, command))
156    matched.sort(key=lambda m: m[:6] if query.strip() else (m[1], m[2], m[5]))
157    return [m[-1] for m in matched]

160class _ItemDelegate(QStyledItemDelegate):
161    """A command: its menus' names muted, then its own; a tick if it's
162    checked; its shortcut on the right."""
163
164    def sizeHint(self, option, index):
165        return QSize(option.rect.width(), QFontMetrics(option.font).height() + 10)
166
167    def paint(self, painter, option, index):
168        command: Command = index.data(COMMAND_ROLE)
169        style = option.widget.style() if option.widget else QApplication.style()
170        style.drawPrimitive(QStyle.PE_PanelItemViewItem, option, painter, option.widget)
171        selected = bool(option.state & QStyle.State_Selected)
172        pal = option.palette
173        text = pal.color(QPalette.HighlightedText if selected else QPalette.Text)
174        muted = QColor(text)
175        muted.setAlpha(150)
176        if not command.action.isEnabled():
177            text.setAlpha(110)
178            muted.setAlpha(80)
179        fm = QFontMetrics(option.font)
180        rect = option.rect.adjusted(10, 0, -10, 0)
181        painter.save()
182        shortcut = command.action.shortcut().toString(QKeySequence.NativeText)
183        if shortcut:
184            painter.setPen(muted)
185            painter.drawText(rect, Qt.AlignRight | Qt.AlignVCenter, shortcut)
186            rect.setRight(rect.right() - fm.horizontalAdvance(shortcut) - 16)
187        x = rect.left()
188        tick = "✓ " if command.action.isCheckable() and command.action.isChecked() else ""
189        menus = SEPARATOR.join(command.path[:-1]) + SEPARATOR if len(command.path) > 1 else ""
190        for part, color in ((tick, text), (menus, muted), (command.name, text)):
191            if not part:
192                continue
193            painter.setPen(color)
194            width = rect.right() - x
195            shown = fm.elidedText(part, Qt.ElideMiddle, max(0, width))
196            painter.drawText(QRect(x, rect.top(), max(0, width), rect.height()), Qt.AlignLeft | Qt.AlignVCenter,
197                             shown)
198            x += fm.horizontalAdvance(shown)
199        painter.restore()

202class CommandPalette(QFrame):
203    """The palette: a field to type in, the matching commands under it.
204    ``recent`` (the titles of the commands last run, latest first) is kept
205    up to date in place; ``recent_changed`` (if set) is called after."""
206
207    WIDTH = 620
208    ROWS = 14
209
210    def __init__(self, window: QWidget, recent: list[str] | None = None):
211        super().__init__(window, Qt.Popup)
212        self.window_ = window
213        self.recent = recent if recent is not None else []
214        self.recent_changed = None
215        self.commands: list[Command] = []
216        self.setFrameShape(QFrame.StyledPanel)
217        self.field = QLineEdit()
218        self.field.setPlaceholderText("Type a command (letters in order: \"mvup\" finds Move Line Up)")
219        self.field.setClearButtonEnabled(True)
220        self.field.textChanged.connect(self._filter)
221        self.field.installEventFilter(self)
222        self.list = QListWidget()
223        self.list.setItemDelegate(_ItemDelegate(self.list))
224        self.list.setUniformItemSizes(True)
225        self.list.setFocusPolicy(Qt.NoFocus)
226        self.list.itemClicked.connect(lambda item: self.run(item.data(COMMAND_ROLE)))
227        layout = QVBoxLayout(self)
228        layout.setContentsMargins(6, 6, 6, 6)
229        layout.setSpacing(6)
230        layout.addWidget(self.field)
231        layout.addWidget(self.list)
232
233    def open(self):
234        """Read the menus, and show the palette at the top of the window."""
235        menu_bar = self.window_.menuBar() if hasattr(self.window_, "menuBar") else None
236        self.commands = collect(menu_bar) if menu_bar is not None else []
237        self.field.clear()
238        self._filter("")
239        width = min(self.WIDTH, max(300, self.window_.width() - 40))
240        row = self.list.sizeHintForRow(0) if self.list.count() else 24
241        self.resize(width, self.field.sizeHint().height() + row * self.ROWS + 24)
242        top = self.window_.mapToGlobal(self.window_.rect().topLeft())
243        self.move(top.x() + (self.window_.width() - width) // 2, top.y() + 40)
244        self.show()
245        self.raise_()
246        self.field.setFocus()
247
248    def _filter(self, text: str):
249        self.list.clear()
250        for command in ranked(text, self.commands, self.recent):
251            item = QListWidgetItem(command.title)
252            item.setData(COMMAND_ROLE, command)
253            if not command.action.isEnabled():
254                item.setToolTip("Not available now")
255            elif command.action.statusTip():
256                item.setToolTip(command.action.statusTip())
257            self.list.addItem(item)
258        if self.list.count():
259            self.list.setCurrentRow(0)
260
261    def eventFilter(self, obj, event):
262        if obj is self.field and event.type() == QEvent.KeyPress:
263            key = event.key()
264            steps = {Qt.Key_Up: -1, Qt.Key_Down: 1, Qt.Key_PageUp: -self.ROWS, Qt.Key_PageDown: self.ROWS}
265            if key in steps and self.list.count():
266                row = max(0, min(self.list.count() - 1, self.list.currentRow() + steps[key]))
267                self.list.setCurrentRow(row)
268                return True
269            if key in (Qt.Key_Return, Qt.Key_Enter):
270                item = self.list.currentItem()
271                if item is not None:
272                    self.run(item.data(COMMAND_ROLE))
273                return True
274            if key == Qt.Key_Escape:
275                self.close()
276                return True
277        return super().eventFilter(obj, event)
278
279    def run(self, command: Command):
280        """Run a command (if it's available now): the palette closes first, so
281        the command acts where the focus was (e.g. the editor)."""
282        if not command.action.isEnabled():
283            return
284        title = command.title
285        if title in self.recent:
286            self.recent.remove(title)
287        self.recent.insert(0, title)
288        del self.recent[MAX_RECENT:]
289        if self.recent_changed is not None:
290            self.recent_changed(self.recent)
291        self.close()
292        QTimer.singleShot(0, self.window_, command.action.trigger)
293
294    def current_titles(self) -> list[str]:
295        """(The titles listed, in order: for the tests.)"""
296        return [self.list.item(i).data(COMMAND_ROLE).title for i in range(self.list.count())]
