vibefpmgui.complexity_panel

The Complexity tab: each procedure's complexity, as lizard measures it (see vibefpmgui.complexity): CCN, lines of code, tokens, parameters, length and nesting depth, the values over lizard's thresholds in red; each file's totals; and the selected procedure's control-flow graph (see vibefpmgui.control_flow, vibefpmgui.cfg_view). Click a procedure to open it; click a file to list only its procedures (its first one selected and drawn); click a node of the graph to open its lines.

192    def _table(self, columns: list[Column], style: RowStyle) -> MetricsTable:
193        """A table whose columns are as wide as their contents, scrolling
194        sideways if the panel is narrower (not squeezing the first column);
195        its cells drawn on demand (see `vibefpmgui.metrics_table`)."""
196        table = MetricsTable(columns, style)
197        table.verticalHeader().setVisible(False)
198        table.setShowGrid(False)
199        table.setEditTriggers(QAbstractItemView.NoEditTriggers)
200        table.setSelectionBehavior(QAbstractItemView.SelectRows)
201        table.setSelectionMode(QAbstractItemView.SingleSelection)
202        table.setSortingEnabled(True)
203        table.setWordWrap(False)
204        table.setTextElideMode(Qt.ElideLeft)  # (a long path: the end of its file name still shows)
205        table.setHorizontalScrollMode(QAbstractItemView.ScrollPerPixel)
206        hh = table.horizontalHeader()
207        hh.setDefaultAlignment(Qt.AlignLeft | Qt.AlignVCenter)
208        hh.setSectionResizeMode(QHeaderView.Interactive)
209        hh.setStretchLastSection(False)
210        hh.sortIndicatorChanged.connect(lambda *_: self._fit_columns(table))
211        return table

213    @staticmethod
214    def _fit_columns(table: MetricsTable):
215        """Each column as wide as its values and its title, room for the sort
216        arrow only in the sorted one (Qt leaves room for it in every title)."""
217        header = table.horizontalHeader()
218        header.setSortIndicatorShown(False)
219        table.resizeColumnsToContents()
220        header.setSortIndicatorShown(True)
221        column = header.sortIndicatorSection()
222        if 0 <= column < header.count():
223            header.resizeSection(column, max(header.sectionSize(column), header.sectionSizeHint(column)))
224        for column in range(header.count()):
225            if header.sectionSize(column) > MAX_TEXT_WIDTH:
226                header.resizeSection(column, MAX_TEXT_WIDTH)

256    def show_metrics(self, files: list[cx.FileMetrics], project_dir: Path, test_dirs: list[Path] | None = None,
257                     dependency_dirs: list[Path] | None = None, description: str = ""):
258        """Every measured file's metrics (the project's tests and the
259        dependencies' are listed only with their boxes ticked)."""
260        place = (project_dir.resolve(), [Path(d).resolve() for d in test_dirs or []],
261                 [Path(d).resolve() for d in dependency_dirs or []])
262        if place != (self.project_dir, self.test_dirs, self.dependency_dirs):
263            self._labels.clear()
264            self._is_test.clear()
265        self.files = files
266        self.project_dir, self.test_dirs, self.dependency_dirs = place
267        self.description = description
268        self._fill()

270    def shown_files(self) -> list[cx.FileMetrics]:
271        """The files listed: the project's own, with its tests if ticked
272        (the dependencies' are only measured when their box is ticked)."""
273        if self.tests_check.isChecked():
274            return list(self.files)

277    def _test_file(self, path: Path) -> bool:
278        found = self._is_test.get(path)
279        if found is None:
280            resolved = path.resolve()
281            found = self._is_test[path] = any(resolved.is_relative_to(d) for d in self.test_dirs)

284    def label(self, path: Path) -> str:
285        """A file's name as listed: its path in the project (or in its dependency)."""

291    def _label(self, path: Path) -> str:
292        path = path.resolve()
293        for top in (*self.dependency_dirs, self.project_dir):
294            if top is not None and path.is_relative_to(top):
295                relative = path.relative_to(top).as_posix()
296                return relative if top == self.project_dir else f"{top.name}/{relative}"
297        return str(path)

300        files = self.shown_files()

312    def _update_totals(self, files: list[cx.FileMetrics]):
313        self.subtitle.setText(" · ".join(part for part in (self.description, f"{len(files)} files") if part))
314        procedures = [p for f in files for p in f.procedures]
315        over = [p for p in procedures if p.warnings(self.thresholds)]
316        nloc = sum(f.nloc for f in files)
317        average_ccn = sum(p.ccn for p in procedures) / len(procedures) if procedures else 0.0
318        average_nloc = sum(p.nloc for p in procedures) / len(procedures) if procedures else 0.0
319        self.totals.setText(f"{nloc:,} lines of code · {len(procedures):,} procedures · average CCN "
320                            f"{average_ccn:.1f}, NLOC {average_nloc:.0f} · "
321                            + (f"<span style='color:{OVER}'>{len(over)} over a threshold</span>" if over
322                               else "none over a threshold"))
323        self.totals.setToolTip("Thresholds (lizard's): " + ", ".join(
324            f"{key.upper() if key == 'ccn' else key} {value}" for key, value in self.thresholds.items()))

334    def _listed(self, files: list[cx.FileMetrics]) -> list[tuple[cx.FileMetrics, cx.Procedure, str]]:
335        """(file, procedure, file's label) of the files' procedures the filter
336        lets through (a file chosen: only that file's)."""
337        needle = self.filter_edit.text().strip().lower()
338        chosen = self.chosen_file()
339        rows = []
340        for f in files:
341            label = self.label(f.path)
342            if chosen is not None and label != chosen:  # (only that file's: not another path it's a part of)
343                continue
344            for proc in f.procedures:
345                if chosen is None and needle and needle not in proc.qualified.lower() and needle not in label.lower():
346                    continue
347                rows.append((f, proc, label))
348        return rows

350    def _fill_procedures(self):
351        rows = [ProcedureRow(f, proc, label, frozenset(proc.warnings(self.thresholds)))
352                for f, proc, label in self._listed(self.shown_files())]
353        self.procedures.setColumnHidden(FILE_COLUMN, self.chosen_file() is not None)  # (one file: no need to say which)
354        table = self.procedures
355        table.blockSignals(True)  # (refilling isn't choosing a procedure: see the end)
356        table.set_rows(rows)
357        self._fit_columns(table)
358        table.blockSignals(False)
359        self._keep_flow()

388    def _keep_flow(self):
389        """After the tables are filled again (a save, a filter, a file chosen):
390        the procedure shown stays selected, its graph drawn again from its file
391        (it may have changed); filtered out, or a file just chosen, the first
392        procedure listed is selected and shown instead; gone from the project,
393        the graph goes too."""
394        shown = self.flow_shown
395        row = self._row_of(shown.path, shown.qualified) if shown is not None else None
396        if row is None and self.procedures.rowCount() and (shown is not None or self.chosen_file()):
397            row = 0  # (filtered out, or a file chosen: its most complex procedure)
398        if row is None and shown is None:
399            return
400        if row is None:
401            if not any(p.path == shown.path and p.qualified == shown.qualified
402                       for f in self.files for p in f.procedures):
403                self.flow_shown = None
404                self.flow_title.setText("")
405                self.flow.show_message("Select a procedure to see its control flow.")
406            return
407        self.procedures.blockSignals(True)
408        self.procedures.setCurrentCell(row, NAME_COLUMN)
409        self.procedures.blockSignals(False)
410        self._show_flow(row, again=True)

412    def _show_flow(self, row: int, again: bool = False):
413        """The control-flow graph of the procedure in ``row``."""
414        item = self.procedures.item(row, NAME_COLUMN) if row >= 0 else None
415        proc = item.data(PROCEDURE_ROLE) if item is not None else None
416        if proc is None or (not again and proc is self.flow_shown):
417            return
418        self.flow_shown = proc
419        try:
420            text = read_source(proc.path).text
421        except OSError as error:
422            self.flow.show_message(f"Couldn't read {proc.path.name}: {error}")
423            return
424        graph = cf.build(text, proc.path, proc)
425        decisions = graph.complexity - 1
426        self.flow_title.setText(f"<b>{html.escape(proc.name)}</b> · lines {proc.start}–{proc.end} · "
427                                f"{decisions} decision{'s' * (decisions != 1)} · click a node to open it, "
428                                "right-click for more")
429        self.flow.show_graph(graph, text)

435    def _procedure_tip(self, proc: cx.Procedure, label: str, over: set[str]) -> str:
436        lines = [f"<b>{html.escape(proc.qualified)}</b> ({proc.kind})",
437                 f"{html.escape(label)}, lines {proc.start}–{proc.end}"]
438        if over:
439            names = {"ccn": "CCN", "length": "length", "params": "parameters", "depth": "depth"}
440            over_text = ", ".join(f"{names[key]} (over {self.thresholds[key]})" for key in sorted(over))
441            lines.append(f"<span style='color:{OVER}'>Over: {over_text}</span>")
442        return "<br>".join(lines)

444    def _fill_files(self, files: list[cx.FileMetrics]):
445        table = self.file_table
446        chosen = self.chosen_file()
447        table.blockSignals(True)  # (refilling isn't choosing a file)
448        table.set_rows([FileRow(f, self.label(f.path), sum(1 for p in f.procedures if p.warnings(self.thresholds)),
449                                max((p.ccn for p in f.procedures), default=0)) for f in files])
450        self._fit_columns(table)
451        table.setCurrentCell(-1, -1)
452        row = next((n for n, r in enumerate(table.rows()) if r.label == chosen), None)
453        if row is not None:  # (still chosen: a save refills the tables)
454            table.setCurrentCell(row, 0)
455        table.blockSignals(False)

335    def _listed(self, files: list[cx.FileMetrics]) -> list[tuple[cx.FileMetrics, cx.Procedure, str]]:
336        """(file, procedure, file's label) of the files' procedures the filter
337        lets through (a file chosen: only that file's)."""

339        chosen = self.chosen_file()

341        for f in files:
342            label = self.label(f.path)
343            if chosen is not None and label != chosen:  # (only that file's: not another path it's a part of)
344                continue
345            for proc in f.procedures:
346                if chosen is None and needle and needle not in proc.qualified.lower() and needle not in label.lower():
347                    continue
348                rows.append((f, proc, label))

389    def _keep_flow(self):
390        """After the tables are filled again (a save, a filter, a file chosen):
391        the procedure shown stays selected, its graph drawn again from its file
392        (it may have changed); filtered out, or a file just chosen, the first
393        procedure listed is selected and shown instead; gone from the project,
394        the graph goes too."""
395        shown = self.flow_shown
396        row = self._row_of(shown.path, shown.qualified) if shown is not None else None
397        if row is None and self.procedures.rowCount() and (shown is not None or self.chosen_file()):
398            row = 0  # (filtered out, or a file chosen: its most complex procedure)
399        if row is None and shown is None:
400            return
401        if row is None:
402            if not any(p.path == shown.path and p.qualified == shown.qualified
403                       for f in self.files for p in f.procedures):
404                self.flow_shown = None
405                self.flow_title.setText("")
406                self.flow.show_message("Select a procedure to see its control flow.")
407            return
408        self.procedures.blockSignals(True)
409        self.procedures.setCurrentCell(row, NAME_COLUMN)
410        self.procedures.blockSignals(False)
411        self._show_flow(row, again=True)

413    def _show_flow(self, row: int, again: bool = False):
414        """The control-flow graph of the procedure in ``row``."""
415        item = self.procedures.item(row, NAME_COLUMN) if row >= 0 else None
416        proc = item.data(PROCEDURE_ROLE) if item is not None else None
417        if proc is None or (not again and proc is self.flow_shown):
418            return
419        self.flow_shown = proc
420        try:
421            text = read_source(proc.path).text
422        except OSError as error:
423            self.flow.show_message(f"Couldn't read {proc.path.name}: {error}")
424            return
425        graph = cf.build(text, proc.path, proc)
426        decisions = graph.complexity - 1
427        self.flow_title.setText(f"<b>{html.escape(proc.name)}</b> · lines {proc.start}–{proc.end} · "
428                                f"{decisions} decision{'s' * (decisions != 1)} · click a node to open it, "
429                                "right-click for more")
430        self.flow.show_graph(graph, text)
