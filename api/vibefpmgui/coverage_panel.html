Clicking a file opens it at its first missed line; the editor's gutter shows which lines ran (see vibefpmgui.coverage).

 44def function_name(name: str) -> str:
 45    """gfortran's name for a procedure, as written: ``add (module calc)``."""
 46    found = MANGLED.match(name)
 47    if found:
 48        return f"{found.group(2)} (module {found.group(1)})"
 49    if name == "MAIN__":
 50        return "(main program)"
 51    return name

 54def shown_functions(file: cov.FileCoverage) -> list[cov.FunctionCoverage]:
 55    """A file's functions without the ``main`` gfortran adds to call a
 56    program (it has no line of its own)."""
 57    has_program = any(f.name == "MAIN__" for f in file.functions)
 58    return [f for f in file.functions if not (has_program and f.name == "main")]

 61def row_tooltip(f: cov.FileCoverage) -> str:
 62    """A file's row's tooltip: its path, and every range of lines the tests
 63    missed (wrapped: a tooltip of one very long line runs off the screen)."""
 64    ranges = f.missed()
 65    if not ranges:
 66        missed = "Every line with code ran."
 67    else:
 68        lines = sum(1 for n in f.lines.values() if n == 0)
 69        missed = (f"<b>Missed lines</b> ({lines:,} in {len(ranges):,} range{'s' * (len(ranges) != 1)}):<br>"
 70                  f"{html.escape(cov.ranges_text(ranges, limit=len(ranges)))}")
 71    return f"<p style='white-space:normal'>{html.escape(str(f.path))}<br><br>{missed}</p>"

234    def shown_files(self) -> list[cov.FileCoverage]:
235        """The files listed: the project's (and the dependencies' or the
236        test programs, if asked), each with code in it."""
237        out = []
238        for path, f in self.files.items():
239            if not f.lines_found:
240                continue
241            inside = self.project_dir is not None and self.project_dir in path.parents
242            dependency = not inside or "dependencies" in path.relative_to(self.project_dir).parts[:2]
243            if dependency and not self.dependencies_check.isChecked():
244                continue
245            if not self.tests_check.isChecked() and any(d == path.parent or d in path.parents for d in self.test_dirs):
246                continue
247            out.append(f)
248        return out

250    def relative(self, path: Path) -> str:
251        try:
252            return str(path.relative_to(self.project_dir)) if self.project_dir else path.name
253        except ValueError:
254            return str(path)

256    def _fill(self, *_):
257        files = self.shown_files()
258        if not self.files:
259            return
260        if not files:
261            self.show_message("No coverage for the project's files: did the tests run code from them? "
262                              "(Include tests and Include dependencies list the others.)")
263            return
264        self.empty.hide()
265        self.splitter.show()
266        muted = self.palette_colors["muted"]
267        t = cov.totals(files)
268        functions = [f for file in files for f in shown_functions(file)]
269        parts = []
270        for name, (hit, found) in (("lines", t["lines"]),
271                                   ("functions", (sum(f.hits > 0 for f in functions), len(functions))),
272                                   ("branches", t["branches"])):
273            if found:
274                percent = 100 * hit / found
275                parts.append(f"<b style='color:{color_for(percent)}'>{percent:.1f}%</b> "
276                             f"<span style='color:{muted}'>{name} ({hit:,} of {found:,})</span>")
277        self.totals.setText("&nbsp;&nbsp;".join(parts))
278        hit, found = t["lines"]
279        self.progress.set(hit / found if found else 0, "ok" if found and hit == found else "running")
280        selected = self._selected_path()
281        self.table.setSortingEnabled(False)
282        self.table.setRowCount(len(files))
283        for row, f in enumerate(sorted(files, key=lambda f: self.relative(f.path))):
284            name = QTableWidgetItem(self.relative(f.path))  # (sorted by its text)
285            if self.icon_for is not None:
286                name.setIcon(self.icon_for(f.path))
287            name.setData(Qt.UserRole, str(f.path))
288            shown = shown_functions(f)
289            fn_hit = sum(x.hits > 0 for x in shown)
290            items = [
291                name,
292                NumberItem(f"{f.percent:.1f}%", f.percent),
293                NumberItem(f"{f.lines_hit} / {f.lines_found}", f.lines_hit),
294                NumberItem(f"{fn_hit} / {len(shown)}" if shown else "—", fn_hit / len(shown) if shown else -1),
295            ]
296            items[1].setForeground(QColor(color_for(f.percent)))
297            tip = row_tooltip(f)  # (the whole row: its path and every missed line)
298            for column, item in enumerate(items):
299                item.setToolTip(tip)
300                self.table.setItem(row, column, item)
301        self.table.setSortingEnabled(True)  # (sorted again, as the header says: the user's choice kept)
302        if selected is not None:
303            for row in range(self.table.rowCount()):
304                if self.table.item(row, 0).data(Qt.UserRole) == selected:
305                    self.table.selectRow(row)
306                    break
307        self._show_details()
308

310    def _selected_path(self) -> str | None:
311        rows = self.table.selectionModel().selectedRows() if self.table.selectionModel() else []
312        return self.table.item(rows[0].row(), 0).data(Qt.UserRole) if rows else None

314    def _show_details(self):
315        path = self._selected_path()
316        f = self.files.get(Path(path)) if path else None
317        if f is None:
318            self.detail_title.setText("Select a file for its functions.")
319            self.functions.setRowCount(0)
320            return
321        self.detail_title.setText(f"<b>{html.escape(self.relative(f.path))}</b> · {f.percent:.1f}% of its "
322                                  f"{f.lines_found} lines with code ran")
323        shown = shown_functions(f)
324        try:  # (what each function is, from its first line: the outline's badge for it)
325            source = f.path.read_text(errors="replace").split("\n")
326        except OSError:
327            source = []
328        self.functions.setSortingEnabled(False)
329        self.functions.setRowCount(len(shown))
330        for row, fn in enumerate(shown):
331            name = QTableWidgetItem(function_name(fn.name))
332            kind = cov.procedure_kind(source, fn.line)
333            if kind:
334                name.setIcon(symbol_icon(kind))
335                name.setToolTip(kind)
336            name.setData(Qt.UserRole, fn.line)
337            calls = NumberItem(f"{fn.hits:,}" if fn.hits else "never", fn.hits)
338            calls.setForeground(QColor(GOOD if fn.hits else POOR))
339            for column, item in enumerate((name, NumberItem(str(fn.line), fn.line), calls)):
340                self.functions.setItem(row, column, item)
341        self.functions.setSortingEnabled(True)
342        self.functions.sortItems(1)
343
344    def _row_clicked(self, row: int, _column: int):
345        path = self.table.item(row, 0).data(Qt.UserRole)
346        f = self.files.get(Path(path))
347        if f is not None:
348            missed = f.missed()
349            self.open_requested.emit(path, missed[0][0] if missed else min(f.lines, default=1))
350
351    def _function_clicked(self, row: int, _column: int):
352        path = self._selected_path()
353        if path:
354            self.open_requested.emit(path, self.functions.item(row, 0).data(Qt.UserRole))
