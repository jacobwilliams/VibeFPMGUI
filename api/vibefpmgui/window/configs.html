vibefpmgui.window.configs

Build configurations (see vibefpmgui.build_configs): the Configuration list at the top of Build options (its ⋯ menu opens another configuration file), and File → Save Build Configuration…. (A project opened with File → Open Project… or Recent Projects comes with its vibefpm.json: see vibefpmgui.window.project.)

"New configuration" is the Build options as before: editable, kept in the app's settings. Choosing a saved configuration shows its options, read-only (the fpm executable stays editable: it isn't part of one); choosing New configuration again brings back what was there. Save… adds what's shown to the project's configuration file, as a configuration chosen from then on. Edit makes a saved configuration's options editable, and Save puts them back in the file under its name; leaving it (choosing another, opening another project or file, quitting) with changes not saved asks first.

168    def _config_chosen(self, index: int):
169        if self._editing:
170            editing = self._editing
171            if not self.maybe_save_configuration():
172                self.config_combo.blockSignals(True)  # (Cancel: still on it, its changes kept)
173                self.config_combo.setCurrentIndex(self._index_of(editing))
174                self.config_combo.blockSignals(False)
175                return
176            self._editing = ""
177        if index <= 0:
178            if self._free_options is not None:
179                free, self._free_options = self._free_options, None
180                self._show_options(free)
181        else:
182            config = self._configuration(self.config_combo.itemText(index))
183            if config is None:
184                return
185            if self._free_options is None:
186                self._free_options = self.current_options()
187            self._show_options(config)
188        self._set_options_locked(index > 0)
189        if self.manifest_path is not None:
190            self.settings.setValue("build_configuration", self.config_combo.currentText() if index > 0 else "")
191        self._update_config_ui()

207    def _load_configurations(self, reread: bool = True):
208        """The list of the project's configurations (the project was opened
209        or reloaded): its configurations read again (not for a file just
210        opened, or ``reread`` False: just read), the one shown kept (for a
211        file just opened, the one shown last time); one being edited keeps
212        its changes."""
213        file = self.config_file
214        if file is None or self.manifest_path is None:
215            return
216        first = file is not self._listed_file
217        if reread and not first:
218            file.reread_configurations()  # (not its options: they're the window's until written)
219        self._listed_file = file
220        if self._editing and not first:
221            self._refill_items(self._editing)
222            return
223        if first:
224            self._editing = ""  # (asked about before the project changed: see maybe_save_configuration)
225            wanted = self.settings.value("build_configuration", "") or ""
226        else:
227            wanted = self.config_combo.currentText() if self.options_locked() else ""
228        if self._config_error:
229            self.statusBar().showMessage(f"Build configurations: {self._config_error}", 8000)
230        self._fill_config_combo(wanted)

232    def reload_configurations(self):
233        """The configuration file was saved in the editor: its options, if
234        they were changed there, applied, and its configurations listed again
235        (the one shown kept: with its options as the file has them now,
236        unless it's being edited: its changes are kept)."""
237        if self.config_file is None or self.manifest_path is None:
238            return
239        self.reload_config_file()  # (it reads it all)
240        self._load_configurations(reread=False)
241        if self._config_error:
242            return  # (the list says why; it's empty until the file is fixed)
243        count = len(self.configurations)
244        self.statusBar().showMessage(f"Build configurations read again from {self.config_path.name}: {count} "
245                                     f"configuration{'s' * (count != 1)}", 5000)

247    def _fill_config_combo(self, wanted: str = ""):
248        """The list again, showing ``wanted`` (New configuration if it's gone):
249        back to New configuration first (its options restored), then to
250        ``wanted`` (its options read again: the file may have changed)."""
251        self.select_configuration("")
252        self._refill_items()
253        if wanted and self._configuration(wanted) is not None:
254            self.select_configuration(wanted)
255        self._update_config_ui()
256
257    def _refill_items(self, shown: str = ""):
258        """The list's names again, ``shown`` chosen (no options changed)."""
259        combo = self.config_combo
260        combo.blockSignals(True)
261        while combo.count() > 1:
262            combo.removeItem(1)
263        names = [config.name for config in self.configurations]
264        combo.addItems(names + ([shown] if shown and shown not in names else []))  # (being edited: kept)
265        combo.setCurrentIndex(self._index_of(shown))
266        combo.blockSignals(False)
267        self._update_config_ui()

315    def save_configuration(self) -> bool:
316        """Build options → Save (File → Save Build Configuration…): the
317        configuration being edited, under its name; otherwise Save As…."""
318        if not self._editing:
319            return self.save_configuration_as()
320        name = self._editing
321        if not self._write_configuration(self.current_options(name)):
322            return False
323        self._editing = ""
324        self._fill_config_combo(name)
325        return True

327    def _config_file_for_saving(self) -> bc.ConfigFile | None:
328        """The file to save in (None: it can't be read, so isn't written over)."""
329        file = self.config_file
330        if file is None:
331            return None
332        if file.error and file.exists():
333            QMessageBox.warning(self, "Save Build Configuration",
334                                f"{file.error}\n\nFix the file (or move it away) to save configurations in it.")
335            return None
336        return file

338    def _write_configuration(self, config: bc.Configuration) -> bool:
339        file = self._config_file_for_saving()
340        if file is None:
341            return False
342        try:
343            file.save(config)
344        except (OSError, bc.ConfigError) as exc:
345            QMessageBox.warning(self, "Save Build Configuration", f"Couldn't save {file.path}:\n\n{exc}")
346            return False
347        self._remember_config_file(self.manifest_path, file.path)
348        self._config_file_saved()  # (a file just made: the project's options go in it from now on)
349        self.statusBar().showMessage(f"Saved build configuration “{config.name}” in {file.path.name}", 5000)
350        return True

352    def save_configuration_as(self) -> bool:
353        """Save… / ⋯ → Save As…: what's shown, as a configuration in the
354        project's file under a name asked for, chosen from then on."""
355        if self.manifest_path is None:
356            return False
357        file = self._config_file_for_saving()
358        if file is None:
359            return False
360        suggestion = self.config_combo.currentText() if self.config_shown() else ""
361        name, ok = QInputDialog.getText(self, "Save Build Configuration",
362                                        f"Save these Build options in {file.path.name} as:", text=suggestion)
363        name = name.strip()
364        if not ok or not name:
365            return False
366        if name == NEW_CONFIGURATION:
367            QMessageBox.warning(self, "Save Build Configuration", f"“{NEW_CONFIGURATION}” is taken: choose "
368                                "another name.")
369            return False
370        if self._configuration(name) is not None and name != self._editing and QMessageBox.question(
371                self, "Save Build Configuration",
372                f"{file.path.name} already has a configuration named “{name}”. Replace it?") != QMessageBox.Yes:
373            return False
374        if not self._write_configuration(self.current_options(name)):
375            return False
376        self._editing = ""  # (saved: under another name, the one edited is left as it was in the file)
377        self._fill_config_combo(name)
378        return True

394    def cancel_editing(self):
395        """Cancel: the configuration as saved, read-only again (changes dropped)."""
396        if not self._editing:
397            return
398        name, self._editing = self._editing, ""

407    def maybe_save_configuration(self) -> bool:
408        """Before leaving a configuration being edited with unsaved changes
409        (choosing another, opening another project or file, quitting): Save,
410        Discard (it's shown as saved again: what follows may not happen) or
411        Cancel. False: Cancel (or saving failed): stay."""
412        if not self.has_unsaved_configuration():
413            return True
414        answer = QMessageBox.question(
415            self, "Unsaved Build Configuration",
416            f"Save the changes to the build configuration “{self._editing}” in {self.config_path.name}?",
417            QMessageBox.Save | QMessageBox.Discard | QMessageBox.Cancel, QMessageBox.Save)
418        if answer == QMessageBox.Cancel:
419            return False
420        if answer == QMessageBox.Save:
421            name = self._editing
422            if not self._write_configuration(self.current_options(name)):
423                return False
424        else:
425            self.cancel_editing()
426        return True

439    def delete_configuration(self):
440        if not self.config_shown() or self.config_path is None:
441            return
442        name = self.config_combo.currentText()
443        if QMessageBox.question(self, "Delete Build Configuration",

463    def open_configuration(self, filename: str | Path):
464        """Open a configuration file: its project (the fpm.toml beside it),
465        if it isn't the one open, and its first configuration."""
466        path = Path(filename).resolve()
467        manifest = bc.manifest_for(path)
468        if manifest is None:
469            QMessageBox.warning(self, "Open Build Configuration",
470                                f"There's no fpm.toml beside {path.name}: a configuration file goes in the "
471                                "project's folder, next to its fpm.toml.")
472            return
473        opened = bc.ConfigFile(path).read()
474        if opened.error:
475            QMessageBox.warning(self, "Open Build Configuration", opened.error)
476            return
477        if not self.maybe_save_configuration():
478            return
479        manifest = manifest.resolve()
480        key = project_key(manifest, "build_config_file")
481        before = self.settings.app.value(key, "")
482        self._remember_config_file(manifest, path)  # (the file the project is opened with)
483        if manifest != self.manifest_path:
484            self.open_manifest(manifest)
485            if self.manifest_path != manifest:  # (not opened: Cancel, or fpm running)
486                self.settings.app.setValue(key, before)
487                return
488        elif path != self.config_path:
489            self._switch_config_file(manifest, force=True)
490            self._load_configurations()
491        configs = self.configurations
492        self._fill_config_combo(configs[0].name if configs else "")
493        self.statusBar().showMessage(f"{path.name}: {len(configs)} configuration"
494                                     f"{'s' * (len(configs) != 1)}", 5000)

496    def _show_config_file(self):
497        if self.config_path and self.config_path.is_file():
498            self.open_file(str(self.config_path), force_editor=True)
