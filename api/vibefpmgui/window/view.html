389    def set_font_delta(self, delta: int, announce: bool = True):
390        """Make all text in the window ``delta`` points bigger (or smaller)
391        than the system size."""
392        lo, hi = self.FONT_DELTA_RANGE
393        self.font_delta = max(lo, min(hi, int(delta)))
394        base = self._base_font.pointSizeF()
395        size = base + self.font_delta
396        factor = size / base
397        set_ui_scale(factor)
398
399        font = QFont(self._base_font)
400        font.setPointSizeF(size)
401        QApplication.setFont(font)
402        # Widgets with their own fonts don't follow the application font.
403        mono = self._mono_font()
404        for view in (self.console, self.test_output):
405            view.setFont(mono)
