 82    def tabSizeHint(self, index):
 83        size = super().tabSizeHint(index)
 84        name = self.italic_name(index)
 85        if name is not None:  # (room for the name it doesn't have as text)
 86            size.setWidth(size.width() + QFontMetrics(self._italic_font()).horizontalAdvance(name) + 4)
 87        # No more room around the name than NAME_GAP on each side: some styles
 88        # (macOS's) leave its buttons at the tab's edges and center the name in
 89        # a much wider space. As wide as the style leaves for the name, less
 90        # what's spare (a style that leaves none, as Fusion, is unchanged).
 91        option = QStyleOptionTab()
 92        self.initStyleOption(option, index)
 93        option.rect = QRect(QPoint(0, 0), size)
 94        room = self.style().subElementRect(QStyle.SE_TabBarTabText, option, self).width()
 95        font = self._italic_font() if name is not None else self.font()
 96        spare = room - QFontMetrics(font).horizontalAdvance(self.tab_name(index)) - 2 * self.NAME_GAP
 97        if spare > 0:
 98            size.setWidth(size.width() - spare)
 99        return size

101    def paintEvent(self, event):
102        super().paintEvent(event)
103        names = [(i, self.italic_name(i)) for i in range(self.count())]
104        if not any(name for _i, name in names):
105            return
106        painter = QStylePainter(self)
107        painter.setFont(self._italic_font())
108        for index, name in names:
109            if name is not None:
110                option = QStyleOptionTab()
111                self.initStyleOption(option, index)
112                option.text = name
113                painter.drawControl(QStyle.CE_TabBarTabLabel, option)

122    def wheelEvent(self, event):
123        arrows = [b for b in self.findChildren(QToolButton) if b.isVisible() and b.arrowType() in
124                  (Qt.LeftArrow, Qt.RightArrow)]
125        if not arrows:
126            super().wheelEvent(event)
127            return
128        delta = event.angleDelta()
129        step = delta.x() if abs(delta.x()) > abs(delta.y()) else delta.y()
130        self._wheel_rest = getattr(self, "_wheel_rest", 0) + step
131        while abs(self._wheel_rest) >= 120:  # one notch (or that much trackpad travel) per tab
132            back = self._wheel_rest > 0
133            self._wheel_rest -= 120 if back else -120
134            for button in arrows:
135                if button.arrowType() == (Qt.LeftArrow if back else Qt.RightArrow) and button.isEnabled():
136                    button.click()
137        event.accept()

277    def open_file(self, path: str, diagnostics: list | None = None, force_editor: bool = False,
278                  preview: bool = False):
279        """Open a file in the built-in editor (or the default application),
280        marking any compiler diagnostics and jumping to the first one.
281        ``preview``: in the preview tab (see preview_file), and the keyboard
282        stays where it is."""
283        file = Path(path).resolve()
284        editor = next((e for e in self.editors() if e.path == file), None)  # (open already: there, even unsaved)
285        use_editor = force_editor or self.use_editor_action.isChecked()
286        if editor is None and (not use_editor or not file.is_file() or not EditorTab.is_probably_text(file)):
287            QDesktopServices.openUrl(QUrl.fromLocalFile(str(file)))
288            return
289        if editor is not None and not preview:
290            self._make_permanent(editor)  # (opened for real: a preview no longer)
291        if editor is None:
292            try:
293                editor = EditorTab(file)
294            except OSError as exc:
295                QMessageBox.warning(self, "Can't open file", f"{file}:\n{exc}")
296                return
297            # A new preview takes the old one's place (unless that one has changes: then it stays).
298            replaced = self._preview_editor() if preview else None
299            index = self.tabs.indexOf(replaced) if replaced is not None and not replaced.is_dirty() else -1
300            if preview:
301                self._preview = editor
302            self._add_editor(editor, index)
303            if index >= 0:
304                self._close_tab(replaced)
305        self.tabs.setCurrentWidget(editor)
306        if diagnostics:
307            editor.show_diagnostics(diagnostics)
308            editor.editor.go_to_line(diagnostics[0].line)
309        elif not preview:
310            editor.editor.setFocus()

451    def rename_file(self, path: str):
452        """Project files → Rename…: rename a file on disk; a tab it's open in
453        follows it (unsaved changes included)."""
454        old = Path(path).resolve()
455        name, ok = QInputDialog.getText(self, "Rename", f"New name for {old.name}:", text=old.name)
456        name = name.strip() if ok else ""
457        if not name or name == old.name:
458            return
459        if "/" in name or "\\" in name or name in (".", ".."):
460            QMessageBox.warning(self, "Rename", f"“{name}” isn't a file name (it can't contain / or \\).")
461            return
462        new = old.with_name(name)
463        if new.exists() and not os.path.samefile(new, old):  # (another file; a change of case is fine)
464            QMessageBox.warning(self, "Rename", f"There's already a file named {name} in {old.parent.name}.")
465            return
466        try:
467            old.rename(new)
468        except OSError as exc:
469            QMessageBox.warning(self, "Rename", f"Couldn't rename {old.name}:\n{exc}")
470            return
471        editor = self._open_editor_for(old)
472        if editor is not None:
473            editor.renamed(new)
474            self._save_session()
475        self._lsp_diagnostics_by_file.pop(old, None)
476        self.search_panel.file_changed(old)
477        self.search_panel.file_changed(new)
478        self.project_tree.refresh()
479        self.project_tree.reveal(str(new))
480        self.statusBar().showMessage(f"Renamed {old.name} to {name}", 4000)

482    def delete_file(self, path: str):
483        """Project files → Move to Trash…: after asking (and, for a file open
484        with unsaved changes, whether to save them first), move a file to the
485        trash and close its tab."""
486        file = Path(path).resolve()
487        editor = self._open_editor_for(file)
488        if editor is not None and editor.is_dirty():
489            self.tabs.setCurrentWidget(editor)
490            answer = QMessageBox.question(
491                self, "Move to Trash",
492                f"{file.name} has unsaved changes. Save them before moving it to the trash?",
493                QMessageBox.Save | QMessageBox.Discard | QMessageBox.Cancel, QMessageBox.Cancel)
494            if answer == QMessageBox.Cancel or (answer == QMessageBox.Save and not editor.save()):

279                  preview: bool = False):
280        """Open a file in the built-in editor (or the default application),
281        marking any compiler diagnostics and jumping to the first one.
282        ``preview``: in the preview tab (see preview_file), and the keyboard
283        stays where it is."""
284        file = Path(path).resolve()
285        editor = next((e for e in self.editors() if e.path == file), None)  # (open already: there, even unsaved)

288            QDesktopServices.openUrl(QUrl.fromLocalFile(str(file)))
289            return
290        if editor is not None and not preview:
291            self._make_permanent(editor)  # (opened for real: a preview no longer)
292        if editor is None:
293            try:
294                editor = EditorTab(file)
295            except OSError as exc:
296                QMessageBox.warning(self, "Can't open file", f"{file}:\n{exc}")
297                return
298            # A new preview takes the old one's place (unless that one has changes: then it stays).
299            replaced = self._preview_editor() if preview else None
300            index = self.tabs.indexOf(replaced) if replaced is not None and not replaced.is_dirty() else -1
301            if preview:
302                self._preview = editor
303            self._add_editor(editor, index)
304            if index >= 0:
305                self._close_tab(replaced)
306        self.tabs.setCurrentWidget(editor)
307        if diagnostics:
308            editor.show_diagnostics(diagnostics)
309            editor.editor.go_to_line(diagnostics[0].line)
310        elif not preview:
311            editor.editor.setFocus()
