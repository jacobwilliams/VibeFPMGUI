 29def _signature(graph: m.DependencyGraph):
 30    return (sorted((p.name, p.kind, p.version) for p in graph.packages.values()),
 31            sorted((e.parent, e.child, e.dev) for e in graph.edges))

143    def open_project(self, folder: str | Path, new_window: bool = False) -> bool:
144        """Open the project in ``folder``: with its vibefpm.json (its build
145        configurations and options) if it has one, else its fpm.toml; a
146        folder without an fpm.toml isn't one (an error message). False: not
147        opened (that, or Cancel)."""
148        folder = Path(folder).expanduser().resolve()
149        manifest, config = folder / "fpm.toml", folder / bc.FILE_NAME
150        if not manifest.is_file():
151            also = (f" It has a {bc.FILE_NAME}, but that goes beside a project's fpm.toml."
152                    if config.is_file() else "")
153            QMessageBox.warning(self, "Open Project", f"{folder}\n\nisn't an fpm project: it has no fpm.toml.{also}")
154            return False
155        key = project_key(manifest, "build_config_file")
156        before = self.settings.app.value(key, "")
157        if config.is_file():
158            self.settings.app.setValue(key, "")  # (its vibefpm.json: not another file opened for it before)
159        if new_window:
160            self.open_in_new_window(manifest)
161            return True
162        if manifest == self.manifest_path:
163            if config.is_file() and config != self.config_path:  # (it had another configuration file)
164                if not self.maybe_save_configuration():
165                    self.settings.app.setValue(key, before)
166                    return False
167                self._switch_config_file(manifest, force=True)
168                self._load_configurations()
169            return True
170        self.open_manifest(manifest)
171        if self.manifest_path != manifest:  # (not opened: Cancel, or fpm running)
172            self.settings.app.setValue(key, before)
173            return False
174        return True

176    def open_in_new_window(self, manifest: str | Path):
177        """Open a project in another VibeFPMGUI window (a separate process, so
178        each window has its own fpm runs, language server and so on), keeping
179        this one as it is."""
180        manifest = Path(manifest).resolve()
181        process = QProcess()
182        process.setProgram(gui_python())  # (on Windows pythonw: no console window beside it)
183        process.setArguments(["-m", "vibefpmgui", "--new-window", str(manifest)])
184        process.setWorkingDirectory(str(manifest.parent))
185        # Make sure the new process finds this package even when it isn't
186        # installed (run from a checkout, e.g. "pixi run start").
187        env = QProcessEnvironment.systemEnvironment()
188        package_root = str(Path(__file__).resolve().parents[2])  # (the folder vibefpmgui is in)
189        env.insert("PYTHONPATH", os.pathsep.join(p for p in (package_root, env.value("PYTHONPATH")) if p))
190        process.setProcessEnvironment(env)
191        ok = process.startDetached()
192        ok = ok[0] if isinstance(ok, tuple) else ok  # (PySide returns (ok, pid))
193        if ok:
194            self.statusBar().showMessage(f"Opening {manifest.parent.name} in a new window…", 5000)
195        else:
196            QMessageBox.warning(self, "Open in New Window",
197                                f"Could not start a new VibeFPMGUI window for {manifest}.")

199    def open_manifest(self, filename: str | Path):
200        path = Path(filename).resolve()
201        new_project = path != self.manifest_path
202        if new_project:
203            # Every question before anything changes: No or Cancel keeps the
204            # current project as it is, its files still open.
205            if not self.maybe_save_configuration():
206                return
207            if self.busy():
208                answer = QMessageBox.question(
209                    self, "fpm is running",
210                    "An fpm command is still running for the current project. Stop it and open the new one?")
211                if answer != QMessageBox.Yes:
212                    return
213            self._save_session()  # the old project's open files, for next time
214            # The open files belong to the old project: ask about unsaved changes.
215            if not self.close_all_editors():
216                return
217            if self.busy():
218                self._stop_and_wait(3000)
219            self._clear_run_panels(show_build=self.manifest_path is not None)
220            self.search_panel.reset()
221            self.references_panel.reset()
222            self._switch_config_file(path)  # (this one's options written, the other's applied: see window.options)
223        self.manifest_path = path
224        if self.remember_last:
225            self.settings.setValue("last_manifest", str(self.manifest_path))
226        self._remember_project(self.manifest_path.parent)
227        self.settings.sync()  # write now, so it survives the app being killed
228        self.setWindowFilePath(str(self.manifest_path))  # macOS title-bar proxy icon
229        self.update_window_title()
230        self.reload()
231        if new_project:
232            self._restore_session()

245    def recent_projects(self) -> list[str]:
246        """The folders of the projects last opened, latest first (File → Recent Projects)."""
247        if not self.settings.contains("recent_projects"):  # (kept before as fpm.toml and .json files: their folders)
248            old = [str(Path(p).parent) for key in ("recent_manifests", "recent_configs")
249                   for p in self.settings.value(key, [], type=list) or [] if p]
250            self.settings.setValue("recent_projects", list(dict.fromkeys(old))[:self.MAX_RECENT])
251            for key in ("recent_manifests", "recent_configs"):
252                self.settings.remove(key)
253        value = self.settings.value("recent_projects", [], type=list) or []
254        return [str(p) for p in value if p]

301    def _save_session(self):
302        if self.manifest_path is None or self._session_busy:
303            return
304        files = []
305        for editor in self.editors():
306            if editor.untitled:
307                continue  # (nothing on disk to open next time)
308            cursor = editor.editor.textCursor()
309            files.append({"path": str(editor.path), "line": cursor.blockNumber() + 1,
310                          "column": cursor.positionInBlock(),
311                          "scroll": editor.editor.verticalScrollBar().value()})
312        current = self.current_editor()
313        state = {"files": files, "current": str(current.path) if current and not current.untitled else None}
314        tree = self.project_tree.view_state()
315        if tree is None:  # filtered: keep what was saved before
316            try:
317                tree = json.loads(self.settings.value(self._session_key(), "{}")).get("tree")
318            except (TypeError, ValueError):
319                tree = None
320        state["tree"] = tree
321        state["breakpoints"] = self.session_breakpoints()  # (see window.debugging)
322        self.settings.setValue(self._session_key(), json.dumps(state))
323
324    def _restore_session(self):
325        try:
326            state = json.loads(self.settings.value(self._session_key(), "") or "{}")
327        except (TypeError, ValueError):
328            state = {}
329        self.set_session_breakpoints(state.get("breakpoints"))  # (before the files: they show theirs)
330        if state.get("tree"):
331            self.project_tree.restore_view_state(state["tree"])
332        self._session_busy = True
333        try:
334            for f in state.get("files") or []:
335                path = Path(f.get("path", ""))
336                if not path.is_file():
337                    continue  # deleted or moved since
338                self.open_file(str(path), force_editor=True)
339                editor = self._open_editor_for(path)
340                if editor is None:
341                    continue
342                editor.editor.go_to_position(int(f.get("line", 1)), int(f.get("column", 0)))
343                scroll = int(f.get("scroll", 0))
344                # (after the tab's first layout, or the scroll position is clamped)
345                QTimer.singleShot(0, editor, lambda e=editor, v=scroll: e.editor.verticalScrollBar().setValue(v))
346        finally:
347            self._session_busy = False
348        current = self._open_editor_for(Path(state["current"])) if state.get("current") else None
349        if current is not None:
350            self.tabs.setCurrentWidget(current)

386    def reload(self):
387        if not self.manifest_path:
388            return
389        try:
390            self.graph = m.resolve(self.manifest_path,
391                                   dependencies_dev=self.dependencies_dev_action.isChecked())
392        except (OSError, tomllib.TOMLDecodeError) as exc:
393            self.graph = None
394            self.graph_view.clear(f"Could not read {self.manifest_path.name}:\n{exc}")
395            self.details.clear()
396            self._set_project_enabled(False)
397            self.statusBar().showMessage(f"Error reading manifest: {exc}")
398            return
399        # Keep the current layout (and any nodes the user moved) unless the
400        # set of packages or their relationships actually changed.
401        if self.graph_view.graph is None or _signature(self.graph_view.graph) != _signature(self.graph):
402            self.graph_view.set_graph(self.graph)
403        else:
404            self.graph_view.update_packages(self.graph)
405        self._update_legend()  # (it mentions version conflicts when there are any)
406        self.show_package(self.graph.root)
407        self.project_tree.set_project(self.graph, self.manifest_path.parent)
408        self.terminal_panel.set_directory(self.manifest_path.parent)
409        self._update_language_server()
410        self.refresh_build_outputs()
411        self.refresh_git()
412        self.refresh_complexity()
413        self._set_project_enabled(True)
414        self._load_features_and_profiles()
415        self._load_configurations()  # (after: a configuration's profile may be one of fpm.toml's)
416        missing = sum(1 for p in self.graph.packages.values() if p.kind == m.MISSING)
417        dev_only = sum(1 for p in self.graph.packages.values() if p.kind == m.DEV_ONLY)
418        msg = f"{len(self.graph.packages) - 1 - dev_only} dependencies"
419        if missing:
420            msg += f" ({missing} not downloaded — run Update deps or Build)"
421        if dev_only:
422            msg += f" · {dev_only} of their dev-dependencies shown (fpm doesn't download those)"
423        conflicts = m.conflicts(self.graph)
424        if conflicts:
425            msg += (f" · ⚠ {len(conflicts)} version conflict{'s' * (len(conflicts) != 1)}: "
426                    + ", ".join(sorted(conflicts)) + " (see the graph)")
427        self.statusBar().showMessage(msg)

430def _run_fpm_new(command: list[str], folder: Path) -> tuple[bool, str]:
431    """(whether it worked, what fpm said) of ``fpm new`` run in ``folder`` (on a pool thread)."""
432    try:
433        code, output, message = console.run(command, timeout=120, cwd=str(folder))
434    except (OSError, TimeoutError) as error:
435        return False, str(error)
436    said = "\n".join(part for part in ((output or b"").decode("utf-8", "replace").strip(), message) if part)
437    return code == 0, said

144    def open_project(self, folder: str | Path, new_window: bool = False) -> bool:
145        """Open the project in ``folder``: with its vibefpm.json (its build
146        configurations and options) if it has one, else its fpm.toml; a
147        folder without an fpm.toml isn't one (an error message). False: not
148        opened (that, or Cancel)."""
149        folder = Path(folder).expanduser().resolve()
150        manifest, config = folder / "fpm.toml", folder / bc.FILE_NAME
151        if not manifest.is_file():
152            also = (f" It has a {bc.FILE_NAME}, but that goes beside a project's fpm.toml."
153                    if config.is_file() else "")
154            QMessageBox.warning(self, "Open Project", f"{folder}\n\nisn't an fpm project: it has no fpm.toml.{also}")
155            return False
156        key = project_key(manifest, "build_config_file")
157        before = self.settings.app.value(key, "")
158        if config.is_file():
159            self.settings.app.setValue(key, "")  # (its vibefpm.json: not another file opened for it before)
160        if new_window:
161            self.open_in_new_window(manifest)
162            return True
163        if manifest == self.manifest_path:
164            if config.is_file() and config != self.config_path:  # (it had another configuration file)
165                if not self.maybe_save_configuration():
166                    self.settings.app.setValue(key, before)
167                    return False
168                self._switch_config_file(manifest, force=True)
169                self._load_configurations()
170            return True
171        self.open_manifest(manifest)
172        if self.manifest_path != manifest:  # (not opened: Cancel, or fpm running)
173            self.settings.app.setValue(key, before)
174            return False
175        return True

297    # positions), the tab in front, and the Project files tree's open

299    def _session_key(self) -> str:
300        return "session"  # (the project's: see project_options.PER_PROJECT)
301
302    def _save_session(self):
303        if self.manifest_path is None or self._session_busy:
304            return
305        files = []
306        for editor in self.editors():
307            if editor.untitled:
308                continue  # (nothing on disk to open next time)
309            cursor = editor.editor.textCursor()
310            files.append({"path": str(editor.path), "line": cursor.blockNumber() + 1,
311                          "column": cursor.positionInBlock(),
312                          "scroll": editor.editor.verticalScrollBar().value()})
313        current = self.current_editor()
314        state = {"files": files, "current": str(current.path) if current and not current.untitled else None}
315        tree = self.project_tree.view_state()
316        if tree is None:  # filtered: keep what was saved before
317            try:
318                tree = json.loads(self.settings.value(self._session_key(), "{}")).get("tree")
319            except (TypeError, ValueError):
320                tree = None
321        state["tree"] = tree
322        state["breakpoints"] = self.session_breakpoints()  # (see window.debugging)
323        self.settings.setValue(self._session_key(), json.dumps(state))
324
325    def _restore_session(self):
326        try:
327            state = json.loads(self.settings.value(self._session_key(), "") or "{}")
328        except (TypeError, ValueError):
329            state = {}
330        self.set_session_breakpoints(state.get("breakpoints"))  # (before the files: they show theirs)
331        if state.get("tree"):
332            self.project_tree.restore_view_state(state["tree"])
333        self._session_busy = True
334        try:
335            for f in state.get("files") or []:
336                path = Path(f.get("path", ""))
337                if not path.is_file():
338                    continue  # deleted or moved since
339                self.open_file(str(path), force_editor=True)
340                editor = self._open_editor_for(path)
341                if editor is None:
342                    continue
343                editor.editor.go_to_position(int(f.get("line", 1)), int(f.get("column", 0)))
344                scroll = int(f.get("scroll", 0))
345                # (after the tab's first layout, or the scroll position is clamped)
346                QTimer.singleShot(0, editor, lambda e=editor, v=scroll: e.editor.verticalScrollBar().setValue(v))
347        finally:
348            self._session_busy = False
349        current = self._open_editor_for(Path(state["current"])) if state.get("current") else None
350        if current is not None:
351            self.tabs.setCurrentWidget(current)

387    def reload(self):
388        if not self.manifest_path:
389            return
390        try:
391            self.graph = m.resolve(self.manifest_path,
392                                   dependencies_dev=self.dependencies_dev_action.isChecked())
393        except (OSError, tomllib.TOMLDecodeError) as exc:
394            self.graph = None
395            self.graph_view.clear(f"Could not read {self.manifest_path.name}:\n{exc}")
396            self.details.clear()
397            self._set_project_enabled(False)
398            self.statusBar().showMessage(f"Error reading manifest: {exc}")
399            return
400        # Keep the current layout (and any nodes the user moved) unless the
401        # set of packages or their relationships actually changed.
402        if self.graph_view.graph is None or _signature(self.graph_view.graph) != _signature(self.graph):
403            self.graph_view.set_graph(self.graph)
404        else:
405            self.graph_view.update_packages(self.graph)
406        self._update_legend()  # (it mentions version conflicts when there are any)
407        self.show_package(self.graph.root)
408        self.project_tree.set_project(self.graph, self.manifest_path.parent)
409        self.terminal_panel.set_directory(self.manifest_path.parent)
410        self._update_language_server()
411        self.refresh_build_outputs()
412        self.refresh_git()
413        self.refresh_complexity()
414        self._set_project_enabled(True)
415        self._load_features_and_profiles()
416        self._load_configurations()  # (after: a configuration's profile may be one of fpm.toml's)
417        missing = sum(1 for p in self.graph.packages.values() if p.kind == m.MISSING)
418        dev_only = sum(1 for p in self.graph.packages.values() if p.kind == m.DEV_ONLY)
419        msg = f"{len(self.graph.packages) - 1 - dev_only} dependencies"
420        if missing:
421            msg += f" ({missing} not downloaded — run Update deps or Build)"
422        if dev_only:
423            msg += f" · {dev_only} of their dev-dependencies shown (fpm doesn't download those)"
424        conflicts = m.conflicts(self.graph)
425        if conflicts:
426            msg += (f" · ⚠ {len(conflicts)} version conflict{'s' * (len(conflicts) != 1)}: "
427                    + ", ".join(sorted(conflicts)) + " (see the graph)")
428        self.statusBar().showMessage(msg)
