218    def start_review(self):
219        editor = self.current_editor()
220        if editor is None:
221            QMessageBox.information(self, "AI review", "Open a file in the editor first "
222                                    "(double-click a file, or File → Open File in Editor).")
223            return
224        if not self._can_ask("AI review"):
225            return
226        self._ai_editor = editor
227        self._after_model = self._run_review
228        focus = ai.FOCUS_LABELS.get(self.ai_focus, self.ai_focus)
229        self._add_entry(Entry("review", focus, editor.path, editor.path.name, self.ai_model))
230        self._ensure_ollama(self._check_model)

232    def _can_ask(self, title: str) -> bool:
233        """The AI is free, and Ollama's Python package is there (if not: why)."""
234        if self._ai_thread is not None or self._ai_starting:
235            self.statusBar().showMessage("The AI is busy: try again when it's done", 5000)
236            return False
237        if not ai.ollama_available():
238            QMessageBox.warning(self, title, "The 'ollama' Python package isn't installed.\n"
239                                "Run 'pixi install' (or 'pip install ollama').")
240            return False
241        return True

252    def _remember(self, entry: Entry, turn: ctx.Turn):
253        """What the model was asked and answered: it's part of what later
254        questions send (see ai_context)."""
255        if self._request_generation != self._memory_generation:
256            return  # (asked before New Chat, Clear or another project: forgotten)
257        turn.answer, turn.edits = entry.answer, entry.edits
258        entry.turn = turn
259        self._turns.append(turn)

267    def _ensure_ollama(self, then):
268        """Start "ollama serve" if no server answers, then call ``then``
269        (Ollama is asked on a pool thread: the window never waits for it)."""
270        self._set_ai_starting(True)
271        call_in_background(ai.server_running, lambda up: self._server_checked(up, then), self)

273    def _server_checked(self, up: bool, then):
274        if up:
275            then()
276            return
277        exe = ai.ollama_executable()
278        if exe is None:
279            self._set_ai_starting(False)
280            self._end_explain_setup("Ollama isn't running")
281            QMessageBox.warning(self, "AI review", "Ollama isn't running and the 'ollama' program "
282                                "wasn't found. Run 'pixi install', or install Ollama from ollama.com.")
283            return
284        self._ai_status("starting the local Ollama server", "Starting Ollama")
285        self._ollama_process = QProcess(self)
286        self._ollama_process.setProcessChannelMode(QProcess.MergedChannels)
287        self._ollama_process.start(exe, ["serve"])
288        deadline = time.monotonic() + 20
289
290        def poll():
291            call_in_background(ai.server_running, polled, self)
292
293        def polled(up: bool):
294            if up:
295                then()
296            elif time.monotonic() > deadline or self._ollama_process.state() == QProcess.NotRunning:
297                self._set_ai_starting(False)
298                self.review_monitor.finish(False, "Could not start the Ollama server.")
299            else:
300                QTimer.singleShot(300, self, poll)
301
302        QTimer.singleShot(300, self, poll)

304    def _check_model(self):

308    def _model_checked(self, installed: list[str]):
309        self._known_models = installed
310        self._set_ai_starting(False)  # (from here a worker thread, or nothing, follows)
311        model = self.ai_model if ":" in self.ai_model else self.ai_model + ":latest"
312        if model in installed or self.ai_model in installed:
313            if self.ai_model not in self._model_contexts:  # (for the next chat question: asked once)
314                self._model_contexts[self.ai_model] = None
315                call_in_background(lambda m=self.ai_model: ai.model_context_length(m),
316                                   lambda n, m=self.ai_model: self._model_contexts.__setitem__(m, n), self)
317            self._after_model()
318            return
319        answer = QMessageBox.question(
320            self, "Download AI model",
321            f"The model '{self.ai_model}' isn't downloaded yet. Download it now?\n\n"
322            "Models are a few GB; this happens once, and everything runs locally afterwards.")
323        if answer != QMessageBox.Yes:
324            self._end_explain_setup(f"{self.ai_model} isn't downloaded")
325            return
326        self._ai_status(f"downloading {self.ai_model}", "Downloading")
327        worker = ai.ReviewWorker(ai.DEFAULT_HOST, self.ai_model, pull_only=True)
328        worker.progress.connect(self._pull_progress)
329        # (A method of the window, not a lambda: a lambda would run in the
330        # worker's thread, which is ending, and the review would never start.)
331        worker.finished.connect(self._pull_finished)
332        worker.failed.connect(self._review_failed)
333        self._start_ai_worker(worker)

351    def _run_review(self):
352        editor = self._ai_editor
353        if editor is None or editor not in self.editors():
354            self.review_monitor.finish(False, "The file was closed")
355            return
356        code = editor.editor.toPlainText()
357        diagnostics = editor.compiler_diagnostics()
358        lines = ctx.file_lines(code)
359        options = self._answer_options()
360        cap = self._chat_context()
361        room = options["num_predict"] + (ai.REASONING_TOKENS if options["think"] else 0) \
362            if options["num_predict"] else ai.REASONING_TOKENS
363
364        def prompt_for(first: int, last: int) -> str:
365            return ai.build_prompt(editor.path.name, editor.highlighter.lexer.name, "\n".join(lines[first - 1:last]),
366                                   self.ai_focus, diagnostics, first, len(lines))
367
368        # The selection (two lines or more), or the whole file if it fits the
369        # model's context with room for the answer; else the procedure at the
370        # cursor. (Sent anyway, a file too long loses its start, instructions
371        # included: Ollama doesn't refuse, it drops what doesn't fit.)
372        first, last, why = 1, len(lines), ""
373        selection = editor.editor.selected_lines()
374        if selection is not None and selection[1] > selection[0]:
375            (first, last), why = selection, "the selection"
376        elif ctx.estimate_tokens(prompt_for(1, len(lines)) + ai.SYSTEM_PROMPT, self._token_ratio) + room \
377                > cap * 0.9:
378            first, last = ctx.window(lines, None, editor.editor.textCursor().blockNumber() + 1)
379            why = "the procedure at the cursor: the file is too long to review at once"
380        prompt = prompt_for(first, last)
381        entry = self._ai_entry
382        if why:
383            entry.where = f"{editor.path.name}:{first}–{last}"
384            entry.question = f"{entry.question} · {why}"
385            self.review_monitor.refresh(entry)
386        num_ctx = ai.context_for(ctx.estimate_tokens(prompt + ai.SYSTEM_PROMPT, self._token_ratio), cap, room)
387        self.review_monitor.set_status("Thinking")
388        self._show_single_output(f"AI review of {editor.path.name} ({self.ai_model})", "")
389        self._review_text = ""
390        self._review_keys = []
391        self._review_code = code
392        editor.clear_review()  # markers from an earlier review would be misleading
393        worker = ai.ReviewWorker(ai.DEFAULT_HOST, self.ai_model, prompt, num_ctx=num_ctx, **options)
394        worker.chunk.connect(self._review_chunk)
395        worker.finished.connect(self._review_finished)
396        worker.failed.connect(self._review_failed)
397        self._start_answer(worker)

425    def _review_chunk(self, text: str):
426        cursor = self.test_output.textCursor()
427        cursor.movePosition(QTextCursor.End)
428        cursor.insertText(text)
429        self.test_output.setTextCursor(cursor)
430        self.test_output.ensureCursorVisible()
431        self.review_monitor.add_answer(text)
432        # Text arrives a line at a time: re-read the answer so far and show
433        # the findings found so far (the entry's list and the editor's markers).
434        self._review_text += text
435        editor, entry = self._ai_editor, self._ai_entry
436        max_line = editor.editor.blockCount() if editor in self.editors() else None
437        findings, _summary = ai.parse_findings(self._review_text, max_line)
438        keys = [(f.line, f.end_line, f.category, f.comment) for f in findings]
439        if keys != self._review_keys:
440            self._review_keys = keys
441            entry.findings = findings
442            self.review_monitor.refresh_soon(entry)  # (with the answer's next showing)
443            if editor in self.editors():
444                editor.show_review(findings)
445
446    def _review_finished(self, text: str, stats: dict):
447        editor, entry = self._ai_editor, self._ai_entry
448        max_line = editor.editor.blockCount() if editor in self.editors() else None
449        findings, summary = ai.parse_findings(text, max_line)
450        entry.findings, entry.summary = findings, summary
451        name = editor.path.name if editor is not None else entry.where.split(":")[0]
452        self._remember(entry, ctx.Turn(
453            f"Review {entry.where}", f"Review my file {entry.where} ({entry.question.lower()}): list the issues "
454            "you find.",
455            ctx.file_attachment(name, self._review_code, numbers=True, context=self._chat_context(),
456                                ratio=self._token_ratio)))
457        detail = f"{stats.get('seconds', 0):.1f} s · {self.ai_model}" if stats else self.ai_model
458        if stats.get("repeated"):
459            detail += " · stopped a repetition loop"
460        elif stats.get("thinking"):
461            detail += " · the model spent its answer's length on reasoning (see AI → Answer Length)"
462        elif stats.get("truncated"):
463            detail += " · answer cut off at the length limit"
464        self.review_monitor.finish(True, detail)
465        if editor in self.editors():
466            editor.show_review(findings)
467        self.statusBar().showMessage(f"AI review: {len(findings)} findings", 5000)

469    def _review_failed(self, message: str):

310        self._known_models = installed
311        self._set_ai_starting(False)  # (from here a worker thread, or nothing, follows)

352    def _run_review(self):
353        editor = self._ai_editor
354        if editor is None or editor not in self.editors():
355            self.review_monitor.finish(False, "The file was closed")
356            return
357        code = editor.editor.toPlainText()
358        diagnostics = editor.compiler_diagnostics()
359        lines = ctx.file_lines(code)
360        options = self._answer_options()
361        cap = self._chat_context()
362        room = options["num_predict"] + (ai.REASONING_TOKENS if options["think"] else 0) \
363            if options["num_predict"] else ai.REASONING_TOKENS
364
365        def prompt_for(first: int, last: int) -> str:
366            return ai.build_prompt(editor.path.name, editor.highlighter.lexer.name, "\n".join(lines[first - 1:last]),
367                                   self.ai_focus, diagnostics, first, len(lines))
368
369        # The selection (two lines or more), or the whole file if it fits the
370        # model's context with room for the answer; else the procedure at the
371        # cursor. (Sent anyway, a file too long loses its start, instructions
372        # included: Ollama doesn't refuse, it drops what doesn't fit.)
373        first, last, why = 1, len(lines), ""
374        selection = editor.editor.selected_lines()
375        if selection is not None and selection[1] > selection[0]:
376            (first, last), why = selection, "the selection"
377        elif ctx.estimate_tokens(prompt_for(1, len(lines)) + ai.SYSTEM_PROMPT, self._token_ratio) + room \
378                > cap * 0.9:
379            first, last = ctx.window(lines, None, editor.editor.textCursor().blockNumber() + 1)
380            why = "the procedure at the cursor: the file is too long to review at once"
381        prompt = prompt_for(first, last)
382        entry = self._ai_entry
383        if why:
384            entry.where = f"{editor.path.name}:{first}–{last}"
385            entry.question = f"{entry.question} · {why}"
386            self.review_monitor.refresh(entry)
387        num_ctx = ai.context_for(ctx.estimate_tokens(prompt + ai.SYSTEM_PROMPT, self._token_ratio), cap, room)
388        self.review_monitor.set_status("Thinking")
389        self._show_single_output(f"AI review of {editor.path.name} ({self.ai_model})", "")
390        self._review_text = ""
391        self._review_keys = []
392        self._review_code = code
393        editor.clear_review()  # markers from an earlier review would be misleading
394        worker = ai.ReviewWorker(ai.DEFAULT_HOST, self.ai_model, prompt, num_ctx=num_ctx, **options)
395        worker.chunk.connect(self._review_chunk)
396        worker.finished.connect(self._review_finished)
397        worker.failed.connect(self._review_failed)
398        self._start_answer(worker)

426    def _review_chunk(self, text: str):
427        cursor = self.test_output.textCursor()
428        cursor.movePosition(QTextCursor.End)
429        cursor.insertText(text)
430        self.test_output.setTextCursor(cursor)
431        self.test_output.ensureCursorVisible()

433        # Text arrives a line at a time: re-read the answer so far and show
434        # the findings found so far (the entry's list and the editor's markers).
435        self._review_text += text
436        editor, entry = self._ai_editor, self._ai_entry
437        max_line = editor.editor.blockCount() if editor in self.editors() else None
438        findings, _summary = ai.parse_findings(self._review_text, max_line)

440        if keys != self._review_keys:

443            self.review_monitor.refresh_soon(entry)  # (with the answer's next showing)
444            if editor in self.editors():

447    def _review_finished(self, text: str, stats: dict):
448        editor, entry = self._ai_editor, self._ai_entry
449        max_line = editor.editor.blockCount() if editor in self.editors() else None
450        findings, summary = ai.parse_findings(text, max_line)
451        entry.findings, entry.summary = findings, summary
452        name = editor.path.name if editor is not None else entry.where.split(":")[0]
453        self._remember(entry, ctx.Turn(
454            f"Review {entry.where}", f"Review my file {entry.where} ({entry.question.lower()}): list the issues "
455            "you find.",
456            ctx.file_attachment(name, self._review_code, numbers=True, context=self._chat_context(),
457                                ratio=self._token_ratio)))
458        detail = f"{stats.get('seconds', 0):.1f} s · {self.ai_model}" if stats else self.ai_model
459        if stats.get("repeated"):
460            detail += " · stopped a repetition loop"
461        elif stats.get("thinking"):
462            detail += " · the model spent its answer's length on reasoning (see AI → Answer Length)"
463        elif stats.get("truncated"):
464            detail += " · answer cut off at the length limit"
465        self.review_monitor.finish(True, detail)
466        if editor in self.editors():
467            editor.show_review(findings)
468        self.statusBar().showMessage(f"AI review: {len(findings)} findings", 5000)
