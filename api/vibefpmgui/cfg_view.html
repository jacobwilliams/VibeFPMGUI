 62    def __init__(self, view: CfgView, node: cf.Node):
 63        super().__init__()
 64        self.view, self.node = view, node
 65        self.hovered = False
 66        metrics = QFontMetricsF(view.code_font)
 67        width = max((metrics.horizontalAdvance(line) for line in node.text), default=20) + 2 * PADDING
 68        if node.kind == cf.DECISION:

 73        self.setZValue(10)

 78            lines = f"line {node.first}" if node.first == node.last else f"lines {node.first}–{node.last}"
 79            where = f"{node.path.name} (included), " if self.included else ""
 80            self.setToolTip(f"{where}{lines}: click to open")

 88    def shape_path(self) -> QPainterPath:
 89        r = self.rect
 90        path = QPainterPath()
 91        kind = self.node.kind
 92        if kind == cf.DECISION:
 93            d = min(12, r.height() / 2)
 94            path.addPolygon(QPolygonF([QPointF(r.left() + d, r.top()), QPointF(r.right() - d, r.top()),
 95                                       QPointF(r.right(), r.center().y()), QPointF(r.right() - d, r.bottom()),
 96                                       QPointF(r.left() + d, r.bottom()), QPointF(r.left(), r.center().y())]))
 97            path.closeSubpath()
 98        elif kind in (cf.START, cf.END, cf.STOP):
 99            path.addRoundedRect(r, r.height() / 2, r.height() / 2)
100        elif kind == cf.LOOP:
101            path.addRoundedRect(r, 9, 9)
102        else:
103            path.addRoundedRect(r, 3, 3)
104        return path

107        colors = self.view.colors
108        fill, border = colors[self.node.kind]

211    def show_graph(self, graph: cf.Graph, source: str | None = None, fit: bool = True):
212        """Draw ``graph`` (``source``: its file's text, for Copy Code; kept if
213        not given); ``fit``: fit it to the view, else keep the view's range."""
214        if source is not None or self.graph is None or graph.path != self.graph.path:
215            self.sources = {graph.path: source.splitlines()} if source is not None and graph.path else {}
216        view_range = self.vb.viewRange()
217        self.graph = graph
218        self.vb.clear()
219        self.labels = []
220        self.nodes = {n.id: CfgNode(self, n) for n in graph.nodes}
221        places = cf.layout(graph, {i: item.size for i, item in self.nodes.items()})
222        for i, item in self.nodes.items():
223            x, y = places.positions[i]
224            item.setPos(x, y)
225            self.vb.addItem(item)
226        right_edge = max((item.scene_rect().right() for item in self.nodes.values()), default=0)
227        backs = 0
228        for index, edge in enumerate(graph.edges):
229            source, target = self.nodes[edge.source], self.nodes[edge.target]
230            if edge.back:
231                backs += 1
232                points, direction = _back_edge(source.scene_rect(), target.scene_rect(), right_edge + 18 * backs,
233                                               8 + 4 * (backs % 3))
234            elif index in places.routes:
235                points, direction = _routed(source.scene_rect(), target.scene_rect(), places.routes[index])
236            else:
237                points, direction = edge_geometry(source.scene_rect(), target.scene_rect(), LAYERED_TB, True)
238            self._add_edge(points, direction, edge.label)
239        self.stack.setCurrentWidget(self.plot)
240        if fit:
241            self.fit()
242        else:
243            self.vb.setRange(xRange=view_range[0], yRange=view_range[1], padding=0)

424def _routed(source: QRectF, target: QRectF, waypoints: list[tuple[float, float]]):
425    """An edge across layers: from the source's bottom through its waypoints
426    to the target's top, an S-curve between each two."""
427    stops = [QPointF(source.center().x(), source.bottom())] + [QPointF(x, y) for x, y in waypoints] \
428        + [QPointF(target.center().x(), target.top())]
429    parts = []
430    for a, b in pairwise(stops):
431        dy = (b.y() - a.y()) / 2
432        parts.append(_bezier(a, QPointF(a.x(), a.y() + dy), QPointF(b.x(), b.y() - dy), b))
433    points = np.vstack(parts)
434    d = points[-1] - points[-3]
435    return points, (d[0], d[1])

438def _back_edge(source: QRectF, target: QRectF, x: float, drop: float):
439    """A loop's way back: down out of the source's bottom by ``drop`` (into
440    the gap below its layer, clear of the nodes beside it), right to ``x``,
441    up, and into the target's right side."""
442    cx, below = source.center().x(), source.bottom() + drop
443    ty = target.center().y()
444    points = np.array([[cx, source.bottom()], [cx, below], [x, below], [x, ty], [target.right(), ty]])
445    return points, (target.right() - x, 0.0)
