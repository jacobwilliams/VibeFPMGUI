vibefpmgui.lsp

A small client for the Fortran language server, fortls.

fortls indexes the project's Fortran sources and answers questions about them: what a name is (hover), where it is defined, the outline of a file and the symbols in the whole project. It runs as a background process speaking the Language Server Protocol (JSON messages over stdin/stdout); LspClient handles that with a QProcess, so the editor never waits for it.

Open editor tabs send their text to the server (didOpen/didChange), so unsaved edits are seen. Unless the project has its own fortls settings (.fortlsrc, .fortls.json or .fortls), fortls is told to index just the fpm folders (src, app, test, example, include and any source-dir) of the project and each downloaded dependency, rather than every folder under the project (a Python environment, say).

 68def source_dirs(graph: m.DependencyGraph | None, project_dir: Path) -> list[str]:
 69    """``folder/**`` globs for the fpm folders of the project and of every
 70    dependency on disk (fortls doesn't look into subfolders otherwise)."""
 71    packages = [project_dir]
 72    if graph is not None:
 73        packages += [p.directory for p in graph.packages.values()
 74                     if p.name != graph.root and p.kind not in (m.META, m.MISSING)
 75                     and p.directory and p.directory.is_dir()]
 76    folders: set[str] = set()
 77    for directory in packages:
 78        manifest = directory / "fpm.toml"
 79        found = fpm_folders(manifest) if manifest.is_file() else {}
 80        folders.update(found or {str(directory.resolve()): ""})
 81    # A folder inside another listed one is already covered by its "/**".
 82    kept = [f for f in sorted(folders)
 83            if not any(other != f and Path(other) in Path(f).parents for other in folders)]
 84    return [f"{folder}/**" for folder in kept]

 87def _packages(graph: m.DependencyGraph | None, project_dir: Path) -> list[tuple[Path, str]]:
 88    """(folder, version) of the project and of every dependency on disk."""
 89    out = [(project_dir, "")]
 90    if graph is not None:
 91        root = graph.packages.get(graph.root)
 92        out[0] = (project_dir, (root.version if root else "") or "")
 93        out += [(p.directory, p.version or "") for p in graph.packages.values()
 94                if p.name != graph.root and p.kind not in (m.META, m.MISSING)
 95                and p.directory and p.directory.is_dir()]
 96    return out

 99def preprocessor_args(graph: m.DependencyGraph | None, project_dir: Path) -> list[str]:
100    """fortls options for the C preprocessor settings in the fpm.toml files
101    (``[preprocess.cpp]``: macros, suffixes) and the packages' include
102    folders, so #ifdef branches are read as fpm compiles them. [] when no
103    package has preprocessor settings (fortls's own defaults then)."""
104    macros: dict[str, str] = {}
105    suffixes: list[str] = []
106    includes: list[str] = []
107    found = False
108    for directory, version in reversed(_packages(graph, project_dir)):  # (the project's own win)
109        manifest = directory / "fpm.toml"
110        try:
111            data = m.load_toml(manifest) if manifest.is_file() else {}
112        except Exception:  # noqa: BLE001 - an unreadable manifest has no settings
113            data = {}
114        cpp = (data.get("preprocess") or {}).get("cpp") if isinstance(data.get("preprocess"), dict) else None
115        if not isinstance(cpp, dict):
116            continue
117        found = True
118        for macro in cpp.get("macros") or []:
119            name, _, value = str(macro).partition("=")
120            if name.strip():
121                macros[name.strip()] = value.strip().replace("{version}", version)
122        for suffix in cpp.get("suffixes") or []:
123            suffix = "." + str(suffix).lstrip(".")
124            if suffix not in suffixes:
125                suffixes.append(suffix)
126        includes += [f for f, role in fpm_folders(manifest).items() if role == "include" and f not in includes]
127    if not found:
128        return []
129    args = []
130    if macros:
131        args += ["--pp_defs", json.dumps(macros)]
132    if suffixes:
133        args += ["--pp_suffixes", *suffixes]
134    if includes:
135        args += ["--include_dirs", *includes]
136    return args

139def server_args(graph: m.DependencyGraph | None, project_dir: Path) -> list[str]:
140    args = ["--disable_autoupdate",  # it would otherwise ask PyPI for updates
141            "--hover_signature",
142            "--use_signature_help",    # complete names only; arguments are hinted on "("

147    return args

151class LspIssue(NamedTuple):
152    """A problem fortls found in a file (it checks a file when it's opened
153    or saved: the saved text), shown like fortitude's issues."""
154    line: int        # 1-based
155    column: int      # 1-based
156    end_line: int

164    source: str = "fortls"

178def diagnostic_issues(diagnostics) -> list[LspIssue]:
179    """publishDiagnostics' list -> issues (anything malformed is skipped)."""
180    out = []
181    for d in diagnostics or []:
182        try:
183            r = d["range"]
184            out.append(LspIssue(r["start"]["line"] + 1, r["start"]["character"] + 1,
185                                r["end"]["line"] + 1, r["end"]["character"] + 1,
186                                " ".join(str(d.get("message", "")).split()), int(d.get("severity") or 2)))
187        except (KeyError, TypeError, ValueError):
188            continue
189    return out

218def code_html(code: str, language: str = "fortran", style: str | None = None) -> str:
219    """A block of code, syntax-highlighted with the editor's Pygments style,
220    on a panel in that style's background color."""
221    from pygments import highlight
222    from pygments.formatters import HtmlFormatter
223    from pygments.lexers import FortranLexer, TextLexer
224    from pygments.styles import get_style_by_name
225
226    style = style or _current_style()
227    lexer = FortranLexer() if language.lower().startswith("fortran") or not language else TextLexer()
228    spans = highlight(code.rstrip("\n"), lexer, HtmlFormatter(nowrap=True, noclasses=True, style=style))
229    # Synopses aren't always valid Fortran ("[,dim] | [mask]"): no error boxes.
230    spans = re.sub(r"(background-color|border):\s*[^;\"]+;?\s*", "", spans)
231    background = get_style_by_name(style).background_color
232    return (f"<table width='100%' cellspacing='0' cellpadding='6' style='margin:3px 0; "
233            f"background-color:{background}'><tr><td><pre style='margin:0'>{spans.rstrip()}</pre>"
234            "</td></tr></table>")

244def markdown_html(text: str, style: str | None = None) -> str:
245    """Markdown as HTML for a rich-text label (rendered by Qt): code blocks
246    syntax-highlighted, headings as small colored labels."""
247    out = []
248    for i, part in enumerate(re.split(r"```([\w+-]*)[ \t]*\n(.*?)```", text, flags=re.DOTALL)):
249        kind = i % 3
250        if kind == 1:
251            language = part
252        elif kind == 2:
253            out.append(code_html(part, language, style))
254        elif part.strip():
255            doc = QTextDocument()
256            doc.setMarkdown(part, QTextDocument.MarkdownDialectGitHub)
257            body = re.search(r"<body[^>]*>(.*)</body>", doc.toHtml(), re.DOTALL)
258            out.append(body.group(1).strip() if body else "")
259    result = "".join(out)
260    from . import highlighter as hl
261    accent = _accent(hl.style_is_dark(style or _current_style()))
262
263    def heading(match):
264        title = html.unescape(re.sub(r"<[^>]+>", "", match.group(2))).strip()
265        return (f"<p style='margin:9px 0 2px 0'><span style='color:{accent}; font-weight:600; "
266                f"font-size:small'>{html.escape(title.upper())}</span></p>")
267
268    return re.sub(r"<h(\d)[^>]*>(.*?)</h\1>", heading, result, flags=re.DOTALL)

278def _clean(text: str) -> str:
279    # "-----" between the declaration and a doc comment: just a gap (the code
280    # panel already sets them apart; markdown would make the line above a heading).
281    text = re.sub(r"^[ \t]*-{3,}[ \t]*$", "\n", text, flags=re.MULTILINE)
282    # Qt shows _word_ underlined (like a link); the manual pages mean italics.
283    parts = re.split(r"(```.*?```|`[^`]*`)", text, flags=re.DOTALL)
284    return "".join(part if i % 2 else re.sub(r"(?<![\w\\*])_([^_\n]+?)_(?!\w)", r"*\1*", part)
285                   for i, part in enumerate(parts))

289              ("program", r"^\s*PROGRAM\b"), ("submodule", r"^\s*SUBMODULE\b"),
290              ("module", r"^\s*MODULE\b(?!\s+PROCEDURE)"), ("interface", r"\bINTERFACE\b"),

294def hover_info(contents, style: str | None = None) -> HoverInfo | None:
295    """fortls's hover answer, taken apart for the hover card (None if empty)."""
296    text = hover_text(contents)
297    if not text.strip():
298        return None
299    match = INTRINSIC_RE.match(text)
300    if match:
301        rest = text[match.end():]
302        category = summary = ""
303        line = re.match(r"\s*\*\*\w+\*\*\s*-\s*(?:\\?\[([^\]\\]*)\\?\])?\s*(.*)\n", rest)
304        if line:
305            category = " · ".join(p.strip() for p in (line.group(1) or "").split(":") if p.strip())
306            summary = line.group(2).strip()
307            rest = rest[line.end():]
308        return HoverInfo("intrinsic", category, summary, markdown_html(_clean(rest), style))
309    kind = ""
310    code = re.match(r"\s*```[\w+-]*[ \t]*\n(.*?)```", text, re.DOTALL)
311    if code:
312        first = code.group(1).strip().splitlines()[0] if code.group(1).strip() else ""
313        if "::" in first and not re.match(r"^\s*TYPE\b\s*(,|::)", first, re.IGNORECASE):
314            kind = "parameter" if re.search(r"\bPARAMETER\b", first, re.IGNORECASE) else "variable"
315        else:
316            kind = next((k for k, pattern in DECL_KINDS if re.search(pattern, first, re.IGNORECASE)), "")
317            if not kind and re.match(r"^\s*TYPE\b", first, re.IGNORECASE):
318                kind = "type"
319    return HoverInfo(kind, "", "", markdown_html(_clean(text), style))

346class LspClient(QObject):
347    """One fortls process for the open project."""
348
349    stopped = Signal(str)  # the server exited on its own (with why)
350    message = Signal(str)  # something the server wants the user to know
351    diagnostics = Signal(object, object)  # file (Path), [LspIssue]: fortls's check of a file
352
353    def __init__(self, parent=None):
354        super().__init__(parent)
355        self.process: QProcess | None = None
356        self.root: Path | None = None
357        self.args: list[str] = []
358        self._reader = FrameReader()
359        self._next_id = 0
360        self._callbacks: dict[int, Callable] = {}
361        self._ready = False
362        self._queue: list[dict] = []        # sent once initialize has answered
363        self._versions: dict[str, int] = {}  # open document uri -> version
364
365    # ---------------------------------------------------------- lifecycle
366    def is_running(self) -> bool:
367        return self.process is not None
368
369    def start(self, root: Path, args: list[str]) -> bool:
370        self.stop()
371        program = executable()
372        if program is None:
373            return False
374        self.root, self.args = root, list(args)
375        process = QProcess(self)
376        process.setWorkingDirectory(str(root))
377        process.setStandardErrorFile(QProcess.nullDevice())
378        process.readyReadStandardOutput.connect(self._read)
379        # (Methods, not lambdas holding the process: when Qt deletes the
380        # process, PySide drops those lambdas, and with them possibly the last
381        # reference to the process's Python wrapper, mid-delete: a crash.)
382        process.finished.connect(self._process_finished)
383        process.errorOccurred.connect(self._process_error)
384        self.process = process
385        process.start(program, self.args)
386        self._send({"jsonrpc": "2.0", "id": self._new_id(self._initialized), "method": "initialize",
387                    "params": {"processId": None, "rootUri": to_uri(root), "capabilities": {}}})
388        return True
389
390    def stop(self):
391        process, self.process = self.process, None
392        self._reset()
393        if process is not None:
394            process.kill()
395            process.waitForFinished(1000)
396            process.deleteLater()
397
398    def _reset(self):
399        self._reader.clear()
400        self._callbacks.clear()
401        self._queue.clear()
402        self._versions.clear()
403        self._ready = False
404
405    def _process_finished(self, *_):
406        if isinstance(self.sender(), QProcess):
407            self._finished(self.sender())
408
409    def _process_error(self, error):
410        if error == QProcess.FailedToStart and isinstance(self.sender(), QProcess):
411            self._finished(self.sender())
412
413    def _finished(self, process: QProcess):
414        if process is not self.process:
415            return  # one we stopped
416        self.process = None
417        self._reset()
418        try:
419            process.deleteLater()
420        except RuntimeError:
421            return  # the app is shutting down (Qt has already deleted it)
422        self.stopped.emit("fortls could not be started" if process.error() == QProcess.FailedToStart
423                          else f"fortls stopped (exit code {process.exitCode()})")
424
425    def _initialized(self, _result):
426        self._ready = True
427        self._send({"jsonrpc": "2.0", "method": "initialized", "params": {}})
428        queue, self._queue = self._queue, []
429        for message in queue:
430            self._send(message)
431
432    # ---------------------------------------------------------- messages
433    def _new_id(self, callback: Callable) -> int:
434        self._next_id += 1
435        self._callbacks[self._next_id] = callback
436        return self._next_id
437
438    def _send(self, message: dict):
439        if self.process is None:
440            return
441        self.process.write(frame(message))
442
443    def _post(self, message: dict):
444        if self._ready:
445            self._send(message)
446        else:
447            self._queue.append(message)
448
449    def request(self, method: str, params: dict, callback: Callable) -> bool:
450        """Ask the server something; ``callback(result)`` gets the answer
451        (None on an error). False if no server is running."""
452        if self.process is None:
453            return False
454        self._post({"jsonrpc": "2.0", "id": self._new_id(callback), "method": method, "params": params})
455        return True
456
457    def notify(self, method: str, params: dict):
458        if self.process is not None:
459            self._post({"jsonrpc": "2.0", "method": method, "params": params})
460
461    def _read(self):
462        if self.process is None:
463            return
464        for message in self._reader.feed(bytes(self.process.readAllStandardOutput())):
465            self._dispatch(message)
466
467    def _dispatch(self, message: dict):
468        if "method" in message:
469            if "id" in message:  # a request from the server: we don't support any
470                self._send({"jsonrpc": "2.0", "id": message["id"], "result": None})
471            elif message["method"] == "window/showMessage":  # e.g. why a rename failed
472                text = (message.get("params") or {}).get("message")
473                if text:
474                    self.message.emit(str(text))
475            elif message["method"] == "textDocument/publishDiagnostics":
476                params = message.get("params") or {}
477                if params.get("uri"):
478                    self.diagnostics.emit(from_uri(params["uri"]), diagnostic_issues(params.get("diagnostics")))
479            return  # other notifications (logs) aren't used
480        callback = self._callbacks.pop(message.get("id"), None)
481        if callback is not None:
482            try:
483                callback(message.get("result") if "error" not in message else None)
484            except RuntimeError:
485                pass  # the editor that asked has been closed
486
487    # ---------------------------------------------------------- documents
488    def open_document(self, path: Path, text: str):
489        uri = to_uri(path)
490        if self.process is None:
491            return
492        if uri in self._versions:
493            self.change_document(path, text)
494            return

567def outline(symbols, line_text: Callable[[int], str] | None = None) -> list[tuple[int, str, str, int, int]]:
568    """documentSymbol answer -> [(depth, kind word, name, first line, last
569    line)] for modules, types and procedures, in file order (lines 1-based).
570    ``line_text(line)`` (the file's text) refines the kind words."""
571    rows = []
572
573    def kind_word(s, first):
574        word = SYMBOL_KINDS.get(s["kind"], "")
575        if line_text is not None and s["kind"] in (2, 6, 12):
576            match = KEYWORD_RE.match(line_text(first).strip())
577            if match:
578                word = match.group(1).lower()
579        return word
580
581    def visit(items, depth, container_depth):
582        for s in items or []:
583            if not isinstance(s, dict) or s.get("kind") not in OUTLINE_KINDS:
584                continue
585            rng = s.get("range") or s.get("location", {}).get("range")
586            if not rng:

590                         rng["start"]["line"] + 1, rng["end"]["line"] + 1))
591            container_depth[s.get("name")] = level
592            visit(s.get("children"), depth + 1, container_depth)
593
594    visit(symbols, 0, {})
595    rows.sort(key=lambda r: (r[3], r[0]))

642def code_spans(line: str) -> list[tuple[int, int]]:
643    """The parts of a line that are code: not in a string or a "!" comment."""
644    spans, start, quote = [], 0, None
645    for i, ch in enumerate(line):
646        if quote:
647            if ch == quote:
648                quote, start = None, i + 1
649        elif ch in "'\"":
650            spans.append((start, i))
651            quote = ch
652        elif ch == "!":
653            spans.append((start, i))
654            return spans
655    if quote is None:
656        spans.append((start, len(line)))
657    return spans

698def argument_index(text_before: str) -> int | None:
699    """Which argument (0-based) the cursor is in, from the code before it on
700    the line: the commas at the level of the innermost open "(" (None if the
701    cursor isn't inside parentheses, or the innermost is a "[")."""
702    open_brackets: list[list] = []  # [bracket, commas] of each one still open
703    for start, end in code_spans(text_before):
704        for ch in text_before[start:end]:
705            if ch in "([":
706                open_brackets.append([ch, 0])
707            elif ch in ")]":
708                if open_brackets:
709                    open_brackets.pop()
710            elif ch == "," and open_brackets:
711                open_brackets[-1][1] += 1
712    if not open_brackets:
713        return None
714    bracket, commas = open_brackets[-1]
715    return commas if bracket == "(" else None

778def add_result_variable_edits(edits: dict[Path, list[TextEdit]], old: str, new: str,
779                              file_text: Callable[[Path], str]) -> dict[Path, list[TextEdit]]:
780    """fortls's rename misses a function's result variable: in a function
781    without a ``result(...)`` clause, ``old = ...`` inside the body also
782    names the function and must be renamed too. Adds those edits."""
783    header = re.compile(rf"\bfunction\s+{re.escape(old)}\s*\(", re.IGNORECASE)
784    word = re.compile(rf"\b{re.escape(old)}\b", re.IGNORECASE)
785    end = re.compile(r"^\s*(end\s*(function\b|$)|contains\b)", re.IGNORECASE)
786    out = {}
787    for path, file_edits in edits.items():
788        try:
789            lines = file_text(path).split("\n")
790        except OSError:
791            out[path] = file_edits
792            continue
793        have = {(e.line, e.column) for e in file_edits}
794        added = list(file_edits)
795        for i, line in enumerate(lines):
796            code = "".join(line[a:b] for a, b in code_spans(line))
797            if not header.search(code):
798                continue
799            # The whole statement (it may continue over "&" lines).
800            statement, j = code, i
801            while statement.rstrip().endswith("&") and j + 1 < len(lines):
802                j += 1
803                statement += "".join(lines[j][a:b] for a, b in code_spans(lines[j]))
804            if re.search(r"\bresult\s*\(", statement, re.IGNORECASE):
805                continue  # the result has its own name
806            for k in range(j + 1, len(lines)):
807                if end.match(lines[k]):
808                    break
809                for a, b in code_spans(lines[k]):
810                    for match in word.finditer(lines[k], a, b):
811                        if (k, match.start()) not in have:
812                            have.add((k, match.start()))
813                            added.append(TextEdit(k, match.start(), k, match.end(), new))
814        out[path] = added
815    return out
