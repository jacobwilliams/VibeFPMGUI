 25def _canvas(ratio: float) -> tuple[QPixmap, QPainter]:
 26    pixmap = QPixmap(int(SIZE * ratio), int(SIZE * ratio))
 27    pixmap.setDevicePixelRatio(ratio)
 28    pixmap.fill(Qt.transparent)
 29    painter = QPainter(pixmap)
 30    painter.setRenderHint(QPainter.Antialiasing)
 31    return pixmap, painter

247def make_icon(name: str, color: QColor | str) -> QIcon:
248    """An icon by name (see ``DRAWERS``) in the given color; Qt derives the
249    greyed-out disabled look from it."""
250    color = QColor(color)
251    icon = QIcon()
252    for ratio in (1, 2, 3):
253        pixmap, painter = _canvas(ratio)
254        DRAWERS[name](painter, color)
255        painter.end()
256        icon.addPixmap(pixmap)
257    return icon

309def badged_icon(base: QIcon, badge: str, sizes=(16, 20, 32)) -> QIcon:
310    """``base`` (e.g. the system folder icon) with a small colored badge in
311    its lower-right corner showing ``badge`` (see ``BADGES``)."""
312    color = QColor(BADGE_COLORS.get(badge, "#868e96"))
313    icon = QIcon()
314    for size in sizes:
315        for ratio in (1, 2):
316            pixmap = QPixmap(int(size * ratio), int(size * ratio))
317            pixmap.setDevicePixelRatio(ratio)
318            pixmap.fill(Qt.transparent)
319            painter = QPainter(pixmap)
320            painter.setRenderHint(QPainter.Antialiasing)
321            base.paint(painter, 0, 0, size, size)
322            d = size * 0.62  # badge diameter
323            x, y = size - d, size - d
324            painter.setPen(QPen(QColor(255, 255, 255, 230), max(1.0, size / 16)))
325            painter.setBrush(color)
326            painter.drawEllipse(QRectF(x, y, d, d))
327            painter.save()
328            painter.translate(x + d * 0.18, y + d * 0.18)
329            painter.scale(d * 0.64 / SIZE, d * 0.64 / SIZE)
330            BADGES[badge](painter, QColor("white"))
331            painter.restore()
332            painter.end()
333            icon.addPixmap(pixmap)
334    return icon

349def fortran_file_icon(include: bool = False) -> QIcon:
350    """The icon for Fortran source files (a page with the Fortran "F"), or
351    in gray for include files, drawn crisply at each size."""
352    icon = _fortran_icons.get(include)
353    if icon is None:
354        from qtpy.QtCore import QByteArray
355        from qtpy.QtSvg import QSvgRenderer
356
357        svg = FORTRAN_FILE_PATH.read_text(encoding="utf-8")
358        if include:
359            svg = svg.replace("#734f96", "#6c757d").replace("#a98cc9", "#adb5bd")
360        renderer = QSvgRenderer(QByteArray(svg.encode()))
361        icon = _fortran_icons[include] = QIcon()
362        if renderer.isValid():
363            for size in (16, 20, 24, 32, 40, 48, 64):
364                pixmap = QPixmap(size, size)
365                pixmap.fill(Qt.transparent)
366                painter = QPainter(pixmap)
367                painter.setRenderHint(QPainter.Antialiasing)
368                renderer.render(painter)
369                painter.end()
370                icon.addPixmap(pixmap)
371    return icon

374def app_icon() -> QIcon:
375    """The application icon: the fpm logo centred on a white rounded square
376    (the usual macOS app-icon shape), drawn crisply at each size."""
377    from qtpy.QtSvg import QSvgRenderer
378
379    renderer = QSvgRenderer(str(LOGO_PATH))
380    icon = QIcon()
381    if not renderer.isValid():
382        return icon
383    for size in (16, 32, 48, 64, 128, 256, 512, 1024):
384        pixmap = QPixmap(size, size)
385        pixmap.fill(Qt.transparent)
386        p = QPainter(pixmap)
387        draw_app_icon(p, size, renderer)
388        p.end()
389        icon.addPixmap(pixmap)
390    return icon

393def draw_app_icon(p: QPainter, size: float, renderer):
394    """Paint the app icon (`app_icon`) at (0, 0), ``size`` across, with the
395    logo from ``renderer`` (a QSvgRenderer)."""
396    p.setRenderHint(QPainter.Antialiasing)
397    p.setRenderHint(QPainter.SmoothPixmapTransform)
398    view = renderer.viewBoxF()
399    aspect = view.width() / view.height() if view.height() else 1.0
400    # macOS icons leave ~10% margin around the rounded square.
401    margin = size * 0.09
402    tile = QRectF(margin, margin, size - 2 * margin, size - 2 * margin)
403    radius = tile.width() * 0.225
404    if size >= 32:  # a soft shadow (too muddy at tiny sizes)
405        for i, alpha in enumerate((18, 12, 6)):
406            grow = size * 0.006 * (i + 1)
407            p.setPen(Qt.NoPen)
408            p.setBrush(QColor(0, 0, 0, alpha))
409            p.drawRoundedRect(tile.adjusted(-grow, -grow + size * 0.012, grow, grow + size * 0.012),
410                              radius + grow, radius + grow)
411    gradient = QLinearGradient(tile.topLeft(), tile.bottomLeft())
412    gradient.setColorAt(0, QColor("#ffffff"))
413    gradient.setColorAt(1, QColor("#eef0f6"))
414    p.setBrush(gradient)
415    p.setPen(QPen(QColor(0, 0, 0, 28), max(1.0, size / 256)))
416    p.drawRoundedRect(tile, radius, radius)
417    # The logo, as large as fits with some breathing room.
418    box = tile.adjusted(tile.width() * 0.16, tile.height() * 0.12,
419                        -tile.width() * 0.16, -tile.height() * 0.12)
420    h = box.height()
421    w = h * aspect
422    if w > box.width():
423        w = box.width()
424        h = w / aspect
425    renderer.render(p, QRectF(box.center().x() - w / 2, box.center().y() - h / 2, w, h))

433def recolored_logo(turn: float) -> bytes:
434    """The fpm logo's SVG with each piece's color moved ``turn`` of the way
435    around the color wheel (0 to 1), and a little more or less for each piece
436    (so the colors ripple as ``turn`` grows), keeping their depth. At 0 (or
437    1) it's the logo's own colors."""
438    svg = _logo_text()
439    pieces = iter(range(100))
440
441    def shift(match: re.Match) -> str:
442        color = QColor(match.group(1))
443        h, s, v, _ = color.getHsvF()
444        phase = 2 * math.pi * next(pieces) / 5
445        ripple = 0.04 * (math.sin(4 * math.pi * turn + phase) - math.sin(phase))  # (0 at turn 0 and 1)
446        color.setHsvF((max(h, 0) + turn + ripple) % 1.0, s, v)
447        return f"fill:{color.name()}"
448
449    return re.sub(r"fill:(#[0-9a-fA-F]{6})", shift, svg).encode()

470def symbol_icon(kind: str) -> QIcon:
471    """A 16 px badge for a kind of symbol ("function", "module"…)."""
472    if kind in _symbol_icons:
473        return _symbol_icons[kind]
474    letter, color = SYMBOL_BADGES.get(kind, ((kind[:1] or "?").upper(), "#868e96"))
475    width, height = 16, 16
476    icon = QIcon()
477    for ratio in (1, 2, 3):
478        pixmap = QPixmap(int(width * ratio), int(height * ratio))
479        pixmap.setDevicePixelRatio(ratio)
480        pixmap.fill(Qt.transparent)
481        p = QPainter(pixmap)
482        p.setRenderHint(QPainter.Antialiasing)
483        p.setRenderHint(QPainter.TextAntialiasing)
484        box = QRectF(1, 1, 14, 14)
485        p.setPen(Qt.NoPen)
486        p.setBrush(QColor(color))
487        p.drawRoundedRect(box, 3.5, 3.5)
488        font = p.font()
489        font.setPixelSize(10)
490        font.setBold(True)
491        p.setFont(font)
492        p.setPen(QColor("white"))
493        p.drawText(box, Qt.AlignCenter, letter)
494        p.end()
495        icon.addPixmap(pixmap)
496    _symbol_icons[kind] = icon

394def draw_app_icon(p: QPainter, size: float, renderer):
395    """Paint the app icon (`app_icon`) at (0, 0), ``size`` across, with the
396    logo from ``renderer`` (a QSvgRenderer)."""
397    p.setRenderHint(QPainter.Antialiasing)
398    p.setRenderHint(QPainter.SmoothPixmapTransform)
399    view = renderer.viewBoxF()
400    aspect = view.width() / view.height() if view.height() else 1.0
401    # macOS icons leave ~10% margin around the rounded square.
402    margin = size * 0.09
403    tile = QRectF(margin, margin, size - 2 * margin, size - 2 * margin)
404    radius = tile.width() * 0.225
405    if size >= 32:  # a soft shadow (too muddy at tiny sizes)
406        for i, alpha in enumerate((18, 12, 6)):
407            grow = size * 0.006 * (i + 1)
408            p.setPen(Qt.NoPen)
409            p.setBrush(QColor(0, 0, 0, alpha))
410            p.drawRoundedRect(tile.adjusted(-grow, -grow + size * 0.012, grow, grow + size * 0.012),
411                              radius + grow, radius + grow)
412    gradient = QLinearGradient(tile.topLeft(), tile.bottomLeft())
413    gradient.setColorAt(0, QColor("#ffffff"))
414    gradient.setColorAt(1, QColor("#eef0f6"))
415    p.setBrush(gradient)
416    p.setPen(QPen(QColor(0, 0, 0, 28), max(1.0, size / 256)))
417    p.drawRoundedRect(tile, radius, radius)
418    # The logo, as large as fits with some breathing room.
419    box = tile.adjusted(tile.width() * 0.16, tile.height() * 0.12,
420                        -tile.width() * 0.16, -tile.height() * 0.12)
421    h = box.height()

423    if w > box.width():

def recolored_logo(turn: float) -> bytes: 
434def recolored_logo(turn: float) -> bytes:
435    """The fpm logo's SVG with each piece's color moved ``turn`` of the way
436    around the color wheel (0 to 1), and a little more or less for each piece
437    (so the colors ripple as ``turn`` grows), keeping their depth. At 0 (or
438    1) it's the logo's own colors."""
439    svg = _logo_text()
440    pieces = iter(range(100))
441
442    def shift(match: re.Match) -> str:
443        color = QColor(match.group(1))
444        h, s, v, _ = color.getHsvF()
445        phase = 2 * math.pi * next(pieces) / 5
446        ripple = 0.04 * (math.sin(4 * math.pi * turn + phase) - math.sin(phase))  # (0 at turn 0 and 1)
447        color.setHsvF((max(h, 0) + turn + ripple) % 1.0, s, v)
448        return f"fill:{color.name()}"
449
450    return re.sub(r"fill:(#[0-9a-fA-F]{6})", shift, svg).encode()
