 26def _canvas(ratio: float) -> tuple[QPixmap, QPainter]:
 27    pixmap = QPixmap(int(SIZE * ratio), int(SIZE * ratio))
 28    pixmap.setDevicePixelRatio(ratio)
 29    pixmap.fill(Qt.transparent)
 30    painter = QPainter(pixmap)
 31    painter.setRenderHint(QPainter.Antialiasing)
 32    return pixmap, painter

451def make_icon(name: str, color: QColor | str) -> QIcon:
452    """An icon by name (see ``DRAWERS``) in the given color; Qt derives the
453    greyed-out disabled look from it."""
454    color = QColor(color)
455    icon = QIcon()
456    for ratio in (1, 2, 3):
457        pixmap, painter = _canvas(ratio)
458        DRAWERS[name](painter, color)
459        painter.end()
460        icon.addPixmap(pixmap)
461    return icon

597def draw_app_icon(p: QPainter, size: float, renderer):
598    """Paint the app icon (`app_icon`) at (0, 0), ``size`` across, with the
599    logo from ``renderer`` (a QSvgRenderer)."""
600    p.setRenderHint(QPainter.Antialiasing)
601    p.setRenderHint(QPainter.SmoothPixmapTransform)
602    view = renderer.viewBoxF()
603    aspect = view.width() / view.height() if view.height() else 1.0
604    # macOS icons leave ~10% margin around the rounded square.
605    margin = size * 0.09
606    tile = QRectF(margin, margin, size - 2 * margin, size - 2 * margin)
607    radius = tile.width() * 0.225
608    if size >= 32:  # a soft shadow (too muddy at tiny sizes)
609        for i, alpha in enumerate((18, 12, 6)):
610            grow = size * 0.006 * (i + 1)
611            p.setPen(Qt.NoPen)
612            p.setBrush(QColor(0, 0, 0, alpha))
613            p.drawRoundedRect(tile.adjusted(-grow, -grow + size * 0.012, grow, grow + size * 0.012),
614                              radius + grow, radius + grow)
615    gradient = QLinearGradient(tile.topLeft(), tile.bottomLeft())
616    gradient.setColorAt(0, QColor("#ffffff"))
617    gradient.setColorAt(1, QColor("#eef0f6"))
618    p.setBrush(gradient)
619    p.setPen(QPen(QColor(0, 0, 0, 28), max(1.0, size / 256)))
620    p.drawRoundedRect(tile, radius, radius)
621    # The logo, as large as fits with some breathing room.
622    box = tile.adjusted(tile.width() * 0.16, tile.height() * 0.12,
623                        -tile.width() * 0.16, -tile.height() * 0.12)
624    h = box.height()
625    w = h * aspect
626    if w > box.width():
627        w = box.width()
628        h = w / aspect
629    renderer.render(p, QRectF(box.center().x() - w / 2, box.center().y() - h / 2, w, h))

637def recolored_logo(turn: float) -> bytes:
638    """The fpm logo's SVG with each piece's color moved ``turn`` of the way
639    around the color wheel (0 to 1), and a little more or less for each piece
640    (so the colors ripple as ``turn`` grows), keeping their depth. At 0 (or
641    1) it's the logo's own colors."""
642    svg = _logo_text()
643    pieces = iter(range(100))
644
645    def shift(match: re.Match) -> str:
646        color = QColor(match.group(1))
647        h, s, v, _ = color.getHsvF()
648        phase = 2 * math.pi * next(pieces) / 5
649        ripple = 0.04 * (math.sin(4 * math.pi * turn + phase) - math.sin(phase))  # (0 at turn 0 and 1)
650        color.setHsvF((max(h, 0) + turn + ripple) % 1.0, s, v)
651        return f"fill:{color.name()}"
652
653    return re.sub(r"fill:(#[0-9a-fA-F]{6})", shift, svg).encode()
