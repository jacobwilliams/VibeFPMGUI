vibefpmgui.build_configs

Build configurations: named sets of Build options for a project, kept in a JSON file beside its fpm.toml (FILE_NAME unless another is opened), so they can be shared (e.g. committed with the project).

The file::

{
  "vibefpm": 1,
  "configurations": [
    {"name": "release (ifx)", "profile": "release", "features": [],
     "compiler": "ifx", "flags": "", "target": "", "extra_args": "",
     "program_args": "", "parallel_tests": 4, "coverage": false,
     "verbose": false}
  ]
}

A configuration has every Build option but the fpm executable (a path on this computer, kept in the app's settings). An option left out of a configuration has its default; unknown keys are kept as they are.

Beside the configurations, "new_configuration" is New configuration's Build options (those not saved as one): a configuration without a name. Then "options" (one set, for every configuration in it) are the GUI's for the project: see vibefpmgui.project_options.

ConfigFile is a project's file as the app has it: read whole (a file that can't be read whole isn't used, nor written over), and written a part at a time, the rest kept as it is.

 48@dataclass
 49class Configuration:
 50    name: str
 51    profile: str = DEFAULT_PROFILE
 52    features: list[str] = field(default_factory=list)
 53    compiler: str = ""
 54    flags: str = ""
 55    target: str = ""
 56    extra_args: str = ""
 57    program_args: str = ""
 58    parallel_tests: int = 1
 59    coverage: bool = False
 60    verbose: bool = False

 62    def to_json(self, named: bool = True) -> dict:
 63        """As the file has it (``named`` False: New configuration's, without a name)."""
 64        data = asdict(self)
 65        data["profile"] = "" if self.profile == DEFAULT_PROFILE else self.profile
 66        if not named:
 67            del data["name"]
 68        return data
 69
 70    @classmethod
 71    def from_json(cls, data: dict, name: str | None = None) -> Configuration:
 72        """A configuration from the file (a value of the wrong type: its
 73        default). ``name``: one without a name in the file (New configuration's)."""
 74        if name is None:
 75            name = data.get("name")
 76            if not isinstance(name, str) or not name.strip():
 77                raise ConfigError("A configuration has no name.")
 78        config = cls(name.strip())
 79        for f in fields(cls):
 80            if f.name == "name" or f.name not in data:
 81                continue
 82            value, default = data[f.name], getattr(config, f.name)
 83            if isinstance(default, bool):
 84                ok = isinstance(value, bool)
 85            elif isinstance(default, int):
 86                ok = isinstance(value, int) and not isinstance(value, bool) and value >= 1
 87            elif isinstance(default, list):
 88                ok = isinstance(value, list) and all(isinstance(v, str) for v in value)
 89            else:
 90                ok = isinstance(value, str)
 91            if ok:
 92                setattr(config, f.name, list(value) if isinstance(value, list) else value)
 93        if not config.profile:
 94            config.profile = DEFAULT_PROFILE
 95        return config

109def _read(path: Path) -> dict:
110    try:
111        data = json.loads(Path(path).read_text(encoding="utf-8"))
112    except OSError as exc:
113        raise ConfigError(f"Can't read {path}: {exc.strerror or exc}") from exc
114    except (ValueError, UnicodeDecodeError) as exc:
115        raise ConfigError(f"{Path(path).name} isn't valid JSON: {exc}") from exc
116    if not isinstance(data, dict) or not isinstance(data.get("configurations", []), list):
117        raise ConfigError(f"{Path(path).name} isn't a VibeFPM configuration file "
118                          "(no \"configurations\" list).")
119    return data
120
121
122def load(path: Path) -> list[Configuration]:
123    """The configurations in a file, in its order (none if it doesn't exist)."""
124    if not Path(path).exists():
125        return []
126    configs, seen = [], set()
127    for item in _read(path).get("configurations", []):
128        if not isinstance(item, dict):
129            raise ConfigError(f"{Path(path).name}: a configuration isn't an object.")
130        config = Configuration.from_json(item)
131        if config.name in seen:
132            raise ConfigError(f"{Path(path).name}: two configurations are named “{config.name}”.")
133        seen.add(config.name)
134        configs.append(config)
135    return configs

138def _write(path: Path, data: dict):
139    """Write the file whole or not at all (a temporary file, then renamed)."""
140    path = Path(path)
141    text = json.dumps(data, indent=2, ensure_ascii=False) + "\n"
142    handle, temporary = tempfile.mkstemp(prefix=".vibefpm-", suffix=".json", dir=path.parent)
143    try:
144        with os.fdopen(handle, "w", encoding="utf-8", newline="\n") as f:
145            f.write(text)
146        os.replace(temporary, path)
147    except BaseException:
148        Path(temporary).unlink(missing_ok=True)
149        raise

152def _with_configurations(path: Path, change) -> list[Configuration]:
153    """Read the file (or start one), let ``change`` edit its list of
154    configurations (as JSON objects), and write it back; the result. Other
155    keys, and keys of a configuration this app doesn't know, are kept; a
156    file that can't be read whole isn't written over (ConfigError)."""
157    load(path)  # (every configuration checked first)
158    data = _read(path) if Path(path).exists() else {}
159    items = [item for item in data.get("configurations", []) if isinstance(item, dict)]
160    change(items)
161    data = {"vibefpm": data.get("vibefpm", FORMAT), **{k: v for k, v in data.items() if k != "vibefpm"},
162            "configurations": items}
163    _write(path, data)
164    return load(path)

167def save(path: Path, config: Configuration) -> list[Configuration]:
168    """Add a configuration to the file, or replace the one with its name."""
169    def change(items: list[dict]):
170        new = config.to_json()
171        for i, item in enumerate(items):
172            if item.get("name") == config.name:
173                items[i] = {**item, **new}
174                return
175        items.append(new)
176    return _with_configurations(path, change)

179def remove(path: Path, name: str) -> list[Configuration]:
180    """Take the configuration named ``name`` out of the file."""
181    return _with_configurations(path, lambda items: items.__setitem__(
182        slice(None), [item for item in items if item.get("name") != name]))

192def _object(data: dict, key: str, path: Path) -> dict | None:
193    value = data.get(key)
194    if value is not None and not isinstance(value, dict):
195        raise ConfigError(f"{Path(path).name}: \"{key}\" isn't an object.")

202class ConfigFile:
203    """A project's configuration file, as the app has it: ``configurations``,
204    New configuration's Build options (``new_configuration``: None if the
205    file has none) and the GUI's ``options`` (None: there's no file, or it
206    can't be read, ``error`` saying why; then none of it is used, and it
207    isn't written over)."""
208
209    def __init__(self, path: Path):
210        self.path = Path(path)
211        self.configurations: list[Configuration] = []
212        self.new_configuration: Configuration | None = None
213        self.options: dict | None = None
214        self.error = ""
215        self._written: tuple | None = None  # (options, new configuration) as the file has them
216
217    def exists(self) -> bool:
218        return self.path.is_file()
219
220    def _load(self) -> tuple[list[Configuration], dict | None, dict | None]:
221        """Everything in the file, checked (ConfigError if any of it can't be read)."""
222        if not self.exists():
223            return [], None, None
224        configurations = load(self.path)
225        data = _read(self.path)
226        return configurations, _object(data, "options", self.path), _object(data, "new_configuration", self.path)
227
228    def read(self) -> ConfigFile:
229        """Read it all (the file's options and New configuration replace those had)."""
230        try:
231            configurations, options, new = self._load()
232        except ConfigError as exc:
233            self.configurations, self.options, self.new_configuration = [], None, None
234            self.error, self._written = str(exc), None
235            return self
236        self.configurations, self.error = configurations, ""
237        self._take_gui(options, new)
238        return self
239
240    def _take_gui(self, options: dict | None, new: dict | None):
241        self._written = (options, new) if self.exists() else None
242        self.options = dict(options or {}) if self.exists() else None
243        new = dict(new or {})
244        for key, name in _OLD_OPTION_FIELDS.items():  # (from a file written before: moved)
245            if self.options is not None and key in self.options:
246                new.setdefault(name, self.options.pop(key))
247        self.new_configuration = Configuration.from_json(new, name="") if new else None
248
249    def reread_configurations(self):
250        """Its configurations read again (the project reloaded: the file may
251        have changed; its options and New configuration are the window's
252        until they're written, or it's saved in the editor: see reread)."""
253        try:
254            self.configurations, self.error = self._load()[0], ""
255        except ConfigError as exc:
256            self.configurations, self.error = [], str(exc)
257
258    def reread(self) -> tuple[set[str], bool]:
259        """Read it all again (it was saved in the editor): (the options
260        changed in it since it was last read or written, whether New
261        configuration was). Those changed are taken (``options`` and
262        ``new_configuration``: the file's)."""
263        try:
264            configurations, options, new = self._load()
265        except ConfigError as exc:
266            self.configurations, self.error = [], str(exc)
267            return set(), False
268        self.configurations, self.error = configurations, ""
269        if not self.exists():
270            return set(), False
271        before, before_new = self._written or (None, None)
272        before, options_now = before or {}, options or {}
273        changed = {key for key in {*before, *options_now} if before.get(key, _NONE) != options_now.get(key, _NONE)}
274        new_changed = (new or {}) != (before_new or {})
275        if changed or new_changed:
276            self._take_gui(options, new)
277        return changed, new_changed
278
279    def start_options(self):
280        """It was just made (the first configuration saved): it holds the
281        project's options from now on (none yet)."""
282        if self.options is None and self.exists() and not self.error:
283            self.options, self._written = {}, ({}, None)
284
285    def _check_writable(self):
286        if self.error and self.exists():  # (not written over: fix it, or move it away)
287            raise ConfigError(self.error)
288
289    def save(self, config: Configuration):
290        """Add a configuration, or replace the one with its name (OSError, ConfigError)."""
291        self._check_writable()
292        self.configurations = save(self.path, config)
293
294    def remove(self, name: str):
295        """Take a configuration out (OSError, ConfigError)."""
296        self._check_writable()
297        self.configurations = remove(self.path, name)
298
299    def write_options(self, options: dict, new: Configuration) -> bool:
300        """Write the options and New configuration, if they changed (True:
301        written; OSError, ConfigError)."""
302        if self.options is None or self.error:
303            return False
304        written = (options, new.to_json(named=False))
305        if written == self._written:
306            return False
307        save_options(self.path, *written)
308        self.options, self.new_configuration, self._written = dict(options), new, written
309        return True
310

312def save_options(path: Path, options: dict, new_configuration: dict):
313    """Put ``options`` and New configuration's Build options in the file
314    (which must exist and be readable), its configurations and anything else
315    in it kept, in the order it's read: the configurations, New
316    configuration, the options."""
317    load(path)  # (every configuration checked first: a file that can't be read isn't written over)
318    data = {k: v for k, v in _read(path).items() if k not in ("new_configuration", "options")}
319    data["new_configuration"] = new_configuration
320    data["options"] = options
321    _write(path, data)
