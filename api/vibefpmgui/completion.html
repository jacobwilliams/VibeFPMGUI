vibefpmgui.completion

Popups for the language server: the completion list and argument hints.

Both are tooltip-style windows that never take the keyboard focus, so typing continues in the editor; vibefpmgui.code_editor.CodeEditor passes them the keys they use (Up/Down/Enter/Tab/Escape for the completion list).

 38def place(popup, anchor: QRect, below: bool = True):
 39    """Put a popup just below (or above) a global rectangle, on screen."""
 40    screen = (QApplication.screenAt(anchor.center()) or QApplication.primaryScreen()).availableGeometry()
 41    x = min(max(anchor.left(), screen.left()), screen.right() - popup.width())
 42    y = anchor.bottom() + 2 if below else anchor.top() - popup.height() - 2
 43    if below and y + popup.height() > screen.bottom():
 44        y = anchor.top() - popup.height() - 2
 45    elif not below and y < screen.top():
 46        y = anchor.bottom() + 2
 47    popup.move(x, max(screen.top(), y))

 53    def paint(self, painter, option, index):
 54        super().paint(painter, option, index)
 55        item: lsp_mod.Completion | None = index.data(ITEM_ROLE)
 56        if item is None or not item.kind:
 57            return
 58        painter.save()
 59        painter.setFont(option.font)
 60        color = option.palette.color(QPalette.HighlightedText if option.state & QStyle.State_Selected
 61                                     else QPalette.Text)
 62        color.setAlpha(140)
 63        painter.setPen(color)
 64        painter.drawText(option.rect.adjusted(0, 0, -8, 0), Qt.AlignRight | Qt.AlignVCenter, item.kind)
 65        painter.restore()

 68class CompletionPopup(QFrame):
 69    """The list of completions for the name being typed."""
 70
 71    accepted = Signal(object)  # the Completion chosen (double-click)
 72
 73    def __init__(self, editor):
 74        super().__init__(editor, Qt.ToolTip | Qt.FramelessWindowHint)
 75        self.editor = editor
 76        self.setFrameShape(QFrame.Box)
 77        self.items: list[lsp_mod.Completion] = []
 78        self.start: tuple[int, int] | None = None  # (block number, column) the name starts at
 79        self.list = QListWidget()
 80        self.list.setFocusPolicy(Qt.NoFocus)
 81        self.list.setFrameShape(QFrame.NoFrame)
 82        self.list.setUniformItemSizes(True)
 83        self.list.setItemDelegate(_KindDelegate(self.list))
 84        self.list.setHorizontalScrollBarPolicy(Qt.ScrollBarAlwaysOff)
 85        self.list.currentRowChanged.connect(self._show_detail)
 86        self.list.itemDoubleClicked.connect(lambda item: self.accepted.emit(item.data(ITEM_ROLE)))
 87        self.detail = QLabel()
 88        self.detail.setWordWrap(True)
 89        self.detail.setTextFormat(Qt.PlainText)
 90        self.detail.setContentsMargins(6, 3, 6, 4)
 91        layout = QVBoxLayout(self)
 92        layout.setContentsMargins(1, 1, 1, 1)
 93        layout.setSpacing(0)
 94        layout.addWidget(self.list)
 95        layout.addWidget(self.detail)
 96
 97    def open(self, items: list[lsp_mod.Completion], start: tuple[int, int], prefix: str,
 98             anchor: QRect) -> bool:
 99        """Show ``items`` matching ``prefix``; False (and hidden) if none do."""
100        self.items, self.start = items, start
101        bg, text, border = popup_colors(self.editor)
102        # The popup is never the active window, so give the selection the
103        # active colors explicitly (the inactive ones are hard to see).
104        pal = QApplication.palette()
105        sel, sel_text = pal.color(QPalette.Active, QPalette.Highlight), pal.color(QPalette.Active,
106                                                                                 QPalette.HighlightedText)
107        self.setStyleSheet(f"CompletionPopup {{ background: {bg.name()}; border: 1px solid "
108                           f"{border.name(QColor.HexArgb)}; }} QListWidget, QLabel {{ background: "
109                           f"{bg.name()}; color: {text.name()}; }}"
110                           f"QListWidget::item:selected {{ background: {sel.name()}; color: {sel_text.name()}; }}")
111        for widget in (self, self.list, self.detail):
112            widget.setFont(self.editor.font())
113        if not self.filter(prefix):
114            return False
115        place(self, anchor)
116        self.show()
117        return True
118
119    def filter(self, prefix: str) -> bool:
120        """Show the items matching what has been typed (names starting with
121        it first, then names containing it); False if nothing matches."""
122        lower = prefix.lower()
123        matches = [c for c in self.items if c.label.lower().startswith(lower)]
124        matches += [c for c in self.items if lower and lower in c.label.lower()[1:]
125                    and not c.label.lower().startswith(lower)]
126        if not matches or (len(matches) == 1 and matches[0].label == prefix):
127            self.hide()  # nothing to offer (or it's already typed out)
128            return False
129        self.list.clear()
130        for c in matches:
131            item = QListWidgetItem(c.label)
132            item.setData(ITEM_ROLE, c)
133            self.list.addItem(item)
134        self.list.setCurrentRow(0)
135        metrics = self.list.fontMetrics()
136        row_h = self.list.sizeHintForRow(0)
137        width = max(metrics.horizontalAdvance(c.label + "    " + c.kind) for c in matches[:200]) + 40
138        width = min(max(width, 260), 560)
139        self.list.setFixedSize(QSize(width, row_h * min(MAX_ROWS, len(matches)) + 4))
140        self.detail.setFixedWidth(width)
141        self._show_detail(0)
142        return True
143
144    def _fit(self):
145        """Size the popup to the list plus the detail line. (adjustSize()
146        only ever grows a window, so a shorter list left a gap.)"""
147        detail_h = self.detail.heightForWidth(self.detail.width()) if self.detail.isVisibleTo(self) else 0
148        self.detail.setFixedHeight(max(detail_h, 0))
149        self.setFixedSize(self.list.width() + 2, self.list.height() + detail_h + 2)
150
151    def _show_detail(self, row: int):
152        item = self.list.item(row)
153        c: lsp_mod.Completion | None = item.data(ITEM_ROLE) if item else None
154        text = c.detail if c else ""
155        self.detail.setText(text)
156        self.detail.setVisible(bool(text))
157        self._fit()
158
159    def move_selection(self, step: int):
160        rows = self.list.count()
161        if rows:
162            row = self.list.currentRow() + step
163            self.list.setCurrentRow(max(0, min(rows - 1, row)))
164
165    def current(self) -> lsp_mod.Completion | None:
166        item = self.list.currentItem()
167        return item.data(ITEM_ROLE) if item else None
168
169    def page_rows(self) -> int:
170        return MAX_ROWS - 1
