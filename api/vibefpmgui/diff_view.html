vibefpmgui.diff_view

Showing two versions of a file: side by side, or as one unified list.

align lines the versions up row by row (difflib), and with_context keeps a few unchanged lines around each change, folding the rest into a "⋯ N" row. DiffView draws the rows: each side in a syntax-colored pane with its own line numbers, removed lines tinted red and added ones green, and the two sides scrolling together.

 50def align(old: list[str], new: list[str]) -> list[Row]:
 51    """Every line of both versions, lined up: unchanged lines side by side,
 52    edited ones paired (CHANGED), the rest REMOVED or ADDED. The lines the
 53    two begin and end with alike are matched as they are; only those
 54    between (usually a small part of a file) are compared line by line,
 55    which takes time growing with their number squared (a 12,000-line file
 56    whole: a third of a second)."""
 57    shorter = min(len(old), len(new))
 58    start = 0
 59    while start < shorter and old[start] == new[start]:
 60        start += 1
 61    end = 0  # (lines alike at the end, after those at the start)
 62    while end < shorter - start and old[-1 - end] == new[-1 - end]:
 63        end += 1
 64    rows: list[Row] = [Row(SAME, i + 1, old[i], i + 1, new[i]) for i in range(start)]
 65    matcher = difflib.SequenceMatcher(None, old[start:len(old) - end], new[start:len(new) - end], autojunk=False)
 66    for op, a0, a1, b0, b1 in matcher.get_opcodes():
 67        a0, a1, b0, b1 = a0 + start, a1 + start, b0 + start, b1 + start
 68        if op == "equal":
 69            rows += [Row(SAME, a0 + i + 1, old[a0 + i], b0 + i + 1, new[b0 + i]) for i in range(a1 - a0)]
 70            continue
 71        paired = min(a1 - a0, b1 - b0) if op == "replace" else 0
 72        rows += [Row(CHANGED, a0 + i + 1, old[a0 + i], b0 + i + 1, new[b0 + i]) for i in range(paired)]
 73        rows += [Row(REMOVED, a, old[a - 1], None, None) for a in range(a0 + paired + 1, a1 + 1)]
 74        rows += [Row(ADDED, None, None, b, new[b - 1]) for b in range(b0 + paired + 1, b1 + 1)]
 75    rows += [Row(SAME, len(old) - end + i + 1, old[len(old) - end + i], len(new) - end + i + 1,
 76                 new[len(new) - end + i]) for i in range(end)]
 77    return rows

 80def compare(old: str | None, new: str | None) -> list[Row]:
 81    """Two versions' rows to show (`align`, folded with `with_context`): no

 86def with_context(rows: list[Row], context: int = CONTEXT) -> list[Row]:
 87    """The rows near a change (``context`` unchanged lines around each), the
 88    other unchanged ones folded into GAP rows."""
 89    near = [False] * len(rows)
 90    for i, row in enumerate(rows):
 91        if row.kind != SAME:
 92            for j in range(max(0, i - context), min(len(rows), i + context + 1)):
 93                near[j] = True
 94    out: list[Row] = []
 95    i = 0
 96    while i < len(rows):
 97        if near[i]:
 98            out.append(rows[i])
 99            i += 1
100            continue
101        j = i
102        while j < len(rows) and not near[j]:
103            j += 1
104        out.append(Row(GAP, None, None, None, None, j - i))
105        i = j
106    return out

109class PaneRows(NamedTuple):

117def side_by_side(rows: list[Row]) -> tuple[PaneRows, PaneRows]:
118    """The older and the newer version's panes: a row each per Row."""
119    old, new = PaneRows([], [], [], []), PaneRows([], [], [], [])
120    for row in rows:
121        for pane, number, line, kind, mark in ((old, row.old_no, row.old, REMOVED, "-"),
122                                              (new, row.new_no, row.new, ADDED, "+")):
123            pane.new_lines.append(row.new_no)
124            if row.kind == GAP:
125                pane.text.append("")
126                pane.gutter.append((f"⋯ {row.hidden:,}", ""))
127                pane.kinds.append(GAP)
128                continue
129            changed = row.kind in (CHANGED, kind)
130            pane.text.append(line if line is not None else "")
131            pane.gutter.append((str(number) if number else "", mark if changed else ""))
132            pane.kinds.append(kind if changed else FILLER if number is None else SAME)
133    return old, new

136def unified(rows: list[Row]) -> PaneRows:
137    """One pane: unchanged lines once; within each run of changes, the
138    removed lines and then the added ones."""
139    pane = PaneRows([], [], [], [])
140    removed: list[Row] = []
141    added: list[Row] = []
142
143    def flush():
144        for row in removed:
145            pane.text.append(row.old)
146            pane.gutter.append((f"{row.old_no:>5} {'':>5}", "-"))
147            pane.kinds.append(REMOVED)
148            pane.new_lines.append(None)
149        for row in added:
150            pane.text.append(row.new)
151            pane.gutter.append((f"{'':>5} {row.new_no:>5}", "+"))
152            pane.kinds.append(ADDED)
153            pane.new_lines.append(row.new_no)
154        removed.clear()
155        added.clear()
156
157    for row in rows:
158        if row.kind in (REMOVED, CHANGED):
159            removed.append(row)
160        if row.kind in (ADDED, CHANGED):
161            added.append(row)
162        if row.kind in (SAME, GAP):
163            flush()
164            pane.text.append(row.new if row.kind == SAME else "")
165            pane.gutter.append((f"{row.old_no:>5} {row.new_no:>5}", "") if row.kind == SAME
166                               else (f"⋯ {row.hidden:,}", ""))
167            pane.kinds.append(row.kind)
168            pane.new_lines.append(row.new_no)
169    flush()
170    return pane

190    def __init__(self, parent=None):

218    def show_rows(self, rows: PaneRows, path: str):

222        if self.highlighter is None:

227        self._place_gutter()
228        self._tint()
229
230    def refresh_style(self):

232            apply_style(self, self.highlighter)
233        self._tint()
234
235    def set_font(self, font: QFont):

237        self._place_gutter()
238
239    def line_at(self, row: int) -> int | None:
240        """The newer version's line at a row, or the nearest one before it (a
241        removed line has none of its own)."""
242        for line in reversed(self.rows.new_lines[:row + 1]):
243            if line is not None:
244                return line
245        return None
246
247    def _tint(self):
248        """A tint behind removed and added lines; rows with no line, and
249        folded unchanged ones, a little darker than the background."""
250        dark = is_dark(self.palette())
251        base = self.palette().color(QPalette.Base)
252        colors = {REMOVED: QColor(REMOVED_COLOR), ADDED: QColor(ADDED_COLOR)}
253        for color in colors.values():
254            color.setAlpha(70 if dark else 45)
255        colors[FILLER] = colors[GAP] = base.lighter(125) if dark else base.darker(106)
256        selections = []
257        block = self.document().firstBlock()
258        kinds = self.rows.kinds
259        while block.isValid():
260            n = block.blockNumber()
261            color = colors.get(kinds[n]) if n < len(kinds) else None
262            if color is not None:
263                selection = QTextEdit.ExtraSelection()
264                selection.format = QTextCharFormat()
265                selection.format.setBackground(color)
266                selection.format.setProperty(QTextFormat.FullWidthSelection, True)
267                selection.cursor = self.textCursor()
268                selection.cursor.setPosition(block.position())
269                selections.append(selection)
270            block = block.next()
271        self.setExtraSelections(selections)
272
273    def paint_gutter(self, event):
274        painter = QPainter(self.gutter)
275        dark = is_dark(self.palette())
276        base = self.palette().color(QPalette.Base)
277        painter.fillRect(event.rect(), base.lighter(112) if dark else base.darker(104))
278        muted = QColor(self.palette().color(QPalette.Text))
279        muted.setAlpha(130)
280        block = self.firstVisibleBlock()
281        top = round(self.blockBoundingGeometry(block).translated(self.contentOffset()).top())
282        height = self.fontMetrics().height()
283        while block.isValid() and top <= event.rect().bottom():
284            n = block.blockNumber()
285            if n < len(self.rows.gutter):
286                numbers, mark = self.rows.gutter[n]
287                if self.rows.kinds[n] == GAP:
288                    painter.setPen(muted)
289                    painter.drawText(QRect(0, top, self.gutter.width(), height), Qt.AlignCenter, numbers)
290                else:
291                    painter.setPen(REMOVED_COLOR if mark == "-" else ADDED_COLOR if mark == "+" else muted)
292                    painter.drawText(QRect(0, top, self.gutter.width() - 6, height),
293                                     Qt.AlignRight | Qt.AlignVCenter, f"{numbers} {mark or ' '}")
294            top += round(self.blockBoundingRect(block).height())
295            block = block.next()

307        self.rows: list[Row] = []

333        self.unified = on

356        else:

359            self.new_pane.show_rows(new, self.path)
360
361    def refresh_style(self):
362        for pane in self.panes():
363            pane.refresh_style()

369    def eventFilter(self, obj, event):
370        if event.type() == event.Type.MouseButtonDblClick:
371            pane = next((p for p in self.panes() if p.viewport() is obj), None)
372            if pane is not None:

138    """One pane: unchanged lines once; within each run of changes, the
139    removed lines and then the added ones."""
140    pane = PaneRows([], [], [], [])
141    removed: list[Row] = []
142    added: list[Row] = []
143
144    def flush():
145        for row in removed:
146            pane.text.append(row.old)
147            pane.gutter.append((f"{row.old_no:>5} {'':>5}", "-"))
148            pane.kinds.append(REMOVED)
149            pane.new_lines.append(None)
150        for row in added:
151            pane.text.append(row.new)
152            pane.gutter.append((f"{'':>5} {row.new_no:>5}", "+"))
153            pane.kinds.append(ADDED)
154            pane.new_lines.append(row.new_no)
155        removed.clear()
156        added.clear()
157
158    for row in rows:
159        if row.kind in (REMOVED, CHANGED):
160            removed.append(row)
161        if row.kind in (ADDED, CHANGED):
162            added.append(row)
163        if row.kind in (SAME, GAP):
164            flush()
165            pane.text.append(row.new if row.kind == SAME else "")
166            pane.gutter.append((f"{row.old_no:>5} {row.new_no:>5}", "") if row.kind == SAME
167                               else (f"⋯ {row.hidden:,}", ""))
168            pane.kinds.append(row.kind)
169            pane.new_lines.append(row.new_no)
170    flush()
171    return pane
