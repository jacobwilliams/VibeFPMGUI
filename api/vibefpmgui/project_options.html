vibefpmgui.project_options

A project's GUI options, in its build configuration file's "options" (see vibefpmgui.build_configs): the View menu's choices, the panels' layout, the editor's tabs and the files open in them… everything the app keeps in its settings that's about how the project is worked on (not how the app looks: that's the person's, wherever they work). (New configuration's Build options are in the file too, but as a configuration: see vibefpmgui.window.options.)

Not in it (kept in the app's settings, for this computer): the look (theme, window style, syntax colors, editor font, text size), the window's size and place, the recent lists and the last project, the command palette's recent commands, the spelling dictionary, the AI choices, the fpm executable, which configuration file a project uses, and what the app has learned (tests' durations, which build folders a build wrote).

LayeredSettings stands in for the app's QSettings: while a project's options are in use (options not None), an option is read from them (the app's value if they don't have it yet) and written to them; anything else, and every option while there are none, is the app's. An option the app keeps per project (PER_PROJECT) is asked for by its name; the app's key for it is the project's (see use). In the file, the open files and the Project files tree's folders inside the project are relative to it (so it works wherever the project is).

 58def kind_of(key: str):
 59    """The option's kind (None: not an option, always the app's)."""
 60    if key in OPTIONS:
 61        return OPTIONS[key]
 62    return next((kind for prefix, kind in PREFIXES.items() if key.startswith(prefix)), None)

 65def _truthy(value) -> bool:
 66    return value in (True, 1, "true", "1") if not isinstance(value, bool) else value

 69def to_file(value, kind):
 70    """A value as the file has it (JSON's types)."""
 71    if kind == BYTES:
 72        return bytes(QByteArray(value).toBase64()).decode("ascii") if value else ""
 73    if kind == JSON:
 74        return json.loads(value) if isinstance(value, str) and value else (value or {})
 75    if kind is bool:
 76        return _truthy(value)
 77    if kind is int:
 78        return int(value)
 79    if kind is float:
 80        return float(value)
 81    if kind is list:
 82        return [str(v) for v in value] if isinstance(value, (list, tuple)) else ([str(value)] if value else [])
 83    return "" if value is None else str(value)

 86def from_file(value, kind, default):
 87    """A value from the file, as the app's settings would give it (``default``
 88    if it's of the wrong type)."""
 89    if kind == BYTES:
 90        return QByteArray.fromBase64(value.encode("ascii")) if isinstance(value, str) and value else default
 91    if kind == JSON:
 92        return json.dumps(value) if isinstance(value, dict) else default
 93    if kind is bool:
 94        return value if isinstance(value, bool) else default
 95    if kind is int:
 96        return value if isinstance(value, int) and not isinstance(value, bool) else default
 97    if kind is float:
 98        return float(value) if isinstance(value, (int, float)) and not isinstance(value, bool) else default
 99    if kind is list:
100        return list(value) if isinstance(value, list) else default
101    return value if isinstance(value, str) else default

104def _relative(path: str, root: Path) -> str:
105    try:
106        return Path(path).relative_to(root).as_posix() or "."
107    except ValueError:
108        return path

111def _absolute(path: str, root: Path) -> str:
112    return path if Path(path).is_absolute() else str(root / path)

115def _session_paths(state: dict, root: Path, change) -> dict:
116    """The session's paths (open files, the one in front, the tree's open
117    folders and selection) changed with ``change`` (to relative or absolute)."""
118    state = json.loads(json.dumps(state))  # (a copy)
119    for f in state.get("files") or []:
120        if isinstance(f, dict) and isinstance(f.get("path"), str):
121            f["path"] = change(f["path"], root)
122    if isinstance(state.get("current"), str):
123        state["current"] = change(state["current"], root)
124    tree = state.get("tree")
125    if isinstance(tree, dict):
126        tree["expanded"] = [p if not isinstance(p, str) or p.startswith("group:") else change(p, root)
127                            for p in tree.get("expanded") or []]
128        if isinstance(tree.get("selected"), str):
129            tree["selected"] = change(tree["selected"], root)
130    return state

134    """The app's settings (``app``), with a project's options over them while

137    def __init__(self, app):
138        self.app = app
139        self.options: dict | None = None
140        self.root: Path | None = None  # the project's folder (the session's paths are relative to it)
141        self.project_keys: dict[str, str] = {}  # the app's key of each PER_PROJECT option, for this project
142
143    def use(self, options: dict | None, root: Path | None = None, project_keys: dict[str, str] | None = None):
144        """Use a project's options from now on (None: the app's settings
145        only); ``project_keys``: the app's key of each PER_PROJECT option for
146        the project (none: there's no project)."""
147        self.options = None if options is None else dict(options)
148        self.root = root
149        self.project_keys = dict(project_keys or {})
150
151    def _app_key(self, key: str) -> str | None:
152        """The app's key for ``key`` (None: a per-project one, with no project)."""
153        return self.project_keys.get(key) if key in PER_PROJECT else key
154
155    def _in_options(self, key: str) -> bool:
156        return self.options is not None and kind_of(key) is not None
157
158    def value(self, key: str, default=None, type=None):
159        kind = kind_of(key)
160        if self._in_options(key) and key in self.options:
161            value = self.options[key]
162            if kind == JSON and isinstance(value, dict) and self.root is not None:
163                value = _session_paths(value, self.root, _absolute)
164            fallback = default if default is not None else self._default(kind)
165            return from_file(value, kind, fallback)
166        source = self._app_key(key)
167        if source is None:
168            return default
169        return self.app.value(source, default) if type is None else self.app.value(source, default, type=type)
170
171    def setValue(self, key: str, value):
172        if self._in_options(key):
173            kind = kind_of(key)
174            value = to_file(value, kind)
175            if kind == JSON and isinstance(value, dict) and self.root is not None:
176                value = _session_paths(value, self.root, _relative)
177            self.options[key] = value
178        elif (source := self._app_key(key)) is not None:
179            self.app.setValue(source, value)
180
181    def contains(self, key: str) -> bool:
182        if self._in_options(key) and key in self.options:
183            return True
184        source = self._app_key(key)
185        return source is not None and self.app.contains(source)
186
187    def remove(self, key: str):
188        if self._in_options(key):
189            self.options.pop(key, None)
190        elif (source := self._app_key(key)) is not None:
191            self.app.remove(source)
192
193    @staticmethod
194    def _default(kind):
195        return {bool: False, int: 0, float: 0.0, list: [], str: ""}.get(kind)
196
197    def complete_options(self) -> dict:
198        """The options to write: those set, and the app's values for the
199        others it has (so the file has them all)."""
200        out = {}
201        prefixed = {k for k in [*self.app.allKeys(), *(self.options or {})] if any(k.startswith(p) for p in PREFIXES)}
202        names = set(OPTIONS) | prefixed
203        for name in sorted(names):
204            if name in (self.options or {}):
205                out[name] = self.options[name]
206                continue
207            source = self._app_key(name)
208            if source is None or not self.app.contains(source):
209                continue
210            kind = kind_of(name)
211            raw = self.app.value(source)
212            try:
213                value = to_file(raw if kind is not bool else _truthy(raw), kind)
214            except (TypeError, ValueError):
215                continue
216            if kind == JSON and isinstance(value, dict) and self.root is not None:
217                value = _session_paths(value, self.root, _relative)
218            out[name] = value
219        return out
220
221    def __getattr__(self, name):
222        return getattr(self.app, name)
