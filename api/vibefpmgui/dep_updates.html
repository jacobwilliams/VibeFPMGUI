vibefpmgui.dep_updates

Check git dependencies for newer versions (only when the user asks).

For each git dependency the remote is asked for its tags and branches with git ls-remote (nothing is downloaded), and compared with what the project requests (tag / branch / rev / nothing) and the commit fpm checked out (build/cache.toml). A dependency pinned to an older tag can be moved to the newest tag: only that tag value in the declaring fpm.toml is changed.

 49@dataclass
 50class DepCheck:
 51    name: str
 52    git: str
 53    ref_kind: str = ""          # "tag", "branch", "rev" or "" (default branch)

 64    @property
 65    def requested(self) -> str:
 66        if self.ref_kind == "rev":
 67            return f"rev {self.ref[:7]}"
 68        return f"{self.ref_kind} {self.ref}" if self.ref_kind else "default branch"

 72def git_dependencies(graph: m.DependencyGraph) -> list[DepCheck]:
 73    """Every git dependency, with how it is requested and where."""
 74    checks = []
 75    for pkg in graph.packages.values():
 76        if not pkg.git or pkg.kind in (m.ROOT, m.DEV_ONLY):  # (a dependency's dev-dependency isn't fetched)
 77            continue
 78        check = DepCheck(pkg.name, pkg.git, current=pkg.git_rev)
 79        # Prefer a declaration in a manifest the user can edit.
 80        parents = [graph.packages[e.parent] for e in graph.parents(pkg.name) if e.parent in graph.packages]
 81        parents.sort(key=lambda p: p.kind not in (m.ROOT, m.PATH))
 82        for parent in parents:
 83            spec = _declared_spec(parent.manifest, pkg.name) if parent.manifest else None
 84            if spec is not None:
 85                check.declared_in = parent.manifest
 86                check.editable = parent.kind in (m.ROOT, m.PATH)
 87                check.git = str(spec.get("git", check.git))
 88                for key in ("tag", "branch", "rev"):
 89                    if key in spec:
 90                        check.ref_kind, check.ref = key, str(spec[key])
 91                        break
 92                break
 93        checks.append(check)
 94    return sorted(checks, key=lambda c: (not c.editable, c.name.lower()))

 97def _declared_spec(manifest: Path, name: str) -> dict | None:
 98    try:
 99        data = m.load_toml(manifest)
100    except (OSError, tomllib.TOMLDecodeError):
101        return None
102    for section in ("dependencies", "dev-dependencies"):
103        spec = (data.get(section) or {}).get(name)
104        if isinstance(spec, dict):
105            return spec
106    return None

113def git_environment(environ=os.environ) -> dict[str, str]:
114    """The environment for git commands that must never wait for a prompt:
115    no password questions (HTTPS), and SSH in batch mode (no host-key or
116    passphrase questions: a key in the agent still works; otherwise it fails
117    at once instead of waiting out the timeout). A GIT_SSH_COMMAND or
118    GIT_SSH of the user's own is left as it is.
119
120    Only real transports are allowed: a URL comes from a project's fpm.toml,
121    which may not be trustworthy, and git's "ext::" transport (and other
122    remote helpers) would run the command in the URL."""
123    env = dict(environ, GIT_TERMINAL_PROMPT="0", GIT_ALLOW_PROTOCOL=ALLOWED_GIT_PROTOCOLS)
124    if not env.get("GIT_SSH_COMMAND") and not env.get("GIT_SSH"):
125        env["GIT_SSH_COMMAND"] = "ssh -o BatchMode=yes"
126    return env

129def git_error(stderr: str) -> str:
130    """What went wrong, from git's message: its first line (the cause, e.g.
131    "Permission denied (publickey)") and its "fatal:" one, with a hint when
132    SSH couldn't ask for a key or a host key (see git_environment)."""
133    lines = [line.strip() for line in stderr.splitlines() if line.strip()]
134    if not lines:
135        return ""
136    fatal = next((line.removeprefix("fatal: ") for line in lines if line.startswith("fatal:")), "")
137    first = lines[0].removeprefix("fatal: ")
138    text = first if not fatal or fatal == first else f"{first} — {fatal}"
139    if "publickey" in stderr or "Host key verification failed" in stderr:
140        text += (" (SSH can't ask for a key or a host's fingerprint here: load the key into ssh-agent "
141                 "and connect once in a terminal, or use the repository's https:// URL)")
142    return text

145def ls_remote(url: str) -> dict[str, str]:
146    """{ref: commit} for HEAD, branches and tags (annotated tags resolved
147    to their commit). Raises RuntimeError with git's message on failure."""
148    env = git_environment()
149    try:
150        # ("--": a URL from fpm.toml that starts with "-", such as
151        # "--upload-pack=<command>", is a URL to git, never an option.)
152        out = subprocess.run(["git", "ls-remote", "--", url, "HEAD", "refs/heads/*", "refs/tags/*"],
153                             capture_output=True, text=True, timeout=LS_REMOTE_TIMEOUT, env=env, check=False,
154                             **NO_WINDOW)
155    except FileNotFoundError:
156        raise RuntimeError("git was not found on PATH")
157    except subprocess.TimeoutExpired:
158        raise RuntimeError(f"no answer from the server within {LS_REMOTE_TIMEOUT} s")
159    if out.returncode != 0:
160        raise RuntimeError(git_error(out.stderr or "") or f"git ls-remote failed ({out.returncode})")
161    refs: dict[str, str] = {}
162    for line in out.stdout.splitlines():
163        sha, _, ref = line.partition("\t")
164        if ref.endswith("^{}"):
165            refs[ref[:-3]] = sha  # the commit an annotated tag points to
166        else:
167            refs.setdefault(ref, sha)
168    return refs

171def version_key(tag: str) -> tuple:
172    """Sort key for tags like v1.2.3 / 1.2 / release-2.0.1."""
173    numbers = tuple(int(n) for n in re.findall(r"\d+", tag))
174    return (numbers, not PRERELEASE_RE.search(tag), tag)

177def newest_tag(tags: list[str], allow_prerelease: bool = False) -> str:
178    candidates = [t for t in tags if re.search(r"\d", t)
179                  and (allow_prerelease or not PRERELEASE_RE.search(t))]
180    return max(candidates, key=version_key) if candidates else ""

183def check_dependency(check: DepCheck) -> DepCheck:
184    try:
185        refs = ls_remote(check.git)
186    except RuntimeError as exc:
187        check.status, check.message = ERROR, f"Could not reach the repository: {exc}"
188        return check
189    tags = {ref[len("refs/tags/"):]: sha for ref, sha in refs.items() if ref.startswith("refs/tags/")}
190    allow_pre = check.ref_kind == "tag" and bool(PRERELEASE_RE.search(check.ref))
191    check.latest_tag = newest_tag(list(tags), allow_pre)
192    check.latest_tag_commit = tags.get(check.latest_tag, "")
193    if check.ref_kind == "branch":
194        check.branch_commit = refs.get(f"refs/heads/{check.ref}", "")
195    else:
196        check.branch_commit = refs.get("HEAD", "")
197    current = check.current
198
199    if check.ref_kind == "tag":
200        if check.ref not in tags:
201            check.status, check.message = ERROR, f"Tag {check.ref} no longer exists in the repository"
202        elif check.latest_tag and version_key(check.latest_tag) > version_key(check.ref):
203            check.status = UPDATE
204            check.message = f"Newer tag available: {check.ref} → {check.latest_tag}"
205        else:
206            check.status, check.message = UP_TO_DATE, f"{check.ref} is the newest tag"
207    elif check.ref_kind == "rev":
208        pinned = check.ref
209        if not check.latest_tag:
210            check.status, check.message = INFO, "Pinned to a commit (the repository has no tags)"
211        elif check.latest_tag_commit and (check.latest_tag_commit.startswith(pinned)
212                                          or pinned.startswith(check.latest_tag_commit)):
213            check.status = TAG_AVAILABLE
214            check.message = f"Pinned to commit {pinned[:7]}, which is tag {check.latest_tag}"
215        else:
216            check.status = TAG_AVAILABLE
217            check.message = (f"Pinned to a commit; the newest tag is {check.latest_tag} "
218                             "(Compare shows whether it is newer)")
219    else:  # a branch, or the default branch
220        where = f"branch {check.ref}" if check.ref_kind == "branch" else "the default branch"
221        if not check.branch_commit:
222            check.status, check.message = ERROR, f"Could not find {where} in the repository"
223        elif not current:
224            check.status, check.message = INFO, "Not downloaded yet — Update deps fetches it"
225        elif check.branch_commit.startswith(current) or current.startswith(check.branch_commit):
226            check.status, check.message = UP_TO_DATE, f"Up to date with {where}"
227        else:
228            check.status = NEW_COMMITS
229            check.message = f"{where.capitalize()} has new commits — Update deps fetches them"
230    return check

234def set_ref(manifest: Path, name: str, old_key: str, old: str, new_key: str, new: str) -> bool:
235    """Change ``old_key = "old"`` to ``new_key = "new"`` (e.g. a ``rev`` pin
236    to a ``tag``) for one dependency, leaving everything else in the file
237    (comments, layout) untouched. Returns False if the declaration couldn't
238    be found unambiguously."""
239    text = manifest.read_text()
240    tag_re = re.compile(r"(\b)" + re.escape(old_key) + r"""(\s*=\s*)(["'])""" + re.escape(old) + r"""\3""")
241    lines = text.split("\n")
242    target = None
243    # 1. Inline table:  name = { git = "...", tag = "old" }
244    key = re.compile(r"""^\s*(?:["']?)""" + re.escape(name) + r"""(?:["']?)\s*=\s*\{""")
245    for i, line in enumerate(lines):
246        if key.match(line) and tag_re.search(line):
247            target = i
248            break
249    # 2. Table form:  [dependencies.name] ... tag = "old"
250    if target is None:
251        header = re.compile(r"^\s*\[\s*(?:dev-)?dependencies\.(?:[\"']?)" + re.escape(name)
252                            + r"(?:[\"']?)\s*\]")
253        for i, line in enumerate(lines):
254            if header.match(line):
255                for j in range(i + 1, len(lines)):
256                    if lines[j].lstrip().startswith("["):
257                        break
258                    if tag_re.search(lines[j]):
259                        target = j
260                        break
261                break
262    if target is None:
263        return False
264    lines[target] = tag_re.sub(lambda mt: f"{mt.group(1)}{new_key}{mt.group(2)}{mt.group(3)}{new}{mt.group(3)}",
265                               lines[target], count=1)
266    updated = "\n".join(lines)
267    # Make sure the result parses and says what we meant.
268    try:
269        data = tomllib.loads(updated)
270    except tomllib.TOMLDecodeError:
271        return False
272    spec = (data.get("dependencies") or {}).get(name) or (data.get("dev-dependencies") or {}).get(name)
273    if not isinstance(spec, dict) or spec.get(new_key) != new or (old_key != new_key and old_key in spec):
274        return False
275    manifest.write_text(updated)
276    return True

279def read_cache_revs(project_dir: Path) -> dict[str, str]:
280    """{dependency: commit} for the downloaded git dependencies, from each
281    checkout itself (build/cache.toml's "rev" isn't updated by fpm update)."""
282    revs: dict[str, str] = {}
283    folder = project_dir / "build" / "dependencies"
284    if folder.is_dir():
285        for dep in folder.iterdir():
286            commit = m.checked_out_commit(dep)
287            if commit:
288                revs[dep.name] = commit
289    return revs

293class CheckWorker(QObject):
294    result = Signal(int, object)   # row, DepCheck
295    finished = Signal()
296
297    def __init__(self, checks: list[DepCheck]):
298        super().__init__()
299        self.checks = checks
300
301    @Slot()
302    def run(self):
303        try:
304            with ThreadPoolExecutor(max_workers=6) as pool:
305                futures = {pool.submit(check_dependency, c): i for i, c in enumerate(self.checks)}
306                for future in as_completed(futures):
307                    row = futures[future]
308                    try:
309                        check = future.result()
310                    except Exception as exc:  # noqa: BLE001 - that row's error, not the end of the check
311                        check = self.checks[row]
312                        check.status, check.message = ERROR, f"Could not be checked: {exc}"
313                    self.result.emit(row, check)
314        finally:
315            self.finished.emit()  # (always: it ends the thread)

def ls_remote(url: str) -> dict[str, str]: 
146def ls_remote(url: str) -> dict[str, str]:
147    """{ref: commit} for HEAD, branches and tags (annotated tags resolved
148    to their commit). Raises RuntimeError with git's message on failure."""
149    env = git_environment()
150    try:
151        # ("--": a URL from fpm.toml that starts with "-", such as
152        # "--upload-pack=<command>", is a URL to git, never an option.)
153        out = subprocess.run(["git", "ls-remote", "--", url, "HEAD", "refs/heads/*", "refs/tags/*"],
154                             capture_output=True, text=True, timeout=LS_REMOTE_TIMEOUT, env=env, check=False,
155                             **NO_WINDOW)
156    except FileNotFoundError:
157        raise RuntimeError("git was not found on PATH")
158    except subprocess.TimeoutExpired:
159        raise RuntimeError(f"no answer from the server within {LS_REMOTE_TIMEOUT} s")
160    if out.returncode != 0:
161        raise RuntimeError(git_error(out.stderr or "") or f"git ls-remote failed ({out.returncode})")
162    refs: dict[str, str] = {}
163    for line in out.stdout.splitlines():
164        sha, _, ref = line.partition("\t")
165        if ref.endswith("^{}"):
166            refs[ref[:-3]] = sha  # the commit an annotated tag points to
167        else:
168            refs.setdefault(ref, sha)
169    return refs

def set_ref( manifest: pathlib.Path, name: str, old_key: str, old: str, new_key: str, new: str) -> bool: 
235def set_ref(manifest: Path, name: str, old_key: str, old: str, new_key: str, new: str) -> bool:
236    """Change ``old_key = "old"`` to ``new_key = "new"`` (e.g. a ``rev`` pin
237    to a ``tag``) for one dependency, leaving everything else in the file
238    (comments, layout) untouched. Returns False if the declaration couldn't
239    be found unambiguously."""
240    text = manifest.read_text()
241    tag_re = re.compile(r"(\b)" + re.escape(old_key) + r"""(\s*=\s*)(["'])""" + re.escape(old) + r"""\3""")
242    lines = text.split("\n")
243    target = None
244    # 1. Inline table:  name = { git = "...", tag = "old" }
245    key = re.compile(r"""^\s*(?:["']?)""" + re.escape(name) + r"""(?:["']?)\s*=\s*\{""")
246    for i, line in enumerate(lines):
247        if key.match(line) and tag_re.search(line):
248            target = i
249            break
250    # 2. Table form:  [dependencies.name] ... tag = "old"
251    if target is None:
252        header = re.compile(r"^\s*\[\s*(?:dev-)?dependencies\.(?:[\"']?)" + re.escape(name)
253                            + r"(?:[\"']?)\s*\]")
254        for i, line in enumerate(lines):
255            if header.match(line):
256                for j in range(i + 1, len(lines)):
257                    if lines[j].lstrip().startswith("["):
258                        break
259                    if tag_re.search(lines[j]):
260                        target = j
261                        break
262                break
263    if target is None:
264        return False
265    lines[target] = tag_re.sub(lambda mt: f"{mt.group(1)}{new_key}{mt.group(2)}{mt.group(3)}{new}{mt.group(3)}",
266                               lines[target], count=1)
267    updated = "\n".join(lines)
268    # Make sure the result parses and says what we meant.
269    try:
270        data = tomllib.loads(updated)
271    except tomllib.TOMLDecodeError:
272        return False
273    spec = (data.get("dependencies") or {}).get(name) or (data.get("dev-dependencies") or {}).get(name)
274    if not isinstance(spec, dict) or spec.get(new_key) != new or (old_key != new_key and old_key in spec):
275        return False
276    manifest.write_text(updated)
277    return True
