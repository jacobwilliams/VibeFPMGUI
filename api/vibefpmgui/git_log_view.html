vibefpmgui.git_log_view

The Git panel's History page (see vibefpmgui.git_log): the commits, newest first, each with the graph's lanes (its dot, the lines to its parents: a column of their own, wide enough for every lane), its branches and tags (badges: the branch checked out in bold), subject, author, date and hash; every branch's, or only what's checked out (All branches). The newest MAX_COMMITS at first; Load Older Commits lists the rest, back to the first. Choosing a commit shows its card (as git blame's: the whole message, links to its web page and issues), the files it changed, and the chosen file's diff (double-click a line: the file, there).

 53COLUMNS = ["Graph", "Commit", "Author", "Date", "Hash"]
 54GRAPH, SUBJECT = 0, 1  # (the columns drawn by GraphDelegate)
 55GRAPH_SHARE = 0.6  # the most of the list's width the graph is given at first (wider: drag its edge)

 70class LogModel(QAbstractTableModel):
 71    def __init__(self, parent=None):
 72        super().__init__(parent)
 73        self.log: gl.Log | None = None
 74        self.lanes = 1  # the graph's width (lanes), the widest row's (every lane: none cut off)
 75
 76    def set_log(self, log: gl.Log | None):
 77        self.beginResetModel()
 78        self.log = log
 79        self.lanes = max((r.width for r in log.rows), default=1) if log else 1
 80        self.endResetModel()
 81
 82    def rowCount(self, parent=QModelIndex()):  # noqa: B008 - Qt's signature
 83        return 0 if parent.isValid() or self.log is None else len(self.log.commits)
 84
 85    def columnCount(self, parent=QModelIndex()):  # noqa: B008 - Qt's signature
 86        return 0 if parent.isValid() else len(COLUMNS)
 87
 88    def headerData(self, section, orientation, role=Qt.DisplayRole):
 89        if orientation == Qt.Horizontal and role == Qt.DisplayRole:
 90            return COLUMNS[section]
 91        return None
 92
 93    def data(self, index, role=Qt.DisplayRole):
 94        if not index.isValid() or self.log is None:
 95            return None
 96        commit = self.log.commits[index.row()]
 97        column = index.column()
 98        if role == Qt.DisplayRole:
 99            return ["", commit.subject, commit.author, date_words(commit.time), commit.sha[:8]][column]
100        if role == Qt.ToolTipRole:
101            refs = ", ".join(r.name for r in commit.refs)
102            subject = f"{commit.subject}" + (f"\n{refs}" if refs else "")
103            return [subject, subject, f"{commit.author} <{commit.email}>", None, commit.sha][column]
104        if role == COMMIT_ROLE:
105            return commit
106        if role == ROW_ROLE:
107            return self.log.rows[index.row()]
108        return None

111def lane_width(font: QFont) -> float:
112    """A lane's width in the graph, for the list's font."""
113    return max(12.0, QFontMetrics(font).height() * 0.8)

116def graph_width(lanes: int, font: QFont) -> int:
117    """The Graph column's width for every one of ``lanes``."""
118    return round(8 + lane_width(font) * lanes)

130        size = super().sizeHint(option, index)

133    def paint(self, painter, option, index):
134        commit: gl.LogCommit = index.data(COMMIT_ROLE)
135        row: gl.GraphRow = index.data(ROW_ROLE)
136        style = option.widget.style() if option.widget else QApplication.style()
137        style.drawPrimitive(QStyle.PE_PanelItemViewItem, option, painter, option.widget)
138        painter.save()
139        painter.setRenderHint(QPainter.Antialiasing)
140        rect = QRectF(option.rect)
141        if index.column() == SUBJECT:
142            self._paint_subject(painter, option, commit, rect)
143            painter.restore()
144            return
145        lane = lane_width(option.font)
146        colors = LANE_COLORS["dark" if self.view.dark else "light"]
147        top, bottom, middle = rect.top(), rect.bottom() + 1, rect.center().y()
148
149        def x(i: int) -> float:
150            return rect.left() + 4 + lane * (i + 0.5)
151
152        def pen(color: int) -> QPen:
153            line = QPen(QColor(colors[color % len(colors)]), 2)
154            line.setCapStyle(Qt.RoundCap)
155            return line
156
157        painter.setClipRect(rect)  # (the column made narrower than the graph: the rest cut off)
158        for i, color in row.through:
159            painter.setPen(pen(color))
160            painter.drawLine(QPointF(x(i), top), QPointF(x(i), bottom))
161        for i, color in row.joining:  # (from its lane above, curving into the commit)
162            painter.setPen(pen(color))
163            path = QPainterPath(QPointF(x(i), top))
164            path.cubicTo(QPointF(x(i), middle), QPointF(x(i), middle), QPointF(x(row.lane), middle))
165            painter.drawPath(path)
166        if row.from_above:
167            painter.setPen(pen(row.color))
168            painter.drawLine(QPointF(x(row.lane), top), QPointF(x(row.lane), middle))
169        for i, color in row.parents:
170            painter.setPen(pen(color))
171            if i == row.lane:
172                painter.drawLine(QPointF(x(i), middle), QPointF(x(i), bottom))
173            else:  # (a merge's other parent, or a lane waiting for it: curving out to it)
174                path = QPainterPath(QPointF(x(row.lane), middle))
175                path.cubicTo(QPointF(x(i), middle), QPointF(x(i), middle), QPointF(x(i), bottom))
176                painter.drawPath(path)
177        radius = lane * 0.28
178        dot = QColor(colors[row.color % len(colors)])
179        center = QPointF(x(row.lane), middle)
180        is_head = self.view.model.log is not None and commit.sha == self.view.model.log.head
181        if len(commit.parents) > 1:  # (a merge: a ring)
182            painter.setPen(QPen(dot, 2))
183            painter.setBrush(QBrush(QColor(self.view.pal.get("bg", "#ffffff"))))
184        else:
185            painter.setPen(Qt.NoPen)
186            painter.setBrush(QBrush(dot))
187        painter.drawEllipse(center, radius, radius)
188        if is_head:  # (what's checked out: a ring around it)
189            painter.setPen(QPen(QColor(self.view.pal.get("text", "#000000")), 1.5))
190            painter.setBrush(Qt.NoBrush)
191            painter.drawEllipse(center, radius + 3, radius + 3)
192        painter.restore()

194    def _paint_subject(self, painter, option, commit: gl.LogCommit, rect: QRectF):
195        """The commit's branches and tags (badges), then its subject."""
196        selected = bool(option.state & QStyle.State_Selected)
197        middle = rect.center().y()
198        left = rect.left() + 4
199        small = QFont(option.font)
200        small.setPointSizeF(max(7.0, option.font.pointSizeF() - 1.5))
201        badge_metrics = QFontMetrics(small)
202        for ref in commit.refs:
203            if left > rect.right() - 40:
204                break
205            bold = QFont(small)
206            bold.setBold(ref.current)
207            text_width = QFontMetrics(bold).horizontalAdvance(ref.name)
208            badge = QRectF(left, middle - badge_metrics.height() / 2 - 1, text_width + 10, badge_metrics.height() + 2)
209            color = QColor(REF_COLORS[ref.kind])
210            solid = ref.current or selected  # (on the selection's color: solid, so it shows)
211            painter.setPen(QPen(QColor("white") if selected else color, 1.2 if ref.current else 1))
212            fill = QColor(color)
213            fill.setAlphaF(0.9 if solid else 0.18)
214            painter.setBrush(fill)
215            painter.drawRoundedRect(badge, 4, 4)
216            painter.setFont(bold)
217            painter.setPen(QColor("white") if solid else color if not self.view.dark else color.lighter(130))
218            painter.drawText(badge, Qt.AlignCenter, ref.name)
219            left = badge.right() + 4
220        painter.setFont(option.font)
221        painter.setPen(option.palette.color(option.palette.ColorRole.HighlightedText if selected
222                                            else option.palette.ColorRole.Text))
223        text_rect = QRectF(left + 2, rect.top(), rect.right() - left - 4, rect.height())
224        painter.drawText(text_rect, Qt.AlignVCenter | Qt.AlignLeft,
225                         QFontMetrics(option.font).elidedText(commit.subject, Qt.ElideRight, int(text_rect.width())))

419    def _commit_chosen(self, index, _previous=None):
420        commit = index.data(COMMIT_ROLE) if index.isValid() else None
421        self.files.clear()
422        self.file_diff.clear()
423        if commit is None or self.log is None:
424            self._details = None
425            self.card.clear()
426            return
427        self._details = (commit.sha, None)
428        self._show_card(commit, None)
429        root = self.log.root
430
431        def read():
432            try:
433                return gb.commit_details(root, commit.sha)
434            except (gs.GitError, OSError) as error:
435                return str(error) or type(error).__name__
436
437        call_in_background(read, lambda result: self._details_read(commit, result), self)

439    def _show_card(self, commit: gl.LogCommit, details):
440        shown = details
441        if isinstance(details, gb.CommitDetails):  # (its files: in the list under it)
442            shown = gb.CommitDetails(details.email, details.message, details.committer, details.commit_time, None)
443        text = QColor(self.pal.get("text", "#000000"))
444        html = gb.commit_card_html(gb.Commit(commit.sha, commit.author, commit.time, commit.subject), shown,
445                                   self.log.remote if self.log else "", text)
446        self.card.setHtml(html)

448    def _details_read(self, commit: gl.LogCommit, result):
449        if self._details is None or self._details[0] != commit.sha:
450            return  # (another commit chosen meanwhile)
451        self._details = (commit.sha, result)
452        self._show_card(commit, result)
453        if not isinstance(result, gb.CommitDetails):
454            return
455        for change in result.files or []:
456            name = f"{change.old_path} → {change.path}" if change.old_path else change.path
457            counts = "binary" if change.added is None else f"+{change.added} −{change.deleted}"
458            item = QTreeWidgetItem([name, counts, change.status])
459            item.setData(0, CHANGE_ROLE, change)
460            word, color = gb.STATUS_CHIPS.get(change.status, (change.status, "#868e96"))
461            item.setToolTip(0, f"{change.path}\n{word}")
462            item.setTextAlignment(1, Qt.AlignRight | Qt.AlignVCenter)
463            item.setForeground(1, QBrush(QColor(self.pal.get("muted", "#888888"))))
