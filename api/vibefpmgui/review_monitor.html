vibefpmgui.review_monitor

The Review tab: a timeline of everything the AI does (reviews, Explain with AI, questions and edits from the box at the bottom), newest at the bottom. Nothing replaces anything: each action adds an entry, older ones fold to a one-line header (click to open), and everything in them stays usable: findings jump to their line or get Fix with AI…, a review's markers go back in the editor, edits can still be accepted. The timeline lasts until the app quits (or Clear); New Chat draws a line: the model forgets what's above it.

The panel only shows Entry objects and says what the user did; vibefpmgui.window.ai and vibefpmgui.window.ai_chat run the model and change files.

107    def markdown(self) -> str:
108        """The entry as text (Copy)."""
109        if self.kind == "review":
110            lines = [f"**Review of {self.where}** ({self.question})", ""]
111            lines += [f"- LINE {f.line}{'' if f.end_line == f.line else f'-{f.end_line}'}: {f.category}: "
112                      f"{f.comment}" for f in self.findings]
113            return "\n".join(lines + ([f"\n**Summary:** {self.summary}"] if self.summary else []))
114        if self.kind == "explain":
115            return f"**{self.where}:** {self.question}\n\n{self.answer}"
116        return f"**You:** {self.question}\n\n{self.answer}"

119def answer_display(answer: str) -> str:
120    """A chat answer as shown: its "LINE n: …" lines each on their own
121    (Markdown would run them together), as a list; its edit blocks as a
122    pointer to their card."""
123    shown: list[str] = []
124    fenced = in_block = drop_fence = False
125    n = 0
126    for line in answer.split("\n"):
127        fence = line.lstrip().startswith("```")
128        if in_block:
129            if EDIT_END_RE.match(line):
130                in_block = False
131                shown[-1] = f"*✎ Edit {n}: see its card below.*"
132            continue
133        if EDIT_START_RE.match(line):
134            in_block, n = True, n + 1
135            if fenced and shown and shown[-1].lstrip().startswith("```"):  # (a fence just around it)
136                shown.pop()
137                fenced, drop_fence = False, True
138            shown.append(f"*✎ Writing edit {n}…*")
139            continue
140        if drop_fence and fence:  # (that fence's end)
141            drop_fence = False
142            continue
143        if line.strip():
144            drop_fence = False
145        if fence:
146            fenced = not fenced
147        elif not fenced and FINDING_LINE_RE.match(line):
148            line = "- " + line.strip()
149        shown.append(line)
150    return "\n".join(shown)

153def style_code(document, family: str, style_name: str | None = None):
154    """Code in a Markdown document in a fixed-width font (Qt's Markdown asks
155    only for "monospace", which macOS turns into a serif font, or ignores),
156    and code blocks labeled with a language (```fortran, ```toml…)
157    highlighted as the editor would, in its colors (``style_name``: a
158    Pygments style; by default the one the editor uses)."""
159    fixed = QTextCharFormat()
160    fixed.setFontFamilies([family])
161    cursor = QTextCursor(document)
162    block = document.begin()
163    while block.isValid():
164        if block.blockFormat().hasProperty(QTextFormat.BlockCodeFence):
165            language = str(block.blockFormat().property(QTextFormat.BlockCodeLanguage) or "")
166            lines = []
167            while block.isValid() and block.blockFormat().hasProperty(QTextFormat.BlockCodeFence) \
168                    and str(block.blockFormat().property(QTextFormat.BlockCodeLanguage) or "") == language:
169                lines.append(block)
170                block = block.next()
171            cursor.setPosition(lines[0].position())
172            cursor.setPosition(lines[-1].position() + lines[-1].length() - 1, QTextCursor.KeepAnchor)
173            cursor.mergeCharFormat(fixed)
174            _highlight(cursor, lines, language, style_name)
175            continue
176        it = block.begin()
177        while not it.atEnd():
178            fragment = it.fragment()
179            if fragment.charFormat().fontFixedPitch():
180                cursor.setPosition(fragment.position())
181                cursor.setPosition(fragment.position() + fragment.length(), QTextCursor.KeepAnchor)
182                cursor.mergeCharFormat(fixed)
183            it += 1
184        block = block.next()

187def _highlight(cursor: QTextCursor, lines: list, language: str, style_name: str | None):
188    """One code block's lines, colored by the lexer for ``language``."""
189    if not language:
190        return
191    try:
192        lexer = hl.keep_positions(get_lexer_by_name(language))
193        style = get_style_by_name(style_name or hl.style_name(QApplication.palette()))
194    except ClassNotFound:
195        return  # (a language Pygments doesn't know: plain)
196    background = QTextBlockFormat()
197    background.setBackground(QColor(style.background_color))
198    plain = QTextCharFormat()
199    plain.setForeground(hl.style_text_color(style))
200    first = lines[0].position()
201    cursor.setPosition(first)
202    cursor.setPosition(lines[-1].position() + lines[-1].length() - 1, QTextCursor.KeepAnchor)
203    cursor.mergeBlockFormat(background)
204    cursor.mergeCharFormat(plain)
205    text = "\n".join(line.text() for line in lines)  # (a line break is one position, as between blocks)
206    for index, ttype, value in lexer.get_tokens_unprocessed(text):
207        if value and not value.isspace():
208            cursor.setPosition(first + index)
209            cursor.setPosition(first + index + len(value), QTextCursor.KeepAnchor)
210            cursor.mergeCharFormat(hl.token_format(style, ttype))

309class FindingsView(QWidget):
310    """Findings, one row each: click to go to the line; Fix with AI…
311
312    The rows are kept and updated in place (only a row whose finding changed
313    gets new text; new findings add rows at the end): rebuilding every row
314    for each finding a review streamed in took ever longer, and with a few
315    dozen findings the window crawled."""
316
317    clicked = Signal(object)
318    fix = Signal(object)
319
320    def __init__(self, parent=None):
321        super().__init__(parent)
322        self.layout_ = QVBoxLayout(self)
323        self.layout_.setContentsMargins(0, 0, 0, 0)
324        self.layout_.setSpacing(2)
325        self.rows: list[FindingRow] = []
326
327    def show_findings(self, findings: list[Finding], pal: dict, is_light: bool):
328        family = monospace_font(editor=True).family()
329        # Each row stays with its finding (by line and category): a finding
330        # coming in between others gets a new row there, rather than every
331        # row below taking the next one's text (and being measured again).
332        spare: dict[tuple, list[FindingRow]] = {}
333        for row in self.rows:
334            spare.setdefault(self._key(row.finding), []).append(row)
335        rows = []
336        for f in sorted(findings, key=lambda f: f.line):
337            same = spare.get(self._key(f))
338            row = same.pop(0) if same else FindingRow(self)
339            row.finding = f
340            color = CATEGORY_COLORS.get(f.category, pal["muted"])
341            lines = str(f.line) if f.end_line == f.line else f"{f.line}–{f.end_line}"
342            text = (f"<span style='font-family:\"{family}\"'>{lines}</span>"
343                    f"&nbsp;&nbsp;<b style='color:{color}'>{CATEGORY_ICONS.get(f.category, '')} "
344                    f"{_esc(f.category)}</b>&nbsp;&nbsp;{_esc(f.comment)}")
345            if row.label.text() != text:  # (only what changed is measured again)
346                row.label.setText(text)
347            rows.append(row)
348        for left in spare.values():
349            for row in left:
350                self.layout_.removeWidget(row)
351                row.deleteLater()
352        for index, row in enumerate(rows):  # (new rows into their places; the others keep theirs)
353            if self.layout_.indexOf(row) != index:
354                self.layout_.insertWidget(index, row)
355        self.rows = rows
356        self.setVisible(bool(findings))
357
358    @staticmethod
359    def _key(f: Finding | None) -> tuple:
360        return (f.line, f.end_line, f.category) if f is not None else ()
