178    def mousePressEvent(self, event):
179        mods = event.modifiers()
180        if event.button() == Qt.LeftButton:
181            point = event.position().toPoint()
182            if mods & Qt.AltModifier and not mods & (Qt.ControlModifier | Qt.ShiftModifier):
183                self.toggle_cursor_at(self.cursorForPosition(point).position())  # ⌥/Alt-click
184                return
185            if mods & Qt.ControlModifier and mods & Qt.ShiftModifier:  # ⌘⇧/Ctrl+Shift-click
186                clicked = self.cursorForPosition(point)
187                self.add_column_cursors(clicked.blockNumber(), clicked.positionInBlock())
188                return
189            self.clear_extra_cursors()  # (any other click: back to one cursor)
190        if (event.button() == Qt.LeftButton and event.modifiers() & Qt.ControlModifier
191                and self.hover_provider is not None):
192            # ⌘-click (Ctrl-click on Linux/Windows): go to the definition.
193            self.setTextCursor(self.cursorForPosition(event.position().toPoint()))

203    def context_menu(self, pos: QPoint, from_mouse: bool = True) -> QMenu:
204        """The usual Cut/Copy/Paste menu, plus fortitude's fixes for the
205        issues at that spot and the language server's commands for the name
206        there (when one is attached). ``pos``: where the menu was asked for
207        (viewport coordinates); ``from_mouse`` False: from the keyboard, for
208        the text cursor's spot."""
209        menu = self.createStandardContextMenu(pos)
210        # fortitude's fixes for the issues at that spot.
211        spot = self.cursorForPosition(pos).position() if from_mouse else self.textCursor().position()
212        fixes = [i for i in self.lint_issues if i.fixable
213                 and (span := self._lint_span(i)) and span[0] <= spot <= span[1]]
214        misspelled = next((i for i in self.lint_issues if i.severity == "spelling"
215                           and (span := self._lint_span(i)) and span[0] <= spot <= span[1]), None)
216        if misspelled is not None:
217            first = menu.actions()[0] if menu.actions() else None
218            words = misspelled.suggestions()
219            for word in words:
220                action = QAction(word, menu)
221                font = action.font()
222                font.setBold(True)
223                action.setFont(font)
224                action.triggered.connect(lambda _=False, i=misspelled, w=word: self.replace_word(i, w))
225                menu.insertAction(first, action)
226            if not words:
227                none = QAction("(No Spelling Suggestions)", menu)
228                none.setEnabled(False)
229                menu.insertAction(first, none)
230            add = QAction(f"Add “{misspelled.word}” to Dictionary", menu)
231            add.triggered.connect(lambda _=False, w=misspelled.word: self.word_added.emit(w))
232            menu.insertAction(first, add)
233            menu.insertSeparator(first)
234        if self.textCursor().hasSelection():
235            first = menu.actions()[0] if menu.actions() else None
236            edit = QAction("Edit Selection with AI…", menu)
237            edit.setToolTip("Ask the local AI model to change the selected lines; you accept or reject each edit")
238            edit.triggered.connect(self.edit_selection_requested.emit)
239            menu.insertAction(first, edit)
240            menu.insertSeparator(first)
241        line = self.document().findBlock(spot).blockNumber() + 1
242        if any(severity not in ("review", "remark") for severity, _ in self.markers.get(line, [])):
243            first = menu.actions()[0] if menu.actions() else None
244            explain = QAction("Explain Error with AI", menu)
245            explain.setToolTip("Ask the local AI model what this line's problem means, and how to fix it")
246            explain.triggered.connect(lambda _=False, n=line: self.explain_requested.emit(n))
247            menu.insertAction(first, explain)
248            menu.insertSeparator(first)
249        if fixes:
250            first = menu.actions()[0] if menu.actions() else None
251            for issue in fixes:
252                label = f"Fix: {issue.fix or issue.message}" + ("" if issue.fix_safe else "  (unsafe — check it)")
253                action = QAction(label, menu)
254                action.triggered.connect(lambda _=False, i=issue: self.lint_fix_requested.emit(i))
255                menu.insertAction(first, action)
256            menu.insertSeparator(first)
257        if self.hover_provider is not None:
258            on_name = False
259            if from_mouse:
260                point = pos  # (already in viewport coordinates)
261                on_name = self.word_at(point) is not None
262                if on_name:  # the commands act on the name at the cursor
263                    cursor = self.cursorForPosition(point)
264                    if not (self.textCursor().selectionStart() <= cursor.position()
265                            <= self.textCursor().selectionEnd() and self.textCursor().hasSelection()):
266                        self.setTextCursor(cursor)
267            else:  # opened from the keyboard: the name at the cursor
268                rect = self.cursorRect()
269                on_name = self.word_at(rect.center()) is not None
270            first = menu.actions()[0] if menu.actions() else None
271            for text, keys, signal in (("Go to Definition", "F12", self.definition_requested),
272                                       ("Go to Implementation", "Ctrl+F12", self.implementation_requested),
273                                       ("Find References", "Shift+F12", self.references_requested),
274                                       ("Rename Symbol…", "F2", self.rename_requested)):
275                action = QAction(text, menu)
276                action.setShortcut(QKeySequence(keys))  # shown in the menu; the Edit menu's work
277                action.setShortcutContext(Qt.WidgetShortcut)
278                action.setEnabled(on_name)
279                action.triggered.connect(signal.emit)
280                menu.insertAction(first, action)
281            menu.insertSeparator(first)
282        return menu

310    def update_extra_selections(self, rebuild: bool = True):
311        """Current line, lines with errors/warnings, underlined issues,
312        gremlins, the extra cursors' selections, the selection's other
313        occurrences and find matches.
314
315        Only the lines around the view get highlights (see `_line_window`):
316        each is a QTextCursor that Qt moves along at every edit and goes
317        through at every cursor move, and a big file's thousands of issues
318        made typing, and replacing its text, take seconds. Scrolling makes
319        those of the lines coming into view. ``rebuild``: the markers,
320        issues or gremlins changed (or the colors, or the line wrapping).
321
322        Not while the text is being replaced (``setPlainText``, as a file
323        changed on disk is reloaded): emptying the document can narrow the
324        gutter and so scroll, calling this with no valid line anywhere, and a
325        cursor made from one crashed Qt. It's called again once the text is in."""
326        if not self.textCursor().block().isValid():
327            return
328        if rebuild:
329            self._highlights_version += 1
330            self._issue_lines = {}
331            for issue in self.lint_issues:
332                if issue.severity != "spelling":  # (drawn by paintEvent: see _paint_squiggles)
333                    self._issue_lines.setdefault(issue.line, []).append(issue)
334        window = self._line_window()
335        key = (self._highlights_version, window)
336        if key != self._fixed_key:
337            self._fixed_selections = self._marker_and_issue_selections(*window)
338            self._fixed_key = key
339        selections = list(self._fixed_selections)
340        if not self.isReadOnly():
341            color = QColor(self.palette().color(QPalette.Highlight))
342            color.setAlpha(28)
343            selections += self._line_highlight(self.textCursor().block(), color)
344        selected = QColor(self.palette().color(QPalette.Highlight))
345        for c in self.extra_cursors:
346            if c.hasSelection():
347                sel = QTextEdit.ExtraSelection()
348                sel.format.setBackground(selected)
349                sel.format.setForeground(self.palette().color(QPalette.HighlightedText))
350                sel.cursor = QTextCursor(c)
351                selections.append(sel)
352        selections += self._span_selections(self.occurrence_selections, self._occurrence_color, window)
353        selections += self._span_selections(self.find_selections, FIND_COLOR, window)
354        self.setExtraSelections(selections)

356    def _line_window(self) -> tuple[int, int]:
357        """The lines (0-based, first and last) to highlight: those on view, and
358        a screenful either side, in steps of a screenful (scrolling a line
359        doesn't make them all again)."""
360        rows = max(8, self.viewport().height() // max(1, self.fontMetrics().lineSpacing()))
361        first = self.firstVisibleBlock().blockNumber() // rows * rows
362        return max(0, first - rows), first + 3 * rows

368    def _span_selections(self, spans: list[tuple[int, int]], color: QColor, window: tuple[int, int]) -> list:
369        """Highlights for the (start, end) positions, in order, that are in the window's lines."""
370        if not spans:
371            return []
372        doc = self.document()
373        low = doc.findBlockByNumber(window[0]).position()
374        last = doc.findBlockByNumber(window[1])
375        high = last.position() + last.length() if last.isValid() else doc.characterCount()
376        out = []
377        for start, end in spans[bisect_left(spans, (low, 0)):]:
378            if start > high:
379                break
380            sel = QTextEdit.ExtraSelection()
381            sel.format.setBackground(color)
382            sel.cursor = QTextCursor(doc)
383            sel.cursor.setPosition(start)
384            sel.cursor.setPosition(end, QTextCursor.KeepAnchor)
385            out.append(sel)
386        return out

388    def _marker_and_issue_selections(self, first: int, last: int) -> list:
389        """The highlights of lines ``first`` to ``last`` (0-based): the tests'
390        missed lines, then the markers' tints, the issues' squiggles and the
391        gremlins' tints (each drawn over the one before)."""
392        missed = QColor(COVERAGE_COLORS["missed"])
393        missed.setAlpha(26)
394        tint = QColor(gremlins_mod.COLOR)
395        tint.setAlpha(70)
396        coverage, markers, issues, gremlins = [], [], [], []
397        block = self.document().findBlockByNumber(first)
398        while block.isValid() and block.blockNumber() <= last:
399            line = block.blockNumber() + 1
400            if self.show_coverage and self.coverage.get(line) == 0:
401                coverage += self._line_highlight(block, missed)
402            notes = self.markers.get(line)
403            if notes:
404                color = QColor(MARKER_COLORS.get(worst_severity(notes), "#868e96"))
405                color.setAlpha(34)
406                markers += self._line_highlight(block, color)
407            for issue in self._issue_lines.get(line, ()):
408                span = self._lint_span(issue)
409                if span is None:
410                    continue
411                sel = QTextEdit.ExtraSelection()
412                sel.format.setUnderlineStyle(QTextCharFormat.WaveUnderline)
413                sel.format.setUnderlineColor(QColor(MARKER_COLORS[issue.severity]))
414                sel.cursor = QTextCursor(self.document())
415                sel.cursor.setPosition(span[0])
416                sel.cursor.setPosition(span[1], QTextCursor.KeepAnchor)
417                issues.append(sel)
418            for g in self.gremlins.get(line, ()):  # (a tint: an invisible one would show nothing else)
419                if g.column >= block.length() - 1:
420                    continue
421                sel = QTextEdit.ExtraSelection()
422                sel.format.setBackground(tint)
423                sel.cursor = QTextCursor(block)
424                sel.cursor.setPosition(block.position() + g.column)
425                sel.cursor.setPosition(block.position() + g.column + 1, QTextCursor.KeepAnchor)
426                gremlins.append(sel)
427            block = block.next()
428        return coverage + markers + issues + gremlins

431    def _line_highlight(block, color: QColor) -> list:
432        """A full-width background for a line: one per row it is shown in
433        (with View → Word Wrap a long line takes several, and Qt fills only
434        the row the selection's cursor is in). Nothing for a block not in a
435        document: a cursor made from it crashes Qt."""
436        if not block.isValid():
437            return []
438        layout = block.layout()
439        starts = [layout.lineAt(i).textStart() for i in range(layout.lineCount())] if layout else []
440        out = []
441        for start in starts or [0]:
442            sel = QTextEdit.ExtraSelection()
443            sel.format.setBackground(color)
444            sel.format.setProperty(QTextFormat.FullWidthSelection, True)
445            sel.cursor = QTextCursor(block)
446            sel.cursor.setPosition(block.position() + start)
447            out.append(sel)
448        return out

450    def update_occurrences(self):
451        """Highlight the other occurrences of the selected text in the file
452        (single-line selections of 2 to 200 characters; Match case and Whole
453        words as set in the find bar)."""
454        found: list[tuple[int, int]] = []
455        cursor = self.textCursor()
456        selected = cursor.selectedText()
457        if (self.highlight_matches and len(self.extra_cursors) == 0 and 2 <= len(selected) <= 200
458                and selected.strip() and "\u2029" not in selected):
459            case, word = self.match_options()
460            pattern = re.escape(selected)

588    def _indent_selection(self, direction: int):
589        """Indent (Tab) or outdent (⇧Tab) the selected lines, or the cursor's;
590        a selection ending at the start of a line leaves that line alone."""
591        cursor = self.textCursor()

593        end = self.document().findBlock(cursor.selectionEnd())
594        if end != start and cursor.selectionEnd() == end.position():
595            end = end.previous()
596        cursor.beginEditBlock()
597        block = start
598        while block.isValid():
599            c = QTextCursor(block)
600            if direction > 0:
601                if block.text().strip() or block == start == end:  # (blank lines stay blank)
602                    c.insertText(self.indent_unit())
603            else:

605                               outdent_length(block.text(), self.indent_width))
606                c.removeSelectedText()
607            if block == end:
608                break
609            block = block.next()
610        cursor.endEditBlock()
