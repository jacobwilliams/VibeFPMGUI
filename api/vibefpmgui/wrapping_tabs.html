vibefpmgui.wrapping_tabs

The editor's tabs on several rows (View → Editor Tabs → Several Rows), instead of one row that scrolls.

Qt's tab bar has one row only. WrappingTabBar lays the editor's tabs out in as many rows as they need, drawn as Qt draws tabs (the style's own tab shapes), and stands in for the tab bar, which is hidden but stays the tabs' model: the names, icons and tooltips are read from it, and choosing, moving and closing tabs is done through it (so everything else about the tabs works as before).

 40class WrappingTabBar(QWidget):
 41    """A ``QTabWidget``'s tabs, wrapped onto rows."""
 42
 43    close_requested = Signal(int)  # a tab's close glyph clicked
 44    empty_menu_requested = Signal(QPoint)  # a right-click on no tab: where, on the screen

 46    def __init__(self, tabs: QTabWidget, parent=None):
 47        super().__init__(parent)
 48        self.tabs = tabs
 49        self.bar: QTabBar = tabs.tabBar()
 50        self._rects: list[QRect] = []
 51        self._pressed: int | None = None   # the tab pressed (being dragged to another place)
 52        self._press_pos = QPoint()
 53        self._closing: int | None = None   # a close glyph pressed
 54        self._hover: int | None = None     # (the close glyph under the mouse: highlighted)
 55        self.setMouseTracking(True)
 56        self.setSizePolicy(QSizePolicy.Preferred, QSizePolicy.Fixed)
 57        tabs.currentChanged.connect(self.refresh)
 58        self.bar.tabMoved.connect(self.refresh)

 60    # ------------------------------------------------------------ layout
 61    def refresh(self, *_):
 62        """The tabs changed (added, removed, renamed, moved, chosen)."""
 63        self._layout(self.width())
 64        self.updateGeometry()
 65        self.update()
 66
 67    def _layout(self, width: int) -> int:
 68        """Place the tabs in rows ``width`` wide; the height they take."""
 69        rects, x, y, row_height = [], 0, 0, 0
 70        for index in range(self.bar.count()):
 71            size = self.bar.tabSizeHint(index)
 72            if x and x + size.width() > width:  # (no room left on this row: the next)
 73                x, y = 0, y + row_height
 74                row_height = 0
 75            rects.append(QRect(x, y, size.width(), size.height()))
 76            x += size.width()
 77            row_height = max(row_height, size.height())
 78        self._rects = rects
 79        return y + row_height
 80
 81    def hasHeightForWidth(self) -> bool:
 82        return True
 83
 84    def heightForWidth(self, width: int) -> int:
 85        rects = self._rects
 86        height = self._layout(width)
 87        self._rects = rects  # (only asked: the tabs stay where they are)
 88        return height
 89
 90    def sizeHint(self) -> QSize:
 91        return QSize(self.width() or 400, self.heightForWidth(self.width() or 400))
 92
 93    def minimumSizeHint(self) -> QSize:
 94        return QSize(0, self.sizeHint().height())
 95
 96    def resizeEvent(self, event):
 97        super().resizeEvent(event)
 98        height = self._layout(event.size().width())
 99        if height != self.height():
100            self.setFixedHeight(height)  # (more rows, or fewer: the editor below gets the rest)
101
102    def index_at(self, pos: QPoint) -> int:
103        return next((i for i, rect in enumerate(self._rects) if rect.contains(pos)), -1)
104
105    def tab_rect(self, index: int) -> QRect:
106        return QRect(self._rects[index]) if 0 <= index < len(self._rects) else QRect()

108    # ------------------------------------------------------------ drawing
109    def _option(self, index: int) -> QStyleOptionTab:
110        option = QStyleOptionTab()
111        self.bar.initStyleOption(option, index)
112        option.rect = self._rects[index]
113        row = [i for i, r in enumerate(self._rects) if r.top() == option.rect.top()]
114        option.position = (QStyleOptionTab.OnlyOneTab if len(row) == 1 else QStyleOptionTab.Beginning
115                           if index == row[0] else QStyleOptionTab.End if index == row[-1]
116                           else QStyleOptionTab.Middle)
117        option.state |= QStyle.State_Enabled
118        if index == self.tabs.currentIndex():
119            option.state |= QStyle.State_Selected
120        else:
121            option.state &= ~QStyle.State_Selected
122        option.text = self.bar.tabText(index)
123        return option
124
125    def _button_rect(self, element, option: QStyleOptionTab) -> QRect:
126        """Where the style puts a tab's button (``element``: its left or right
127        one), for a tab on any row. The macOS style places them from the top
128        of the bar whatever the tab's own top (one row's always 0): asked for
129        the tab on the first row, then moved down to its own."""
130        top = QStyleOptionTab(option)
131        top.rect = QRect(option.rect)
132        top.rect.moveTop(0)
133        return self.style().subElementRect(element, top, self).translated(0, option.rect.top())
134
135    def _close_rect(self, option: QStyleOptionTab) -> QRect:
136        area = self._button_rect(QStyle.SE_TabBarTabRightButton, option)
137        rect = QRect(0, 0, CLOSE_SIZE + 6, CLOSE_SIZE + 6)
138        rect.moveCenter(area.center())
139        return rect

141    def paintEvent(self, event):
142        if len(self._rects) != self.bar.count():
143            self._layout(self.width())
144        painter = QStylePainter(self)
145        italic = QFont(self.font())
146        italic.setItalic(True)
147        close = close_pixmap(self.devicePixelRatioF())
148        current = self.tabs.currentIndex()
149        for index in [i for i in range(len(self._rects)) if i != current] + ([current] if current >= 0 else []):
150            option = self._option(index)  # (the current tab last: drawn over its neighbours)
151            painter.setClipRect(option.rect)  # (a style may draw the current tab down into the page: not the next row)
152            painter.drawControl(QStyle.CE_TabBarTabShape, option)
153            name = self.bar.tab_name(index) if hasattr(self.bar, "tab_name") else option.text
154            is_italic = hasattr(self.bar, "italic_name") and self.bar.italic_name(index) is not None
155            option.text = name
156            painter.setFont(italic if is_italic else self.font())
157            painter.drawControl(QStyle.CE_TabBarTabLabel, option)
158            icon = self.bar.tabButton(index, QTabBar.LeftSide)
159            if icon is not None and hasattr(icon, "pixmap") and icon.pixmap() is not None:
160                area = self._button_rect(QStyle.SE_TabBarTabLeftButton, option)
161                pixmap = icon.pixmap()
162                # (Its size as drawn, not in device pixels: on a Retina screen a 16-point icon is 32 pixels.)
163                target = QRect(QPoint(0, 0), pixmap.deviceIndependentSize().toSize())
164                target.moveCenter(area.center())
165                painter.drawPixmap(target, pixmap)
166            rect = self._close_rect(option)
167            if index == self._hover:
168                painter.setRenderHint(QPainter.Antialiasing)
169                painter.setPen(Qt.NoPen)
170                painter.setBrush(QColor(128, 128, 128, 80))
171                painter.drawRoundedRect(rect, 4, 4)
172            glyph = QRect(0, 0, CLOSE_SIZE, CLOSE_SIZE)  # (its size as drawn, whatever the screen's pixels)
173            glyph.moveCenter(rect.center())
174            painter.drawPixmap(glyph, close)
175            painter.setClipping(False)

177    # ------------------------------------------------------------ the mouse
178    def _on_close(self, index: int, pos: QPoint) -> bool:
179        return index >= 0 and self._close_rect(self._option(index)).contains(pos)
180
181    def mousePressEvent(self, event):
182        index = self.index_at(event.position().toPoint())
183        if event.button() != Qt.LeftButton or index < 0:
184            super().mousePressEvent(event)
185            return
186        if self._on_close(index, event.position().toPoint()):
187            self._closing = index
188            return
189        self.tabs.setCurrentIndex(index)
190        self._pressed, self._press_pos = index, event.position().toPoint()
191
192    def mouseMoveEvent(self, event):
193        pos = event.position().toPoint()
194        index = self.index_at(pos)
195        hover = index if self._on_close(index, pos) else None
196        if hover != self._hover:
197            self._hover = hover
198            self.update()
199        if self._pressed is None or not event.buttons() & Qt.LeftButton:
200            return
201        if (pos - self._press_pos).manhattanLength() < QApplication.startDragDistance():
202            return
203        if index >= 0 and index != self._pressed:  # (dragged over another tab: it takes that one's place)
204            self.bar.moveTab(self._pressed, index)
205            self._pressed = index
206
207    def mouseReleaseEvent(self, event):
208        closing, self._closing, self._pressed = self._closing, None, None
209        index = self.index_at(event.position().toPoint())
210        if closing is not None and index == closing and self._on_close(index, event.position().toPoint()):
211            self.close_requested.emit(index)
212
213    def mouseDoubleClickEvent(self, event):
214        index = self.index_at(event.position().toPoint())
215        if index >= 0 and not self._on_close(index, event.position().toPoint()):
216            self.tabs.tabBarDoubleClicked.emit(index)
217
218    def contextMenuEvent(self, event):
219        if self.index_at(event.pos()) < 0:
220            self.empty_menu_requested.emit(event.globalPos())
221            event.accept()
222        else:
223            super().contextMenuEvent(event)
224
225    def leaveEvent(self, event):
226        if self._hover is not None:
227            self._hover = None
228            self.update()
229        super().leaveEvent(event)
230
231    def event(self, event):
232        if event.type() == event.Type.ToolTip:
233            pos = event.pos()
234            index = self.index_at(pos)
235            if index < 0:
236                QToolTip.hideText()
237            elif self._on_close(index, pos):
238                button = self.bar.tabButton(index, QTabBar.RightSide)
239                QToolTip.showText(event.globalPos(), button.toolTip() if button is not None else "Close", self)
240            else:
241                QToolTip.showText(event.globalPos(), self.bar.tabToolTip(index), self)
242            return True
243        return super().event(event)
