vibefpmgui.graph_view

Interactive dependency diagram drawn with pyqtgraph.

The diagram lives in a pyqtgraph ViewBox (mouse-wheel zoom, drag to pan, auto-fit, image/SVG export). Edges are PlotCurveItem curves with ArrowItem heads; packages are custom GraphicsObject boxes that can be dragged around.

Several layouts (LAYOUTS), color themes (THEMES) and edge styles (EDGE_STYLES) are available. Clicking a package highlights everything it depends on and everything that depends on it.

 143def _asked_for(package: m.Package) -> str:
 144    """The tag, branch or commit a package not downloaded asks for, or the
 145    kind it would be ("Git"…) if none ("" if it's downloaded: its version
 146    is shown)."""
 147    if package.kind != m.DEV_ONLY:
 148        return ""
 149    if not package.git_ref:
 150        return KIND_LABELS.get(package.asked_kind, "")
 151    key, _, value = package.git_ref.partition(" ")
 152    if key == "tag":
 153        return f"v{value}" if value[:1].isdigit() else value  # ("1.6.1" as the others' "v1.6.1")
 154    return value[:7] if key == "rev" else package.git_ref

 313    def hoverEvent(self, ev):
 314        if ev.isExit():
 315            if self.hovered:
 316                self.hovered = False
 317                self.view._hover_changed(self, False)
 318            return
 319        ev.acceptClicks(Qt.LeftButton)

 321        if not self.hovered:
 322            self.hovered = True
 323            self.view._hover_changed(self, True)

 325    def mouseClickEvent(self, ev):
 326        if ev.button() == Qt.RightButton:  # the package's menu (see MainWindow)
 327            ev.accept()
 328            self.view.select(self.package.name, center=False)
 329            self.view.package_context_menu.emit(self.package.name, ev.screenPos().toPoint())
 330            return
 331        if ev.button() != Qt.LeftButton:
 332            ev.ignore()
 333            return
 334        ev.accept()
 335        # pyqtgraph only flags ev.double() for some event orders, so also
 336        # detect a second click within the system double-click interval.
 337        now = time.monotonic()
 338        interval = QApplication.doubleClickInterval() / 1000
 339        if ev.double() or now - self._last_click < interval:
 340            self._last_click = 0.0
 341            self.view.package_activated.emit(self.package.name)
 342        else:
 343            self._last_click = now
 344            self.view.select(self.package.name, center=False)
 345
 346    def mouseDragEvent(self, ev):
 347        if ev.button() != Qt.LeftButton:
 348            ev.ignore()
 349            return
 350        ev.accept()
 351        if ev.isStart():
 352            self._drag_offset = self.pos() - self.mapToParent(ev.buttonDownPos())
 353            self.view._stop_animation()
 354        self.setPos(self.mapToParent(ev.pos()) + self._drag_offset)
 355        self.view._update_edges(self)

 399def _clip_to_rect(rect: QRectF, toward: QPointF) -> QPointF:
 400    """Point where the ray from the rect centre toward ``toward`` leaves it."""
 401    c = rect.center()
 402    dx, dy = toward.x() - c.x(), toward.y() - c.y()
 403    if abs(dx) < 1e-9 and abs(dy) < 1e-9:
 404        return c
 405    hw, hh = rect.width() / 2 + 1, rect.height() / 2 + 1
 406    t = min(hw / abs(dx) if dx else math.inf, hh / abs(dy) if dy else math.inf)
 407    return QPointF(c.x() + dx * t, c.y() + dy * t)

 410def _bezier(p0, c1, c2, p3, n=28) -> np.ndarray:
 411    t = np.linspace(0, 1, n)[:, None]
 412    pts = np.array([[p.x(), p.y()] for p in (p0, c1, c2, p3)])
 413    return ((1 - t) ** 3 * pts[0] + 3 * (1 - t) ** 2 * t * pts[1]
 414            + 3 * (1 - t) * t ** 2 * pts[2] + t ** 3 * pts[3])

 417def edge_geometry(s: QRectF, t: QRectF, layout: str, curved: bool):
 418    """Return (Nx2 points, end direction) for an arrow from rect s to rect t."""
 419    sc, tc = s.center(), t.center()
 420    if curved and layout == LAYERED_TB and t.top() > s.bottom() + 4:
 421        p0, p3 = QPointF(sc.x(), s.bottom()), QPointF(tc.x(), t.top())
 422        dy = (p3.y() - p0.y()) / 2
 423        c1, c2 = QPointF(p0.x(), p0.y() + dy), QPointF(p3.x(), p3.y() - dy)
 424    elif curved and layout == LAYERED_LR and t.left() > s.right() + 4:
 425        p0, p3 = QPointF(s.right(), sc.y()), QPointF(t.left(), tc.y())
 426        dx = (p3.x() - p0.x()) / 2
 427        c1, c2 = QPointF(p0.x() + dx, p0.y()), QPointF(p3.x() - dx, p3.y())
 428    elif curved:
 429        # A gentle arc bowing to one side of the straight line.
 430        v = tc - sc
 431        ctrl = (sc + tc) / 2 + QPointF(-v.y(), v.x()) * 0.18
 432        p0, p3 = _clip_to_rect(s, ctrl), _clip_to_rect(t, ctrl)
 433        c1, c2 = p0 + (ctrl - p0) * (2 / 3), p3 + (ctrl - p3) * (2 / 3)
 434    else:
 435        p0, p3 = _clip_to_rect(s, tc), _clip_to_rect(t, sc)
 436        pts = np.array([[p0.x(), p0.y()], [p3.x(), p3.y()]])
 437        return pts, (p3.x() - p0.x(), p3.y() - p0.y())
 438    pts = _bezier(p0, c1, c2, p3)
 439    d = pts[-1] - pts[-3]
 440    return pts, (d[0], d[1])

 444def _adjacency(graph: m.DependencyGraph):
 445    children = {name: [] for name in graph.packages}
 446    parents = {name: [] for name in graph.packages}
 447    for e in graph.edges:
 448        if e.child not in children[e.parent]:
 449            children[e.parent].append(e.child)
 450            parents[e.child].append(e.parent)
 451    return children, parents

 454def compute_layers(graph: m.DependencyGraph) -> list[list[str]]:
 455    """Assign each package to a layer (longest path from the root) and order
 456    the layers to reduce crossings."""
 457    children, parents = _adjacency(graph)
 458    depth = {graph.root: 0}
 459    # Longest-path layering; the iteration cap guards against cycles.
 460    for _ in range(len(graph.packages)):
 461        changed = False
 462        for e in graph.edges:
 463            new_depth = depth[e.parent] + 1 if e.parent in depth else None
 464            if new_depth is not None and depth.get(e.child, -1) < new_depth < len(graph.packages):
 465                depth[e.child] = new_depth
 466                changed = True
 467        if not changed:
 468            break
 469    for name in graph.packages:
 470        depth.setdefault(name, 1)
 471
 472    layers: list[list[str]] = [[] for _ in range(max(depth.values()) + 1)]
 473    for name in graph.packages:  # BFS order gives a sensible starting order
 474        layers[depth[name]].append(name)
 475
 476    def barycenter(name, neighbours, index):
 477        pos = [index[n] for n in neighbours[name] if n in index]
 478        return sum(pos) / len(pos) if pos else index.get(name, 0)
 479
 480    for _ in range(4):
 481        for i in range(1, len(layers)):  # downward sweep
 482            index = {n: k for k, n in enumerate(layers[i - 1])}
 483            layers[i].sort(key=lambda n: barycenter(n, parents, index))
 484        for i in range(len(layers) - 2, 0, -1):  # upward sweep
 485            index = {n: k for k, n in enumerate(layers[i + 1])}
 486            layers[i].sort(key=lambda n: barycenter(n, children, index))
 487    return [layer for layer in layers if layer]
 488

 490def layout_layered(graph, sizes, horizontal=False) -> dict[str, QPointF]:
 491    layers = compute_layers(graph)
 492    pos = {}
 493    if not horizontal:
 494        y = 0.0
 495        for layer in layers:
 496            width = sum(sizes[n][0] for n in layer) + H_GAP * (len(layer) - 1)
 497            height = max(sizes[n][1] for n in layer)
 498            x = -width / 2
 499            for name in layer:
 500                pos[name] = QPointF(x + sizes[name][0] / 2, y + height / 2)
 501                x += sizes[name][0] + H_GAP
 502            y += height + V_GAP
 503    else:
 504        gap = 22
 505        x = 0.0
 506        for layer in layers:
 507            col_w = max(sizes[n][0] for n in layer)
 508            total = sum(sizes[n][1] for n in layer) + gap * (len(layer) - 1)
 509            y = -total / 2
 510            for name in layer:
 511                pos[name] = QPointF(x + col_w / 2, y + sizes[name][1] / 2)
 512                y += sizes[name][1] + gap
 513            x += col_w + V_GAP * 1.3
 514    return pos

 517def layout_radial(graph, sizes) -> dict[str, QPointF]:
 518    """Root in the middle, a breadth-first spanning tree fanned out in rings."""
 519    children, _ = _adjacency(graph)
 520    rank = {n: i for i, n in enumerate(n for layer in compute_layers(graph) for n in layer)}
 521    tree = {n: [] for n in graph.packages}
 522    depth = {graph.root: 0}
 523    queue = [graph.root]
 524    while queue:
 525        node = queue.pop(0)
 526        for child in sorted(children[node], key=rank.get):
 527            if child not in depth:
 528                depth[child] = depth[node] + 1
 529                tree[node].append(child)
 530                queue.append(child)
 531    for n in graph.packages:  # anything unreachable hangs off the root
 532        if n not in depth:
 533            depth[n] = 1
 534            tree[graph.root].append(n)
 535
 536    leaves: dict[str, int] = {}
 537
 538    def count(n):
 539        leaves[n] = sum(count(c) for c in tree[n]) or 1
 540        return leaves[n]
 541
 542    angle: dict[str, float] = {}
 543
 544    def assign(n, a0, a1):
 545        angle[n] = (a0 + a1) / 2
 546        a = a0
 547        for c in tree[n]:
 548            span = (a1 - a0) * leaves[c] / leaves[n]
 549            assign(c, a, a + span)
 550            a += span
 551
 552    count(graph.root)
 553    assign(graph.root, -math.pi / 2, 3 * math.pi / 2)
 554
 555    # Every leaf gets an equal slice of the circle, so neighbours on a ring
 556    # are at least one slice apart; make each ring big enough for that.
 557    radius = [0.0]
 558    slice_angle = 2 * math.pi / leaves[graph.root]
 559    step = max(max(h for _, h in sizes.values()) * 3, 150)
 560    for d in range(1, max(depth.values()) + 1):
 561        widest = max(sizes[n][0] for n in graph.packages if depth[n] == d)
 562        radius.append(max(radius[-1] + step, (widest + H_GAP) / slice_angle))
 563    pos = {n: QPointF(radius[depth[n]] * math.cos(angle[n]),
 564                      radius[depth[n]] * math.sin(angle[n])) for n in graph.packages}
 565    return _remove_overlaps(pos, sizes)
 566

 568def layout_circular(graph, sizes) -> dict[str, QPointF]:
 569    order = [n for layer in compute_layers(graph) for n in layer]
 570    r = max(sum(sizes[n][0] + H_GAP for n in order) / (2 * math.pi), 150)
 571    step = 2 * math.pi / max(len(order), 1)
 572    return {n: QPointF(r * math.cos(-math.pi / 2 + i * step), r * math.sin(-math.pi / 2 + i * step))
 573            for i, n in enumerate(order)}

 576def layout_force(graph, sizes) -> dict[str, QPointF]:
 577    """Fruchterman-Reingold, seeded from the layered layout (deterministic)."""
 578    names = list(graph.packages)
 579    start = layout_layered(graph, sizes)
 580    n = len(names)
 581    if n < 2:
 582        return start
 583    pos = np.array([[start[k].x(), start[k].y()] for k in names], dtype=float)
 584    index = {k: i for i, k in enumerate(names)}
 585    edges = np.array([(index[e.parent], index[e.child]) for e in graph.edges], dtype=int).reshape(-1, 2)
 586    k = 190.0
 587    temperature = 120.0
 588    for _ in range(400):
 589        delta = pos[:, None, :] - pos[None, :, :]
 590        dist2 = (delta ** 2).sum(-1) + np.eye(n)
 591        disp = (delta * (k * k / dist2)[:, :, None]).sum(1)  # repulsion
 592        if len(edges):  # attraction along edges
 593            d = pos[edges[:, 1]] - pos[edges[:, 0]]
 594            pull = d * (np.sqrt((d ** 2).sum(-1))[:, None] / k)
 595            np.add.at(disp, edges[:, 0], pull)
 596            np.add.at(disp, edges[:, 1], -pull)
 597        disp -= (pos - pos.mean(0)) * 0.01  # gentle gravity
 598        length = np.sqrt((disp ** 2).sum(-1))[:, None] + 1e-9
 599        pos += disp / length * np.minimum(length, temperature)
 600        temperature = max(1.0, temperature * 0.985)
 601    result = {name: QPointF(*pos[i]) for i, name in enumerate(names)}
 602    return _remove_overlaps(result, sizes)

 605def _remove_overlaps(pos: dict[str, QPointF], sizes, margin=14, iterations=80):
 606    """Push apart any boxes that overlap, along the axis of least overlap."""
 607    names = list(pos)
 608    p = {k: [pos[k].x(), pos[k].y()] for k in names}
 609    for _ in range(iterations):
 610        moved = False
 611        for i, a in enumerate(names):
 612            for b in names[i + 1:]:
 613                ox = (sizes[a][0] + sizes[b][0]) / 2 + margin - abs(p[a][0] - p[b][0])
 614                oy = (sizes[a][1] + sizes[b][1]) / 2 + margin - abs(p[a][1] - p[b][1])
 615                if ox > 0 and oy > 0:
 616                    moved = True
 617                    axis, amount = (0, ox) if ox < oy else (1, oy)
 618                    s = 1 if p[a][axis] >= p[b][axis] else -1
 619                    p[a][axis] += s * amount / 2
 620                    p[b][axis] -= s * amount / 2
 621        if not moved:
 622            break
 623    return {k: QPointF(*v) for k, v in p.items()}

 960    def _reachable(self, name: str | None, forward: bool) -> set[str]:
 961        if name is None or not self.graph:
 962            return set()
 963        nxt: dict[str, set[str]] = {}
 964        for e in self.graph.edges:
 965            a, b = (e.parent, e.child) if forward else (e.child, e.parent)
 966            nxt.setdefault(a, set()).add(b)
 967        seen, stack = set(), [name]
 968        while stack:
 969            for n in nxt.get(stack.pop(), ()):
 970                if n not in seen:
 971                    seen.add(n)
 972                    stack.append(n)
 973        seen.discard(name)
 974        return seen

455def compute_layers(graph: m.DependencyGraph) -> list[list[str]]:
456    """Assign each package to a layer (longest path from the root) and order
457    the layers to reduce crossings."""
458    children, parents = _adjacency(graph)
459    depth = {graph.root: 0}
460    # Longest-path layering; the iteration cap guards against cycles.
461    for _ in range(len(graph.packages)):
462        changed = False
463        for e in graph.edges:
464            new_depth = depth[e.parent] + 1 if e.parent in depth else None
465            if new_depth is not None and depth.get(e.child, -1) < new_depth < len(graph.packages):
466                depth[e.child] = new_depth
467                changed = True
468        if not changed:
469            break
470    for name in graph.packages:
471        depth.setdefault(name, 1)
472
473    layers: list[list[str]] = [[] for _ in range(max(depth.values()) + 1)]
474    for name in graph.packages:  # BFS order gives a sensible starting order
475        layers[depth[name]].append(name)
476
477    def barycenter(name, neighbours, index):
478        pos = [index[n] for n in neighbours[name] if n in index]
479        return sum(pos) / len(pos) if pos else index.get(name, 0)
480
481    for _ in range(4):
482        for i in range(1, len(layers)):  # downward sweep
483            index = {n: k for k, n in enumerate(layers[i - 1])}
484            layers[i].sort(key=lambda n: barycenter(n, parents, index))
485        for i in range(len(layers) - 2, 0, -1):  # upward sweep
486            index = {n: k for k, n in enumerate(layers[i + 1])}
487            layers[i].sort(key=lambda n: barycenter(n, children, index))
488    return [layer for layer in layers if layer]

518def layout_radial(graph, sizes) -> dict[str, QPointF]:
519    """Root in the middle, a breadth-first spanning tree fanned out in rings."""
520    children, _ = _adjacency(graph)
521    rank = {n: i for i, n in enumerate(n for layer in compute_layers(graph) for n in layer)}
522    tree = {n: [] for n in graph.packages}
523    depth = {graph.root: 0}
524    queue = [graph.root]
525    while queue:
526        node = queue.pop(0)
527        for child in sorted(children[node], key=rank.get):
528            if child not in depth:
529                depth[child] = depth[node] + 1
530                tree[node].append(child)
531                queue.append(child)
532    for n in graph.packages:  # anything unreachable hangs off the root
533        if n not in depth:
534            depth[n] = 1
535            tree[graph.root].append(n)
536
537    leaves: dict[str, int] = {}
538
539    def count(n):
540        leaves[n] = sum(count(c) for c in tree[n]) or 1
541        return leaves[n]
542
543    angle: dict[str, float] = {}
544
545    def assign(n, a0, a1):
546        angle[n] = (a0 + a1) / 2
547        a = a0
548        for c in tree[n]:
549            span = (a1 - a0) * leaves[c] / leaves[n]
550            assign(c, a, a + span)
551            a += span
552
553    count(graph.root)
554    assign(graph.root, -math.pi / 2, 3 * math.pi / 2)
555
556    # Every leaf gets an equal slice of the circle, so neighbours on a ring
557    # are at least one slice apart; make each ring big enough for that.
558    radius = [0.0]
559    slice_angle = 2 * math.pi / leaves[graph.root]
560    step = max(max(h for _, h in sizes.values()) * 3, 150)
561    for d in range(1, max(depth.values()) + 1):
562        widest = max(sizes[n][0] for n in graph.packages if depth[n] == d)
563        radius.append(max(radius[-1] + step, (widest + H_GAP) / slice_angle))
564    pos = {n: QPointF(radius[depth[n]] * math.cos(angle[n]),
565                      radius[depth[n]] * math.sin(angle[n])) for n in graph.packages}
566    return _remove_overlaps(pos, sizes)

577def layout_force(graph, sizes) -> dict[str, QPointF]:
578    """Fruchterman-Reingold, seeded from the layered layout (deterministic)."""
579    names = list(graph.packages)
580    start = layout_layered(graph, sizes)
581    n = len(names)
582    if n < 2:
583        return start
584    pos = np.array([[start[k].x(), start[k].y()] for k in names], dtype=float)
585    index = {k: i for i, k in enumerate(names)}
586    edges = np.array([(index[e.parent], index[e.child]) for e in graph.edges], dtype=int).reshape(-1, 2)
587    k = 190.0
588    temperature = 120.0
589    for _ in range(400):
590        delta = pos[:, None, :] - pos[None, :, :]
591        dist2 = (delta ** 2).sum(-1) + np.eye(n)
592        disp = (delta * (k * k / dist2)[:, :, None]).sum(1)  # repulsion
593        if len(edges):  # attraction along edges
594            d = pos[edges[:, 1]] - pos[edges[:, 0]]
595            pull = d * (np.sqrt((d ** 2).sum(-1))[:, None] / k)
596            np.add.at(disp, edges[:, 0], pull)
597            np.add.at(disp, edges[:, 1], -pull)
598        disp -= (pos - pos.mean(0)) * 0.01  # gentle gravity
599        length = np.sqrt((disp ** 2).sum(-1))[:, None] + 1e-9
600        pos += disp / length * np.minimum(length, temperature)
601        temperature = max(1.0, temperature * 0.985)
602    result = {name: QPointF(*pos[i]) for i, name in enumerate(names)}
603    return _remove_overlaps(result, sizes)
