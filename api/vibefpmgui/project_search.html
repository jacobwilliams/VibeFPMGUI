vibefpmgui.project_search

Find in Files (⌘/Ctrl+Shift+F): search, and replace, across the project and its dependencies.

The search runs on a worker thread over every text file of the project and of the dependencies on disk. Files open in the editor are searched as they are in the editor, unsaved changes included. Like the Project files tree it leaves out fpm's build folder, .git and compiled files; files ignored by git and hidden files only with Ignored files checked.

Results are grouped by file in the Search tab, each line in the colors the editor would give it; click one to open the file there. Replacing changes open files in the editor (not saved: Undo reverts it) and saves the others, like Rename Symbol.

 83def compile_query(query: Query) -> re.Pattern | None:
 84    """The query as a regular expression (None if there's nothing to find).
 85    Raises `re.error` for a bad regular expression."""
 86    if not query.text:
 87        return None
 88    pattern = query.text if query.regex else re.escape(query.text)
 89    if query.word:
 90        pattern = rf"(?<!\w)(?:{pattern})(?!\w)"
 91    return re.compile(pattern, re.MULTILINE | (0 if query.case else re.IGNORECASE))

 94def search_roots(graph: m.DependencyGraph | None, project_dir: Path) -> list[Root]:
 95    """The project folder, then each dependency on disk."""
 96    name = project_dir.name
 97    if graph is not None and graph.root in graph.packages:
 98        name = graph.packages[graph.root].name
 99    roots, seen = [Root(name, project_dir, False)], {project_dir.resolve()}
100    if graph is not None:
101        for pkg in graph.packages.values():
102            if pkg.name == graph.root or pkg.kind in (m.META, m.MISSING):
103                continue
104            if pkg.directory and pkg.directory.is_dir() and pkg.directory.resolve() not in seen:
105                seen.add(pkg.directory.resolve())
106                roots.append(Root(pkg.name, pkg.directory, True))
107    return roots

114def glob_matches(relative: str, globs: list[str]) -> bool:
115    """Whether a file (its path in its package, "/"-separated) matches one
116    of the globs: "*.f90" (a name anywhere), "src/" or "src" (a folder and
117    everything in it), "src/*.f90" (a path)."""
118    relative = relative.lower()  # (*.f90 matches .F90 files too)
119    name = relative.rsplit("/", 1)[-1]
120    for glob in globs:
121        glob = glob.lower().removeprefix("./")
122        folder = glob.rstrip("/")
123        if fnmatch.fnmatch(name, glob) or fnmatch.fnmatch(relative, glob) \
124                or fnmatch.fnmatch(relative, f"*/{glob}") \
125                or fnmatch.fnmatch(relative, f"{folder}/*") or fnmatch.fnmatch(relative, f"*/{folder}/*"):
126            return True
127    return False

130def iter_files(roots: list[Root], query: Query, cancelled: Callable[[], bool] = lambda: False):
131    """(root, path) for every file to search, package by package (the
132    dependencies only with ``query.dependencies``; their folders are left out
133    of the project's even when not searched, e.g. a local dependency in it)."""
134    others = {r.directory.resolve() for r in roots}
135    include, exclude = _globs(query.include), _globs(query.exclude)
136    for root in roots:
137        if root.dependency and not query.dependencies:
138            continue
139        top = root.directory.resolve()
140        rules = GitIgnore(top)
141        for folder, dirs, files in os.walk(top):
142            if cancelled():
143                return
144            here = Path(folder)  # (resolved: the walk starts at a resolved folder, and doesn't follow links)
145            keep = []
146            for name in sorted(dirs, key=str.lower):
147                path = here / name
148                if name == ".git" or name in SKIP_DIRS or path in others:
149                    continue  # (other packages are searched as themselves)
150                if not query.ignored and (name.startswith(".") or rules.check_name(here, name, True)):
151                    continue
152                keep.append(name)
153            dirs[:] = keep
154            prefix = here.relative_to(top).as_posix()
155            prefix = "" if prefix == "." else prefix + "/"
156            for name in sorted(files, key=str.lower):
157                suffix = os.path.splitext(name)[1]
158                if suffix in SKIP_SUFFIXES or suffix.lower() in SKIP_SUFFIXES:
159                    continue
160                if not query.ignored and (name.startswith(".") or rules.check_name(here, name, False)):
161                    continue
162                path = here / name
163                relative = prefix + name
164                if include and not glob_matches(relative, include):
165                    continue
166                if exclude and glob_matches(relative, exclude):
167                    continue
168                yield root, path

171def is_searched(root: Root, path: Path, query: Query, others: set[Path] = frozenset()) -> bool:
172    """Whether `iter_files` would search a file (checked on its own)."""
173    top = root.directory.resolve()
174    try:
175        parts = path.relative_to(top).parts
176    except ValueError:
177        return False
178    rules = GitIgnore(top)
179    folder = top
180    for name in parts[:-1]:
181        folder = folder / name
182        if name == ".git" or name in SKIP_DIRS or folder in others:
183            return False
184        if not query.ignored and (name.startswith(".") or rules.check(folder, True)):
185            return False
186    name = parts[-1]
187    if path.suffix in SKIP_SUFFIXES or path.suffix.lower() in SKIP_SUFFIXES:
188        return False
189    if not query.ignored and (name.startswith(".") or rules.check(path, False)):
190        return False
191    relative = "/".join(parts)
192    include, exclude = _globs(query.include), _globs(query.exclude)
193    return (not include or glob_matches(relative, include)) and not (exclude and glob_matches(relative, exclude))

196def find_matches(text: str, pattern: re.Pattern, limit: int = MAX_RESULTS) -> list[Match]:
197    """Every match in a text (empty matches are skipped)."""
198    out: list[Match] = []
199    line, last = 0, 0  # (counted incrementally: big files stay quick)
200    for found in pattern.finditer(text):
201        start, end = found.span()
202        if start == end:
203            continue
204        line += text.count("\n", last, start)
205        last = start
206        line_start = text.rfind("\n", 0, start) + 1
207        line_end = text.find("\n", start)
208        line_end = len(text) if line_end < 0 else line_end
209        end_line = line + text.count("\n", start, end)
210        end_column = end - (text.rfind("\n", 0, end) + 1)
211        out.append(make_match(text[line_start:line_end], line + 1, start - line_start,
212                              end_line + 1, end_column))
213        if len(out) >= limit:
214            break
215    return out

218def replacement(found: re.Match, query: Query, template: str) -> str:
219    """What one match is replaced with: with *regex*, ``\\1`` and
220    ``\\g<name>`` in the template are the match's groups."""
221    return found.expand(template) if query.regex else template

224def replacement_edits(text: str, pattern: re.Pattern, query: Query, template: str,
225                      keep: set[tuple[int, int]] | None = None) -> list[TextEdit]:
226    """The edits that replace the matches in a text: all of them, or those
227    starting at the (1-based line, column) positions in ``keep``.
228    Raises `re.error` for a bad template."""
229    edits = []
230    line, line_start, last = 0, 0, 0  # (counted as it goes: recounting from the top was quadratic)
231    for found in pattern.finditer(text):
232        start, end = found.span()
233        if start == end:
234            continue
235        newlines = text.count("\n", last, start)
236        if newlines:
237            line += newlines
238            line_start = text.rfind("\n", 0, start) + 1
239        last = start
240        column = start - line_start
241        if keep is not None and (line + 1, column) not in keep:
242            continue
243        end_line = line + text.count("\n", start, end)
244        end_column = end - (text.rfind("\n", 0, end) + 1)
245        edits.append(TextEdit(line, column, end_line, end_column, replacement(found, query, template)))
246    return edits

249def preview_replacement(match_text: str, pattern: re.Pattern, query: Query, template: str) -> str:
250    """The replacement shown in the results list (the match on its own:
251    good enough for a preview)."""
252    if not query.regex:
253        return template
254    try:
255        return pattern.sub(lambda f: f.expand(template), match_text, count=1)
256    except (re.error, IndexError):
257        return template

260def search(roots: list[Root], query: Query, pattern: re.Pattern, open_texts: dict[Path, str],
261           found: Callable[[FileResult], None], cancelled: Callable[[], bool] = lambda: False,
262           limit: int = MAX_RESULTS) -> tuple[int, bool]:
263    """Search the files; ``found`` gets each file with matches.
264    Returns (files searched, whether it stopped at ``limit`` matches)."""
265    searched, total = 0, 0
266    for root, path in iter_files(roots, query, cancelled):
267        if cancelled():
268            break
269        resolved = path.resolve()
270        text = open_texts.get(resolved)
271        if text is None:
272            text = search_text(path)
273        if text is None:
274            continue
275        searched += 1
276        matches = find_matches(text, pattern, limit - total)
277        if matches:
278            total += len(matches)
279            found(FileResult(resolved, root, matches))
280            if total >= limit:
281                return searched, True
282    return searched, False

285class SearchWorker(QThread):
286    """Runs `search` on a thread; results arrive in batches."""
287    found = Signal(object)          # list[FileResult]
288    progress = Signal(int)          # files searched so far
289    done = Signal(int, bool)        # files searched, stopped at the limit
290
291    def __init__(self, roots, query, pattern, open_texts, parent=None):
292        super().__init__(parent)
293        self.roots, self.query, self.pattern, self.open_texts = roots, query, pattern, open_texts
294        self._cancel = False
295
296    def cancel(self):
297        self._cancel = True
298
299    def run(self):
300        batch: list[FileResult] = []
301        last = time.monotonic()
302
303        def found(result):
304            nonlocal last
305            batch.append(result)
306            if time.monotonic() - last > 0.15:
307                self.found.emit(batch[:])
308                batch.clear()
309                last = time.monotonic()
310
311        searched, truncated = search(self.roots, self.query, self.pattern, self.open_texts, found,
312                                     lambda: self._cancel)
313        if not self._cancel:
314            if batch:
315                self.found.emit(batch)
316            self.done.emit(searched, truncated)
