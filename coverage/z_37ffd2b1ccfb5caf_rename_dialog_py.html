40def renamed_spans(edits: list[lsp_mod.TextEdit], line: int) -> list[tuple[int, int]]: 

41 """Where a line's edits put their new text, in the line as it will read 

42 (each edit shifts those after it on the line).""" 

43 spans, shift = [], 0 

44 for edit in sorted((e for e in edits if e.line == line == e.end_line), key=lambda e: e.column): 

45 start = edit.column + shift 

46 spans.append((start, start + len(edit.new_text))) 

47 shift += len(edit.new_text) - (edit.end_column - edit.column) 

48 return spans

68 style = opt.widget.style() if opt.widget else QApplication.style() 

69 style.drawControl(QStyle.CE_ItemViewItem, opt, painter, opt.widget) 

70 selected = bool(opt.state & QStyle.State_Selected) 

71 group = QPalette.Active if opt.state & QStyle.State_Active else QPalette.Inactive 

72 plain = opt.palette.color(group, QPalette.HighlightedText if selected else QPalette.Text)

75 mono = self.dialog.mono

78 column = len(full) - len(code)

81 number = f"{line + 1:>5} " 

82 painter.setFont(mono) 

83 painter.setPen(dim)

85 painter.drawText(QRect(opt.rect.left() + 4, opt.rect.top(), width, opt.rect.height()), 

86 Qt.AlignVCenter | Qt.AlignLeft, number)

88 metrics = QFontMetrics(mono)

93 opt.rect.height() - 4), MATCH_COLOR)

96 formats, column, mono, plain)

100def is_dependency(path: Path, project_dir: Path | None) -> bool: 

101 if project_dir is None: 101 ↛ 102line 101 didn't jump to line 102 because the condition on line 101 was never true

102 return False
