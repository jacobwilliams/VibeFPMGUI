54 def _lint_span(self, issue: lint.LintIssue) -> tuple[int, int] | None: 

55 """Document positions to underline: the issue's text, only on its 

56 first line (a syntax error can span the rest of the file); at least 

57 one character so it can be seen and hovered.""" 

58 block = self.document().findBlockByNumber(issue.line - 1) 

59 if not block.isValid(): 59 ↛ 60line 59 didn't jump to line 60 because the condition on line 59 was never true

60 return None 

61 length = max(0, block.length() - 1) 

62 start = min(max(issue.column - 1, 0), length) 

63 end = min(issue.end_column - 1, length) if issue.end_line == issue.line else length 

64 if end <= start: 

65 start, end = (start, start + 1) if start < length else (max(0, length - 1), length) 

66 if end <= start: 66 ↛ 67line 66 didn't jump to line 67 because the condition on line 66 was never true

67 return None # an empty line

155 def squiggle_spans(self, area: QRect) -> list[tuple[float, float, float, str]]: 

156 """Where the misspelled words' squiggles go in ``area`` (viewport 

157 coordinates): (left, right, y, color) per row a word is on.""" 

158 issues = self._spelling 

159 if not issues: 

160 return [] 

161 dark = self.palette().color(QPalette.Base).lightness() < 128 

162 color = (DARK_UNDERLINE_COLORS if dark else MARKER_COLORS)["spelling"] 

163 first = self.firstVisibleBlock() 

164 top_line = first.blockNumber() + 1 

165 offset = self.contentOffset() 

166 out = [] 

167 # (Only the words in view: from the first visible line, until one is below the area.) 

168 for issue in issues[bisect_left(self._spelling_lines, top_line):]: 

169 block = self.document().findBlockByNumber(issue.line - 1) 

170 geometry = self.blockBoundingGeometry(block).translated(offset) 

171 if geometry.top() > area.bottom(): 

172 break 

173 if geometry.bottom() < area.top() or not block.isVisible(): 

174 continue 

175 span = self._lint_span(issue) 

176 layout = block.layout() 

177 if span is None or layout is None: 177 ↛ 178line 177 didn't jump to line 178 because the condition on line 177 was never true

178 continue 

179 start, end = span[0] - block.position(), span[1] - block.position() 

180 for n in range(layout.lineCount()): 

181 row = layout.lineAt(n) 

182 a, b = max(start, row.textStart()), min(end, row.textStart() + row.textLength()) 

183 if a >= b: 183 ↛ 184line 183 didn't jump to line 184 because the condition on line 183 was never true

184 continue 

185 x1, x2 = row.cursorToX(a), row.cursorToX(b) 

186 x1 = x1[0] if isinstance(x1, tuple) else x1 # (PySide: (x, position)) 

187 x2 = x2[0] if isinstance(x2, tuple) else x2 

188 y = geometry.top() + row.y() + row.ascent() + max(1.5, row.descent() * 0.45) 

189 out.append((geometry.left() + x1, geometry.left() + x2, y, color)) 

190 return out

192 def _paint_squiggles(self, area: QRect): 

193 """Misspelled words' wavy underline, drawn here (thicker than the one 

194 Qt draws for the other issues, which is easy to miss under a 

195 comment's thin text, especially on a dark background).""" 

196 spans = self.squiggle_spans(area) 

197 if not spans: 

198 return 

199 painter = QPainter(self.viewport()) 

200 painter.setRenderHint(QPainter.Antialiasing) 

201 size = self.font().pointSizeF() if self.font().pointSizeF() > 0 else 12.0 

202 width = max(1.4, size / 8) # pen width 

203 height = max(1.6, size / 7) # half the wave's height 

204 step = max(2.5, size / 4.5) # half a wave's length 

205 for left, right, y, color in spans: 

206 path = QPainterPath(QPointF(left, y)) 

207 x, up = left, True 

208 while x < right: 

209 nxt = min(x + step, right) 

210 path.quadTo(QPointF((x + nxt) / 2, y - height if up else y + height), QPointF(nxt, y)) 

211 x, up = nxt, not up 

212 pen = QPen(QColor(color), width) 

213 pen.setCapStyle(Qt.RoundCap) 

214 painter.setPen(pen) 

215 painter.drawPath(path) 

216 painter.end()
