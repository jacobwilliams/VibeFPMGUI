1"""AI code review with a local model served by Ollama (https://ollama.com). 

2 

3Everything runs on this machine: the GUI talks to an ``ollama serve`` 

4process on localhost (starting one if needed), so no code leaves the 

5computer. Models are downloaded once with ``ollama pull`` (the GUI can do 

6that too) and are then available offline. 

7 

8The model is asked to answer with one finding per line:: 

9 

10 LINE 12: bug: the loop never updates i 

11 LINE 30-34: style: use an assumed-shape array instead of (*) 

12 

13followed by ``SUMMARY: ...``. `parse_findings` reads those lines back 

14(tolerating the small variations models produce), so findings can be listed 

15and marked in the editor's gutter. 

16 

17The work happens in a `ReviewWorker` on a background thread; the 

18``ollama`` Python package is optional and imported lazily. 

19"""

95def parse_findings(text: str, max_line: int | None = None) -> tuple[list[Finding], str]: 

96 """Findings and the summary from a model's answer.""" 

97 findings: list[Finding] = [] 

98 summary_lines: list[str] = [] 

99 in_summary = False 

100 for raw in text.splitlines(): 

101 if in_summary: 101 ↛ 102line 101 didn't jump to line 102 because the condition on line 101 was never true

102 if raw.strip(): 

103 summary_lines.append(raw.strip()) 

104 continue 

105 match = SUMMARY_RE.match(raw) 

106 if match: 

107 in_summary = True 

108 if match.group(1).strip(): 108 ↛ 110line 108 didn't jump to line 110 because the condition on line 108 was always true

109 summary_lines.append(match.group(1).strip()) 

110 continue 

111 match = FINDING_RE.match(raw) 

112 if not match: 

113 # Models often explain a finding on the following line(s). 

114 extra = raw.strip().lstrip("-*\u2022 ").strip() 

115 if extra and findings and not extra.startswith("```"): 

116 last = findings[-1] 

117 joiner = " " if last.comment.rstrip("\x00").endswith((".", "!", "?", ":")) else " — " 

118 last.comment = f"{last.comment}{joiner}{extra}" 

119 elif not extra and findings: 

120 findings[-1].comment += "\x00" # a blank line ends the explanation 

121 continue 

122 start = int(match.group(1)) 

123 end = int(match.group(2)) if match.group(2) else start 

124 if end < start: 124 ↛ 125line 124 didn't jump to line 125 because the condition on line 124 was never true

125 start, end = end, start 

126 if max_line is not None and (start < 1 or start > max_line): 

127 continue # a hallucinated line number 

128 category = (match.group(3) or "suggestion").lower() 

129 category = {"error": "bug", "info": "suggestion", "note": "suggestion"}.get(category, category) 

130 comment = match.group(4).strip().strip("*").strip() 

131 if comment: 131 ↛ 100line 131 didn't jump to line 100 because the condition on line 131 was always true

132 findings.append(Finding(start, min(end, max_line or end), category, comment)) 

133 unique: list[Finding] = [] 

134 seen = set() 

135 for f in findings: 

136 # Text after a blank line wasn't part of the finding. 

137 f.comment = f.comment.split("\x00")[0].strip() 

138 key = (f.line, f.end_line, f.category, " ".join(f.comment.lower().split())) 

139 if key not in seen: # small models sometimes repeat themselves 

140 seen.add(key) 

141 unique.append(f) 

142 return unique, " ".join(summary_lines)

145class RepeatGuard: 

146 """Spots a model stuck in a loop (writing a line it already wrote). 

147 

148 Text is passed on a line at a time, so a repeated line can be held back: 

149 `feed` returns the text that's safe to show and whether to stop. 

150 

151 Lines in code (``` blocks) and in edits (SEARCH/REPLACE blocks) aren't 

152 looked at: there, repeating is normal (an edit's new text repeats the 

153 unchanged lines of the old one). A loop there still ends at the answer's 

154 length limit. 

155 """ 

156 

157 MIN_LENGTH = 12 # short lines ("", "```", "SUMMARY:") may legitimately repeat 

158 

159 def __init__(self): 

160 self._partial = "" 

161 self._seen: set[str] = set() 

162 self.repeated = False 

163 self._in_code = False # in a ``` block 

164 self._in_edit = False # in a SEARCH/REPLACE block 

165 

166 @staticmethod 

167 def _key(line: str) -> str: 

168 return " ".join(line.lower().split()) 

169 

170 def feed(self, text: str) -> tuple[str, bool]: 

171 self._partial += text 

172 out = [] 

173 while "\n" in self._partial: 

174 line, self._partial = self._partial.split("\n", 1) 

175 key = self._key(line) 

176 if EDIT_START_RE.match(line): 

177 self._in_edit = True 

178 elif EDIT_END_RE.match(line): 

179 self._in_edit = False 

180 elif line.lstrip().startswith("```"): 

181 self._in_code = not self._in_code 

182 elif len(key) >= self.MIN_LENGTH and not (self._in_code or self._in_edit): 

183 if key in self._seen: 

184 self.repeated = True 

185 self._partial = "" 

186 return "".join(out), True 

187 self._seen.add(key) 

188 out.append(line + "\n") 

189 return "".join(out), False 

190 

191 def flush(self) -> str: 

192 """Whatever is left at the end (a last line without a newline).""" 

193 rest, self._partial = self._partial, "" 

194 if rest and len(self._key(rest)) >= self.MIN_LENGTH and self._key(rest) in self._seen \ 194 ↛ 196line 194 didn't jump to line 196 because the condition on line 194 was never true

195 and not (self._in_code or self._in_edit): 

196 self.repeated = True 

197 return "" 

198 return rest

201def build_prompt(path: str, language: str, code: str, focus: str, 

202 diagnostics: list[tuple[int, str, str]], first_line: int = 1, total_lines: int = 0) -> str: 

203 """The user message: instructions, compiler messages and numbered code. 

204 ``code`` can be part of the file: its lines from ``first_line`` (of 

205 ``total_lines``), numbered as in the file.""" 

206 # Split only at newlines, like the editor: str.splitlines() would also 

207 # split at form feeds, U+2028 etc. and shift every later line number. 

208 lines = code.split("\n") 

209 if len(lines) > 1 and lines[-1] == "": 

210 lines.pop() # a final newline doesn't start another line 

211 last_line = first_line + len(lines) - 1 

212 width = len(str(last_line)) 

213 numbered = "\n".join(f"{i:>{width}}| {line}" for i, line in enumerate(lines, first_line)) 

214 parts = [f"Review this {language} file: {path}", FOCUS.get(focus, FOCUS["general"])] 

215 if total_lines and (first_line > 1 or last_line < total_lines): 

216 parts.append(f"Only lines {first_line}–{last_line} of the file's {total_lines} are shown: review " 

217 "only these. Don't report problems that may come from code that isn't shown " 

218 "(declarations elsewhere in the file, other procedures).") 

219 diagnostics = [d for d in diagnostics if first_line <= d[0] <= last_line] 

220 if diagnostics: 

221 notes = "\n".join(f"line {ln}: {sev}: {msg}" for ln, sev, msg in diagnostics[:40]) 

222 parts.append("The compiler already reported these messages in the last build, and they " 

223 "are shown to the user separately. Do NOT list them again; look for other " 

224 "problems the compiler can't see:\n" + notes) 

225 parts.append("Source (line numbers on the left):\n" + numbered) 

226 return "\n\n".join(parts)

291def context_for(tokens: int, cap: int = MAX_CONTEXT, answer: int = MAX_ANSWER_TOKENS) -> int: 

292 """A context window for ``tokens`` of conversation plus the answer: a 

293 power of two from 4096, at most ``cap``.""" 

294 size = 4096 

295 while size < tokens + answer and size < cap: 

296 size *= 2 

297 return min(size, cap)

438 # cap the answer so a runaway one always ends.

449 # also much faster. A model that can't think, or can't stop, says so:
