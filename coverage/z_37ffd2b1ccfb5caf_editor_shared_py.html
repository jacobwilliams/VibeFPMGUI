16def visual_column(text: str, width: int) -> int: 

17 """How far across a line's ``text`` reaches, with tabs to the next 

18 multiple of ``width``.""" 

19 column = 0 

20 for ch in text: 

21 column = (column // width + 1) * width if ch == "\t" else column + 1 

22 return column

24def tab_text(before: str, width: int, use_tabs: bool) -> str: 

25 """What Tab types after ``before`` (the line up to the cursor): a tab, 

26 or spaces to the next tab stop.""" 

27 if use_tabs: 

28 return "\t" 

29 return " " * (width - visual_column(before, width) % width)

31def outdent_length(line: str, width: int) -> int: 

32 """How many characters at the start of a line ⇧Tab removes: one tab, or 

33 spaces up to one indent level (or both, up to that width).""" 

34 spaces = min(len(line) - len(line.lstrip(" ")), width) 

35 if spaces < width and line[spaces:spaces + 1] == "\t": 

36 return spaces + 1 # (a tab, after any spaces short of a whole level) 

37 return spaces

39def backspace_length(before: str, width: int) -> int: 

40 """In leading spaces, Backspace removes back to the previous tab stop: 

41 how many spaces (0: an ordinary Backspace).""" 

42 if not before or before.strip(" "): 

43 return 0 # (not just spaces before the cursor) 

44 return (len(before) - 1) % width + 1

96def line_changes(opened: str, saved: str, current: str) -> tuple[dict[int, str], dict[int, tuple[str, int]]]: 

97 """Which lines of ``current`` differ from the saved text ("unsaved") or, 

98 if not, from the text as opened ("saved"); and where lines were deleted: 

99 {line: (kind, count)}, meaning removed just above that line (line count 

100 + 1: at the end). Lines are 1-based.""" 

101 lines = current.split("\n") 

102 if len(lines) > CHANGE_DIFF_MAX_LINES: 102 ↛ 103line 102 didn't jump to line 103 because the condition on line 102 was never true

103 return {}, {}

105 def diff(base: str): 

106 changed: set[int] = set() 

107 deleted: dict[int, int] = {} 

108 old = base.split("\n") 

109 # Only the middle differs (edits are usually in one place): skip the 

110 # lines that are the same at the start and the end before diffing. 

111 top, limit = 0, min(len(old), len(lines)) 

112 while top < limit and old[top] == lines[top]: 

113 top += 1 

114 bottom = 0 

115 while bottom < limit - top and old[-1 - bottom] == lines[-1 - bottom]: 

116 bottom += 1 

117 matcher = difflib.SequenceMatcher(None, old[top:len(old) - bottom], lines[top:len(lines) - bottom], 

118 autojunk=False) 

119 for tag, i1, i2, j1, j2 in matcher.get_opcodes(): 

120 if tag in ("replace", "insert"): 

121 changed.update(range(top + j1 + 1, top + j2 + 1)) 

122 elif tag == "delete": 

123 deleted[top + j1 + 1] = i2 - i1 

124 return changed, deleted 

125 

126 unsaved, unsaved_deleted = diff(saved) 

127 since_open, open_deleted = (unsaved, unsaved_deleted) if opened == saved else diff(opened) 

128 marks = {line: "saved" for line in since_open - unsaved} 

129 marks.update({line: "unsaved" for line in unsaved}) 

130 deletions = {line: ("saved", n) for line, n in open_deleted.items() if line not in unsaved_deleted} 

131 deletions.update({line: ("unsaved", n) for line, n in unsaved_deleted.items()}) 

132 return marks, deletions

135def occurrence_color(background: QColor) -> QColor: 

136 """The highlight of the selection's other occurrences on a syntax 

137 style's background: a blue mixed into it until it's plainly lighter (on 

138 a dark background) or darker (on a light one). The app's selection color 

139 was all but invisible on some styles (github-dark's near-black).""" 

140 tint = QColor(90, 150, 255) 

141 dark = background.lightness() < 128 

142 wanted = OCCURRENCE_CONTRAST[dark] 

143 color = QColor(background) 

144 for amount in (0.25, 0.35, 0.45, 0.55, 0.65, 0.75): 144 ↛ 150line 144 didn't jump to line 150 because the loop on line 144 didn't complete

145 color = QColor(round(background.red() + (tint.red() - background.red()) * amount), 

146 round(background.green() + (tint.green() - background.green()) * amount), 

147 round(background.blue() + (tint.blue() - background.blue()) * amount)) 

148 if abs(color.lightness() - background.lightness()) >= wanted: 

149 break 

150 return color

160def word_start(text: str, column: int) -> int: 

161 """Where the name ending at ``column`` of a line starts (``column`` if 

162 there's none just before it).""" 

163 start = column 

164 while start > 0 and WORD_RE.match(text[start - 1]): 

165 start -= 1 

166 return start

169def word_bounds(text: str, column: int) -> tuple[int, int] | None: 

170 """(start, end) of the name at ``column`` of a line, or just before it 

171 (the cursor right after a name); None if there's none there, or it's a 

172 number.""" 

173 i = column 

174 if not (i < len(text) and WORD_RE.match(text[i])): 

175 if i > 0 and WORD_RE.match(text[i - 1]): 

176 i -= 1 

177 else: 

178 return None 

179 start, end = word_start(text, i), i 

180 while end < len(text) and WORD_RE.match(text[end]): 

181 end += 1 

182 return None if text[start].isdigit() else (start, end)
