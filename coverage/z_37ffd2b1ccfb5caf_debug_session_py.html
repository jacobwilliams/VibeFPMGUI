1"""A debug session: a program run under the debugger (`vibefpmgui.dap`'s 

2client, with an adapter from `vibefpmgui.debug_adapters`), and what it's 

3doing: running, stopped (where, with its call stack and variables), or ended. 

4 

5The order of things, as adapters expect it: ``initialize``, then ``launch`` 

6(gdb answers it only after the configuration is done), and on the 

7``initialized`` event the breakpoints, then ``configurationDone``. At each 

8stop: the threads, the stopped thread's call stack, then the scopes (Locals, 

9Arguments, module variables…) of the frame shown; variables are asked for 

10as they're expanded. 

11"""

174 def _breakpoints_set(self, path: str, lines: list[int], success: bool, body: dict, message: str): 

175 states: dict[int, BreakpointState] = {} 

176 found = body.get("breakpoints") or [] if success else [] 

177 self._breakpoints.pending -= {(path, n) for n in self._breakpoints.lines.get(path, [])} 

178 for line, answer in zip(lines, found, strict=False): 

179 if isinstance(answer.get("id"), int): 179 ↛ 181line 179 didn't jump to line 181 because the condition on line 179 was always true

180 self._breakpoints.ids[answer["id"]] = (path, line) 

181 if not answer.get("verified") and answer.get("reason") == "pending": 

182 self._breakpoints.pending.add((path, line)) 

183 states[line] = BreakpointState(_placed(answer), _why_not(answer)) 

184 for line in lines[len(found):]: 184 ↛ 185line 184 didn't jump to line 185 because the loop on line 184 never started

185 states[line] = BreakpointState(False, message or "the debugger didn't place it") 

186 self.breakpoints_checked.emit(path, states)

233 def _event(self, name: str, body: dict): 

234 if name == "initialized": 

235 self._configure() 

236 elif name == "stopped": 

237 self.thread_id = body.get("threadId", self.thread_id) 

238 # (Which breakpoints, as they were set: the debugger may have moved one to 

239 # the next line with code, where the program stops.) 

240 ids = self._breakpoints.ids 

241 self.hit_breakpoints = [ids[n] for n in body.get("hitBreakpointIds") or [] if n in ids] 

242 self._set_state(STOPPED) 

243 self._unplaced() 

244 why = str(body.get("reason") or "") 

245 self.client.request("threads", {}, lambda ok, b, m: self._threads(ok, b, why)) 

246 elif name == "continued": 

247 if self.state == STOPPED: 247 ↛ 248line 247 didn't jump to line 248 because the condition on line 247 was never true

248 self.frames = [] 

249 self._set_state(RUNNING) 

250 elif name == "output": 

251 self._output(body) 

252 elif name == "breakpoint": 

253 self._breakpoint_event(body) 

254 elif name == "exited":
