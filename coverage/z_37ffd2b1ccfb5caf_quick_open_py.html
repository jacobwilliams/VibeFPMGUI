1"""Go to File (⌘P / Ctrl+P): any file of the project, or of its dependencies, 

2found by typing part of its name, opened in the editor. 

3 

4The files are those the Project files tree lists (`ProjectTree.all_files`): 

5the ones ignored by git and the greyed-out ones only with its *Ignored* 

6ticked. What's typed matches the file's name first (its start, then 

7anywhere, then its letters in order: "dstat" finds demo_stats.f90), then 

8its path ("src/dem"); the project's own files before its dependencies',

37def _in_order(letters: str, text: str) -> bool: 

38 """Whether ``letters`` appear in ``text`` in that order (not together).""" 

39 found = 0 

40 for ch in text: 

41 if found < len(letters) and ch == letters[found]: 

42 found += 1 

43 return found == len(letters)

46def score(query: str, item: ProjectFile) -> int | None: 

47 """How well a file matches what's typed (None: not at all).""" 

48 typed = "".join(query.lower().split()) 

49 name, path = PurePosixPath(item.relative).name.lower(), item.relative.lower() 

50 if name.startswith(typed): 

51 return 1000 

52 if typed in name: 

53 return 800 

54 if _in_order(typed, name): 

55 return 600 

56 if typed in path: 

57 return 400 

58 if _in_order(typed, path): 58 ↛ 59line 58 didn't jump to line 59 because the condition on line 58 was never true

59 return 200 

60 return None

63def ranked(query: str, files: list[ProjectFile], open_paths: list[str] | None = None) -> list[ProjectFile]: 

64 """The files matching ``query``, best first (see the module's description).""" 

65 open_order = {path: n for n, path in enumerate(open_paths or [])} 

66 if not query.strip(): 

67 return sorted(files, key=lambda f: (open_order.get(f.path, len(open_order)), f.dependency, f.package, 

68 f.relative.lower())) 

69 matched = [] 

70 for item in files: 

71 points = score(query, item) 

72 if points is not None: 

73 matched.append((-points, item.dependency, open_order.get(item.path, len(open_order)), 

74 len(item.relative), item.relative.lower(), item)) 

75 matched.sort(key=lambda m: m[:5]) 

76 return [m[-1] for m in matched]
