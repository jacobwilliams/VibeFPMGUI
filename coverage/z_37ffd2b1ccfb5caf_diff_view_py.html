1"""Showing two versions of a file: side by side, or as one unified list. 

2 

3`align` lines the versions up row by row (difflib), and `with_context` 

4keeps a few unchanged lines around each change, folding the rest into a 

5"⋯ N" row. `DiffView` draws the rows: each side in a syntax-colored pane 

6with its own line numbers, removed lines tinted red and added ones green, 

7and the two sides scrolling together.

27SAME, CHANGED, REMOVED, ADDED, GAP = "same", "changed", "removed", "added", "gap"

50def align(old: list[str], new: list[str]) -> list[Row]: 

51 """Every line of both versions, lined up: unchanged lines side by side, 

52 edited ones paired (CHANGED), the rest REMOVED or ADDED.""" 

53 rows: list[Row] = [] 

54 matcher = difflib.SequenceMatcher(None, old, new, autojunk=False) 

55 for op, a0, a1, b0, b1 in matcher.get_opcodes(): 

56 if op == "equal": 

57 rows += [Row(SAME, a0 + i + 1, old[a0 + i], b0 + i + 1, new[b0 + i]) for i in range(a1 - a0)] 

58 continue 

59 paired = min(a1 - a0, b1 - b0) if op == "replace" else 0 

60 rows += [Row(CHANGED, a0 + i + 1, old[a0 + i], b0 + i + 1, new[b0 + i]) for i in range(paired)] 

61 rows += [Row(REMOVED, a, old[a - 1], None, None) for a in range(a0 + paired + 1, a1 + 1)] 

62 rows += [Row(ADDED, None, None, b, new[b - 1]) for b in range(b0 + paired + 1, b1 + 1)] 

63 return rows

66def with_context(rows: list[Row], context: int = CONTEXT) -> list[Row]: 

67 """The rows near a change (``context`` unchanged lines around each), the 

68 other unchanged ones folded into GAP rows.""" 

69 near = [False] * len(rows) 

70 for i, row in enumerate(rows): 

71 if row.kind != SAME: 

72 for j in range(max(0, i - context), min(len(rows), i + context + 1)): 

73 near[j] = True 

74 out: list[Row] = [] 

75 i = 0 

76 while i < len(rows): 

77 if near[i]: 

78 out.append(rows[i]) 

79 i += 1 

80 continue 

81 j = i 

82 while j < len(rows) and not near[j]: 

83 j += 1 

84 out.append(Row(GAP, None, None, None, None, j - i)) 

85 i = j 

86 return out

89class PaneRows(NamedTuple):

97def side_by_side(rows: list[Row]) -> tuple[PaneRows, PaneRows]: 

98 """The older and the newer version's panes: a row each per Row.""" 

99 old, new = PaneRows([], [], [], []), PaneRows([], [], [], []) 

100 for row in rows: 

101 for pane, number, line, kind, mark in ((old, row.old_no, row.old, REMOVED, "-"), 

102 (new, row.new_no, row.new, ADDED, "+")): 

103 pane.new_lines.append(row.new_no) 

104 if row.kind == GAP: 

105 pane.text.append("") 

106 pane.gutter.append((f"⋯ {row.hidden:,}", "")) 

107 pane.kinds.append(GAP) 

108 continue 

109 changed = row.kind in (CHANGED, kind) 

110 pane.text.append(line if line is not None else "") 

111 pane.gutter.append((str(number) if number else "", mark if changed else "")) 

112 pane.kinds.append(kind if changed else FILLER if number is None else SAME) 

113 return old, new

116def unified(rows: list[Row]) -> PaneRows: 

117 """One pane: unchanged lines once; within each run of changes, the 

118 removed lines and then the added ones.""" 

119 pane = PaneRows([], [], [], []) 

120 removed: list[Row] = [] 

121 added: list[Row] = [] 

122 

123 def flush(): 

124 for row in removed: 

125 pane.text.append(row.old) 

126 pane.gutter.append((f"{row.old_no:>5} {'':>5}", "-")) 

127 pane.kinds.append(REMOVED) 

128 pane.new_lines.append(None) 

129 for row in added: 

130 pane.text.append(row.new) 

131 pane.gutter.append((f"{'':>5} {row.new_no:>5}", "+")) 

132 pane.kinds.append(ADDED) 

133 pane.new_lines.append(row.new_no) 

134 removed.clear() 

135 added.clear() 

136 

137 for row in rows: 

138 if row.kind in (REMOVED, CHANGED): 

139 removed.append(row) 

140 if row.kind in (ADDED, CHANGED): 

141 added.append(row) 

142 if row.kind in (SAME, GAP): 

143 flush() 

144 pane.text.append(row.new if row.kind == SAME else "") 

145 pane.gutter.append((f"{row.old_no:>5} {row.new_no:>5}", "") if row.kind == SAME 

146 else (f"⋯ {row.hidden:,}", "")) 

147 pane.kinds.append(row.kind) 

148 pane.new_lines.append(row.new_no) 

149 flush() 

150 return pane
