1"""A small client for the Fortran language server, fortls. 

2 

3`fortls <https://fortls.fortran-lang.org>`_ indexes the project's Fortran 

4sources and answers questions about them: what a name is (hover), where it 

5is defined, the outline of a file and the symbols in the whole project. It 

6runs as a background process speaking the Language Server Protocol (JSON 

7messages over stdin/stdout); `LspClient` handles that with a QProcess, 

8so the editor never waits for it. 

9 

10Open editor tabs send their text to the server (``didOpen``/``didChange``), 

11so unsaved edits are seen. Unless the project has its own fortls settings 

12(``.fortlsrc``, ``.fortls.json`` or ``.fortls``), fortls is told to index 

13just the fpm folders (``src``, ``app``, ``test``, ``example``, ``include`` and 

14any ``source-dir``) of the project and each downloaded dependency, rather 

15than every folder under the project (a Python environment, say). 

16"""

277def _clean(text: str) -> str: 

278 # "-----" between the declaration and a doc comment: just a gap (the code 

279 # panel already sets them apart; markdown would make the line above a heading). 

280 text = re.sub(r"^[ \t]*-{3,}[ \t]*$", "\n", text, flags=re.MULTILINE) 

281 # Qt shows _word_ underlined (like a link); the manual pages mean italics. 

282 parts = re.split(r"(```.*?```|`[^`]*`)", text, flags=re.DOTALL) 

283 return "".join(part if i % 2 else re.sub(r"(?<![\w\\*])_([^_\n]+?)_(?!\w)", r"*\1*", part) 

284 for i, part in enumerate(parts))

293def hover_info(contents, style: str | None = None) -> HoverInfo | None: 

294 """fortls's hover answer, taken apart for the hover card (None if empty)."""

297 return None

300 rest = text[match.end():]

304 category = " · ".join(p.strip() for p in (line.group(1) or "").split(":") if p.strip()) 

305 summary = line.group(2).strip() 

306 rest = rest[line.end():] 

307 return HoverInfo("intrinsic", category, summary, markdown_html(_clean(rest), style))
