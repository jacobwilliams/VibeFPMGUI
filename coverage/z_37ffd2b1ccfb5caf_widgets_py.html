22class HistoryLineEdit(QLineEdit): 

23 """A field for commands, as a shell's: Enter sends its text 

24 (``submitted``), remembers it, and clears the field; Up and Down go 

25 through what was sent (Down past the last one: what was being typed). 

26 The history is this field's own, for as long as it lives (not saved).""" 

27 

28 submitted = Signal(str) 

29 

30 def __init__(self, parent=None): 

31 super().__init__(parent) 

32 self.history: list[str] = [] 

33 self._index = 0 # (in history; len(history): the text being typed, not sent yet) 

34 self._draft = "" 

35 self.returnPressed.connect(self._submit) 

36 

37 def _submit(self): 

38 text = self.text().strip() 

39 if not text: 

40 return 

41 if not self.history or self.history[-1] != text: # (the same twice in a row: kept once) 

42 self.history.append(text) 

43 self._index, self._draft = len(self.history), "" 

44 self.clear() 

45 self.submitted.emit(text) 

46 

47 def keyPressEvent(self, event): 

48 key = event.key() 

49 if key == Qt.Key_Up and self._index > 0: 

50 if self._index == len(self.history): 

51 self._draft = self.text() 

52 self._index -= 1 

53 self.setText(self.history[self._index]) 

54 elif key == Qt.Key_Down and self._index < len(self.history): 

55 self._index += 1 

56 self.setText(self._draft if self._index == len(self.history) else self.history[self._index]) 

57 elif key in (Qt.Key_Up, Qt.Key_Down): 

58 pass # (no earlier, or later, entry) 

59 else: 

60 super().keyPressEvent(event)
