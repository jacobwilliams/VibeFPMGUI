1"""Clickable file locations in the Output panel. 

2 

3Runtime errors and compiler messages name where they happened: 

4gfortran's "At line 1753 of file build/dependencies/slsqp/src/slsqp_core.f90", 

5its backtraces' "… at src/foo.f90:12", compiler messages' "src/foo.f90:12:5:", 

6and ifx's traceback rows ("test 00007FF7010C7F25 ROUTINE 1266 x.f90"). 

7`OutputLinks` underlines those whose file exists (relative paths are the 

8project's, where fpm runs; a bare file name, as ifx prints, is looked for in 

9the project and its dependencies) and opens the file at that line when one 

10is clicked.

25_EXTENSIONS = r"f|for|ftn|f77|f90|f95|f03|f08|f18|fpp|fypp|inc|c|h|cc|cpp|cxx|hpp" 

26LOCATION_RES = [ 

27 # gfortran at run time: "At line 1753 of file src/x.f90" (then maybe " (unit = 6, …)"). 

28 re.compile(r"At line (?P<line>\d+) of file (?P<path>[^\s()]+)"), 

29 # Compiler messages and backtraces: "src/x.f90:12:5:", "at /abs/x.f90:12", "x.f90(12)" (Intel). 

30 re.compile(r"(?P<path>(?:[A-Za-z]:[\\/])?[^\s:'\"()<>]*\.(?i:" + _EXTENSIONS + r"))" 

31 r"(?::(?P<line2>\d+)(?::(?P<col>\d+))?|\((?P<line3>\d+)\))"), 

32 # ifx's traceback (image, PC, routine, line, source): "x 00007FF7010C7F25 ROUTINE 1266 x.f90". 

33 # The link is the line and the source. 

34 re.compile(r"^\s*\S+\s+[0-9A-Fa-f]{8,16}\s+\S+\s+(?P<link>(?P<line>\d+)\s+" 

35 r"(?P<path>[^\s:'\"()<>]+\.(?i:" + _EXTENSIONS + r")))\s*$"),

130 def locations(self, text: str) -> list[Location]: 

131 """The locations in one line of output whose files exist.""" 

132 if "." not in text: # (every location names a file: name.ext) 

133 return [] 

134 found: list[Location] = [] 

135 for pattern in LOCATION_RES: 

136 for match in pattern.finditer(text): 

137 if any(loc.start < match.end() and match.start() < loc.end for loc in found): 137 ↛ 138line 137 didn't jump to line 138 because the condition on line 137 was never true

138 continue # (already part of one) 

139 path = self.resolve(match.group("path")) 

140 if path is None: 

141 continue 

142 groups = match.groupdict() 

143 line = int(groups.get("line") or groups.get("line2") or groups.get("line3") or 1) 

144 column = max(int(groups.get("col") or 1) - 1, 0) 

145 start, end = match.span("link") if groups.get("link") else match.span() 

146 found.append(Location(start, end, path, line, column)) 

147 return found
