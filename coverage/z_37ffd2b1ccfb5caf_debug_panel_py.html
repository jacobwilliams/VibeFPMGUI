145 def _stopped(self, frames: list[Frame], _why: str): 

146 self.stack.blockSignals(True) 

147 self.stack.clear() 

148 for frame in frames: 

149 where = f"{frame.path.rsplit('/', 1)[-1]}:{frame.line}" if frame.path else "(no source)" 

150 item = QListWidgetItem(f"{frame_name(frame.name)} · {where}") 

151 item.setToolTip(f"{frame.path}:{frame.line}" if frame.path else frame.name) 

152 if not frame.path: 

153 item.setForeground(self.palette().placeholderText()) 

154 self.stack.addItem(item) 

155 self.stack.setCurrentRow(self.session.frame_index) 

156 self.stack.blockSignals(False)

162 # ---------------------------------------------------------- the variables 

163 def _scopes(self, _index: int, scopes: list[Scope]): 

164 self.variables.clear() 

165 for number, scope in enumerate(scopes): 

166 item = QTreeWidgetItem([scope.name, ""]) 

167 item.setData(0, REFERENCE_ROLE, scope.reference) 

168 font = item.font(0) 

169 font.setBold(True) 

170 item.setFont(0, font) 

171 self.variables.addTopLevelItem(item) 

172 item.addChild(QTreeWidgetItem([PLACEHOLDER, ""])) 

173 if number == 0 or not scope.expensive and scope.name.lower() in ("locals", "arguments"): 

174 item.setExpanded(True) 

175 

176 def _expanded(self, item: QTreeWidgetItem): 

177 if item.data(0, LOADED_ROLE): 177 ↛ 178line 177 didn't jump to line 178 because the condition on line 177 was never true

178 return 

179 item.setData(0, LOADED_ROLE, True) 

180 reference = int(item.data(0, REFERENCE_ROLE) or 0) 

181 self.session.variables(reference, lambda variables, more, it=item: self._fill(it, variables, more)) 

182 

183 def _fill(self, item: QTreeWidgetItem, variables: list[Variable], more: bool): 

184 try: 

185 item.takeChildren() 

186 except RuntimeError: 

187 return # (cleared meanwhile: the program moved on) 

188 for variable in variables: 

189 child = QTreeWidgetItem([variable.name, variable.value]) 

190 child.setToolTip(1, variable.value) 

191 if variable.type: 

192 child.setToolTip(0, variable.type) 

193 child.setData(0, REFERENCE_ROLE, variable.reference) 

194 if variable.reference: 

195 child.addChild(QTreeWidgetItem([PLACEHOLDER, ""])) 

196 item.addChild(child) 

197 if more: 197 ↛ 198line 197 didn't jump to line 198 because the condition on line 197 was never true

198 item.addChild(QTreeWidgetItem(["…", "more: evaluate a section, e.g. a(1001:1100)"])) 

199 if not variables: 

200 item.addChild(QTreeWidgetItem(["(none)", ""]))

202 def variable_rows(self) -> list[tuple[str, str]]: 

203 """(name, value) of the variables shown, depth first (for the tests).""" 

204 rows = [] 

205 

206 def visit(item: QTreeWidgetItem): 

207 for i in range(item.childCount()): 

208 child = item.child(i) 

209 if child.text(0) != PLACEHOLDER: 

210 rows.append((child.text(0), child.text(1))) 

211 visit(child) 

212 

213 for i in range(self.variables.topLevelItemCount()): 

214 visit(self.variables.topLevelItem(i)) 

215 return rows

218 def _evaluate(self, expression: str): 

219 self.append("console", f"> {expression}\n") 

220 self.session.evaluate(expression, lambda ok, text, _ref: self.append("console", f"{text}\n"))
