15PROGRESS_RE = re.compile(r"^\[\s*(\d{1,3})%\]")

18def progress_color(percent: int) -> QColor: 

19 """Blue at 0%, green at 100% (through teal), bright enough for white text.""" 

20 fraction = max(0, min(100, percent)) / 100 

21 return QColor.fromHsvF(0.60 - 0.24 * fraction, 0.70, 0.78)

29 out = [] 

30 block = self.firstVisibleBlock() 

31 offset = self.contentOffset() 

32 height = self.viewport().height() 

33 while block.isValid(): 

34 geometry = self.blockBoundingGeometry(block).translated(offset) 

35 if geometry.top() > height: 35 ↛ 36line 35 didn't jump to line 36 because the condition on line 35 was never true

36 break 

37 match = PROGRESS_RE.match(block.text()) 

38 if match and block.isVisible(): 

39 start, end = QTextCursor(block), QTextCursor(block) 

40 start.setPosition(block.position() + match.start()) 

41 end.setPosition(block.position() + match.end()) 

42 left, right = self.cursorRect(start), self.cursorRect(end) 

43 if left.top() == right.top(): # (not wrapped in the middle) 43 ↛ 46line 43 didn't jump to line 46 because the condition on line 43 was always true

44 out.append((int(match.group(1)), QRectF(left.left(), left.top() + 1, right.left() - left.left(),

46 block = block.next()
